# 倍增并查集（区间合并）

长度固定 n>=0；第 k 层覆盖长度 2^k 的合法区间，两段起点 x,y 须满足 x+2^k<=n、y+2^k<=n。 下标从 0 编号；mergeRange(k,x,y) 合并 [x,x+2^k) 与 [y,y+2^k) 的对应位置。

每次底层成功合并后调用 callback(a,b)，a,b 是合并前两个集合的代表，不能假定哪一个成为新根。

find/merge 为对应层的反阿克曼均摊；mergeRange 最多 O(2^k alpha(n)) 次基础合并。

```cpp
class RangeDSU {
    vector<vector<int>> p, sz;

  public:
    // 初始化所有可容纳长度 2^k 区间的层；空数组合法。
    explicit RangeDSU(int n) {
        assert(n >= 0);
        for (int len = 1; len <= max(1, n); len <<= 1) {
            int m = n - len + 1;
            if (m < 0)
                break;
            p.emplace_back(m);
            sz.emplace_back(m, 1);
            iota(p.back().begin(), p.back().end(), 0);
            if (len > (1 << 29))
                break;
        }
    }
    // 返回第 k 层位置 x 所属块的代表。
    int find(int k, int x) {
        assert(0 <= k && k < int(p.size()) && 0 <= x && x < int(p[k].size()));
        return p[k][x] == x ? x : p[k][x] = find(k, p[k][x]);
    }

  private:
    // 合并第 k 层两个长度 2^k 区间的代表；成功返回 true。
    bool merge(int k, int x, int y) {
        x = find(k, x);
        y = find(k, y);
        if (x == y)
            return false;
        if (sz[k][x] < sz[k][y])
            swap(x, y);
        p[k][y] = x;
        sz[k][x] += sz[k][y];
        return true;
    }

  public:
    // 把两个长度 2^k 区间逐层对应合并；每次底层合并成功后调用 callback。
    template <class F> void mergeRange(int k, int x, int y, F callback) {
        assert(0 <= k && k < int(p.size()) && x >= 0 && y >= 0 &&
               x + (1 << k) <= int(p[0].size()) && y + (1 << k) <= int(p[0].size()));
        if (find(k, x) == find(k, y))
            return;
        if (k == 0) {
            int a = find(0, x), b = find(0, y);
            if (merge(0, a, b))
                callback(a, b);
            return;
        }
        mergeRange(k - 1, x, y, callback);
        mergeRange(k - 1, x + (1 << (k - 1)), y + (1 << (k - 1)), callback);
        merge(k, x, y);
    }
};
```
