# 可并权值线段树（合并与分裂）

值域 [0,U)，U>0；各根独占节点，必须来自同一 pool；不能共享节点或用作持久化树。 每个值频次非负，各根总计数及合并后总计数须在 i64 范围内。 根 0 是空集合；值域区间 [l,r)（含左端，不含右端），0<=l<=r<=U；k 从 0 起，重复副本分别计名次。

add 按引用更新根；merge/split 消费输入根，须接收新根并弃用旧根。

单点、计数、kth O(log U)；split O(log U) 新节点；merge O(两树重叠节点数)，空间 O(分配节点数)。

```cpp
class MergeableCountTree {
    struct Node {
        int l = 0, r = 0;
        i64 sum = 0;
    };
    int U;
    vector<Node> t{Node{}};
    int alloc() {
        t.push_back({});
        return t.size() - 1;
    }
    void pull(int p) {
        t[p].sum = t[t[p].l].sum + t[t[p].r].sum;
    }
    int update(int p, int l, int r, int x, i64 d) {
        if (!p)
            p = alloc();
        if (r - l == 1) {
            assert(i128(t[p].sum) + d >= 0 && i128(t[p].sum) + d <= LLONG_MAX);
            t[p].sum += d;
            return p;
        }
        int m = l + (r - l) / 2;
        if (x < m)
            t[p].l = update(t[p].l, l, m, x, d);
        else
            t[p].r = update(t[p].r, m, r, x, d);
        pull(p);
        return p;
    }
    i64 query(int p, int l, int r, int a, int b) const {
        if (!p || b <= l || r <= a)
            return 0;
        if (a <= l && r <= b)
            return t[p].sum;
        int m = l + (r - l) / 2;
        return query(t[p].l, l, m, a, b) + query(t[p].r, m, r, a, b);
    }
    int unite(int a, int b, int l, int r) {
        if (!a || !b)
            return a ? a : b;
        if (r - l == 1) {
            t[a].sum += t[b].sum;
            return a;
        }
        int m = l + (r - l) / 2;
        t[a].l = unite(t[a].l, t[b].l, l, m);
        t[a].r = unite(t[a].r, t[b].r, m, r);
        pull(a);
        return a;
    }
    pair<int, int> partition(int p, int l, int r, int a, int b) {
        if (!p || b <= l || r <= a)
            return {0, p};
        if (a <= l && r <= b)
            return {p, 0};
        int m = l + (r - l) / 2;
        auto [il, ol] = partition(t[p].l, l, m, a, b);
        auto [ir, orr] = partition(t[p].r, m, r, a, b);
        int q = alloc();
        t[p].l = il;
        t[p].r = ir;
        pull(p);
        t[q].l = ol;
        t[q].r = orr;
        pull(q);
        return {p, q};
    }

  public:
    // 建立公共值域 [0,U) 的节点池；U>0。
    explicit MergeableCountTree(int universe) : U(universe) {
        assert(U > 0);
    }
    // 返回此根的元素总数；根 0 返回 0。
    i64 size(int p) const {
        assert(0 <= p && p < int(t.size()));
        return t[p].sum;
    }
    // 给合法值 x 增减副本数量，更新 root；最终每个值频次非负，无返回值。
    void add(int &root, int x, i64 delta) {
        assert(0 <= x && x < U);
        size(root);
        root = update(root, 0, U, x, delta);
    }
    // 返回值域 [l,r) 的副本总数；空区间返回 0。
    i64 count(int root, int l, int r) const {
        assert(0 <= l && l <= r && r <= U);
        size(root);
        return query(root, 0, U, l, r);
    }
    // 返回第 k 小原值；k 从 0 开始。
    int kth(int root, i64 k) const {
        assert(0 <= k && k < size(root));
        int l = 0, r = U;
        while (r - l > 1) {
            int m = l + (r - l) / 2;
            i64 left = t[t[root].l].sum;
            if (k < left) {
                root = t[root].l;
                r = m;
            } else {
                k -= left;
                root = t[root].r;
                l = m;
            }
        }
        return l;
    }
    // 合并独立根 a,b，返回合并根；调用后必须弃用两个输入根。
    int merge(int a, int b) {
        size(a);
        size(b);
        assert(!a || a != b);
        return unite(a, b, 0, U);
    }
    // 返回 [l,r) 内、外两个根；输入根从此作废。
    pair<int, int> split(int root, int l, int r) {
        assert(0 <= l && l <= r && r <= U);
        size(root);
        return partition(root, 0, U, l, r);
    }
    // ponytail: 被合并的节点不回收，空间随历史分配增长；大量反复合并时增加空闲节点栈。
};
```
