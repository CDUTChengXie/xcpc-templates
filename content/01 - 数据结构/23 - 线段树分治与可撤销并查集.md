# 线段树分治与可撤销并查集

n>=0，无向边与询问先给出；本文件自带 RollbackDSU，勿再贴 02 的同名结构。 顶点 0..n-1；询问 {t,u,v} 表示加入 edges[0..t] 后，已经加入 t+1 条边。 有边时 t 会夹到 [0,边数-1]；无边时只有同点连通；结果按原询问顺序返回。

建树与回放各 O(m log m + q log m)，可撤销并查集按大小合并且不压缩路径， 每次操作 O(log n)，总体 O((m log m + q) log n)；空间 O(n + m log m + q)。

```cpp
// ---------- 可撤销并查集 ----------
// 只合并、不压缩路径，支持快照与按快照回滚
class RollbackDSU {
  public:
    std::vector<int> p, sz;
    // 每次成功合并压一条 (child, parent, parentSizeBefore)，撤销时按顺序回退
    std::vector<std::array<int, 3>> hist;

    // 创建 n 个独立集合
    explicit RollbackDSU(int n) : p(n), sz(n, 1) {
        for (int i = 0; i < n; i++) {
            p[i] = i;
        }
    }

    // 返回 x 所在集合的代表；不做路径压缩，便于撤销
    int find(int x) const {
        while (p[x] != x) {
            x = p[x];
        }
        return x;
    }

    // 返回 a、b 是否属于同一集合
    bool same(int a, int b) const {
        return find(a) == find(b);
    }

    // 合并 a、b 所在集合；同集合无操作
    void unite(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b) {
            return;
        }
        if (sz[a] < sz[b]) {
            std::swap(a, b);
        }
        hist.push_back({b, a, sz[a]});
        p[b] = a;
        sz[a] += sz[b];
    }

    // 返回当前历史栈深，供 rollback 使用
    int snapshot() const {
        return (int)hist.size();
    }

    // 撤销到快照 snap 时的状态
    void rollback(int snap) {
        while ((int)hist.size() > snap) {
            auto [child, parent, parentSize] = hist.back();
            hist.pop_back();
            p[child] = child;
            sz[parent] = parentSize;
        }
    }
};

// ---------- 线段树分治 ----------
// queries[i] = (t, u, v)：返回「加入 edges[0..t] 之后 u 与 v 是否连通」。
// 返回每条询问的连通性（按原顺序）
std::vector<char> divideConquerConnectivity(int n, const std::vector<std::pair<int, int>> &edges,
                                            const std::vector<std::tuple<int, int, int>> &queries) {
    int q = (int)queries.size();
    if (q == 0) {
        return {};
    }
    int m = (int)edges.size();
    if (m == 0) {
        std::vector<char> res(q, false);
        for (int i = 0; i < q; i++) {
            auto [t, u, v] = queries[i];
            res[i] = (u == v); // 没有边时只有自己和自己连通
        }
        return res;
    }
    std::vector<std::vector<int>> tree(4 * m + 5);
    // 第 e 条边存在于时间区间 [e, m-1]，把它挂到线段树上
    std::function<void(int, int, int, int, int, int)> add = [&](int node, int l, int r, int ql,
                                                                int qr, int eid) {
        if (ql <= l && r <= qr) {
            tree[node].push_back(eid);
            return;
        }
        int mid = (l + r) / 2;
        if (ql <= mid) {
            add(node * 2, l, mid, ql, qr, eid);
        }
        if (qr > mid) {
            add(node * 2 + 1, mid + 1, r, ql, qr, eid);
        }
    };
    for (int e = 0; e < m; e++) {
        add(1, 0, m - 1, e, m - 1, e);
    }
    // 按时间把询问挂到叶子上
    std::vector<std::vector<int>> at(m);
    for (int i = 0; i < q; i++) {
        auto [t, u, v] = queries[i];
        t = std::min(std::max(t, 0), m - 1);
        at[t].push_back(i);
    }
    RollbackDSU dsu(n);
    std::vector<char> res(q, false);
    std::function<void(int, int, int)> dfs = [&](int node, int l, int r) {
        int snap = dsu.snapshot();
        for (int eid : tree[node]) {
            dsu.unite(edges[eid].first, edges[eid].second);
        }
        if (l == r) {
            for (int qi : at[l]) {
                auto [t, u, v] = queries[qi];
                res[qi] = dsu.same(u, v);
            }
        } else {
            int mid = (l + r) / 2;
            dfs(node * 2, l, mid);
            dfs(node * 2 + 1, mid + 1, r);
        }
        dsu.rollback(snap);
    };
    dfs(1, 0, m - 1);
    return res;
}

// 一般边生存区间：增加/删除边 + 连通性查询，复用本文件 RollbackDSU。
```

## 离线动态连通性（扩展）

依赖上方的 RollbackDSU。

所有操作预先给出；Remove 必须对应一条仍存在的同端点边，重复边逐条删除。

顶点 0-based [0,n)；操作按 vector 顺序执行；返回数组只含 Query 答案。

q 次操作 O(q log q log n)，空间 O(n+q log q)。

```cpp
// 一条离线操作：Add/Remove 端点 (u,v)，或 Query 查询 (u,v) 是否连通
struct DynamicConnectivityOp {
    enum Kind { Add, Remove, Query };
    Kind kind;
    int u, v;
};
// 返回全部 Query 的连通性；支持加删重边、自环，不修改输入。
vector<bool> offlineDynamicConnectivity(int n, const vector<DynamicConnectivityOp> &ops) {
    assert(n >= 0);
    int q = ops.size();
    if (!q)
        return {};
    vector<vector<pair<int, int>>> tree(4 * q);
    auto cover = [&](auto &&self, int p, int l, int r, int a, int b, pair<int, int> e) -> void {
        if (b <= l || r <= a)
            return;
        if (a <= l && r <= b) {
            tree[p].push_back(e);
            return;
        }
        int m = (l + r) / 2;
        self(self, p * 2, l, m, a, b, e);
        self(self, p * 2 + 1, m, r, a, b, e);
    };
    map<pair<int, int>, vector<int>> active;
    for (int i = 0; i < q; i++) {
        auto [kind, u, v] = ops[i];
        assert(0 <= u && u < n && 0 <= v && v < n);
        pair<int, int> e = {min(u, v), max(u, v)};
        if (kind == DynamicConnectivityOp::Add)
            active[e].push_back(i);
        else if (kind == DynamicConnectivityOp::Remove) {
            auto it = active.find(e);
            assert(it != active.end() && !it->second.empty());
            int begin = it->second.back();
            it->second.pop_back();
            cover(cover, 1, 0, q, begin, i, e);
        } else
            assert(kind == DynamicConnectivityOp::Query);
    }
    for (const auto &[e, times] : active)
        for (int begin : times)
            cover(cover, 1, 0, q, begin, q, e);
    RollbackDSU dsu(n);
    vector<bool> answer;
    auto visit = [&](auto &&self, int p, int l, int r) -> void {
        int snapshot = dsu.snapshot();
        for (auto [u, v] : tree[p])
            dsu.unite(u, v);
        if (r - l == 1) {
            if (ops[l].kind == DynamicConnectivityOp::Query)
                answer.push_back(dsu.same(ops[l].u, ops[l].v));
        } else {
            int m = (l + r) / 2;
            self(self, p * 2, l, m);
            self(self, p * 2 + 1, m, r);
        }
        dsu.rollback(snapshot);
    };
    visit(visit, 1, 0, q);
    return answer;
}
```
