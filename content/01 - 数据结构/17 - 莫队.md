# 莫队

所有询问先给出；a 为 `vector<int>`，长度 n，0<=a[i]<=n，超值域须先离散化。 下标从 0 编号，区间 [l,r)（含左端，不含右端），0<=l<=r<=n；Query 为 {l,r,id}，id 是 [0,询问数) 的排列。 空区间答案 0；work 每次重置计数，可重复调用；改统计量时需配套修改状态维护。

O((n + q) sqrt n) 次 add/del；空间 O(n+q)。

```cpp
// 基础莫队：分块排序询问并维护半开窗口的计数状态
struct Mo {
    struct Query {
        int l;
        int r;
        int id;
    };

    int n;
    int B;
    std::vector<Query> qs;

    // 用 n 与询问序列建莫队；分块大小取 sqrt(n)
    Mo(int n_, const std::vector<Query> &qs_) : n(n_), B(std::max(1, int(std::sqrt(n)))), qs(qs_) {}

    // ---- 状态维护：区间不同数个数 ----
    std::vector<int> cnt;
    int distinct = 0;

    // 分配计数表并把 distinct 归零
    void initState(const std::vector<int> &a) {
        cnt.assign(n + 1, 0);
        distinct = 0;
    }

    // 把位置 p 加入当前窗口
    void add(const std::vector<int> &a, int p) {
        if (cnt[a[p]]++ == 0) {
            distinct++;
        }
    }

    // 把位置 p 移出当前窗口
    void del(const std::vector<int> &a, int p) {
        if (--cnt[a[p]] == 0) {
            distinct--;
        }
    }

    // 返回当前窗口的统计值
    int answer() const {
        return distinct;
    }
    // --------------------------------

    // 按莫队排序处理全部询问，返回按 id 对应的答案
    std::vector<int> work(const std::vector<int> &a) {
        initState(a);
        int q = qs.size();
        std::vector<int> ans(q);
        std::sort(qs.begin(), qs.end(), [&](const Query &x, const Query &y) {
            int bx = x.l / B, by = y.l / B;
            if (bx != by) {
                return bx < by;
            }
            return bx % 2 ? x.r > y.r : x.r < y.r;
        });
        int l = 0, r = 0;
        for (auto &qu : qs) {
            while (l > qu.l) {
                add(a, --l);
            }
            while (r < qu.r) {
                add(a, r++);
            }
            while (l < qu.l) {
                del(a, l++);
            }
            while (r > qu.r) {
                del(a, --r);
            }
            ans[qu.id] = answer();
        }
        return ans;
    }
};
```
