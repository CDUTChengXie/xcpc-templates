# 二维区间加区间和（四系数树状数组）

n,m>=0，delta 不为 LLONG_MIN，系数乘积和所有中间和均不溢出 i64。 坐标 从 0 编号，矩形 [x1,x2)×[y1,y2)；空矩形合法。

修改、查询 O(log n log m)，空间 O(nm)，每个位置 4 个 i64。

```cpp
class RangeAddSum2D {
    int n, m;
    vector<vector<array<i64, 4>>> bit;
    void corner(int x, int y, i64 v) {
        if (x == n || y == m)
            return;
        array<i64, 4> values = {v, v * x, v * y, v * x * y};
        for (int i = x + 1; i <= n; i += i & -i)
            for (int j = y + 1; j <= m; j += j & -j)
                for (int k = 0; k < 4; k++)
                    bit[i][j][k] += values[k];
    }
    i64 prefix(int x, int y) const {
        array<i64, 4> s{};
        for (int i = x; i; i -= i & -i)
            for (int j = y; j; j -= j & -j)
                for (int k = 0; k < 4; k++)
                    s[k] += bit[i][j][k];
        return s[0] * x * y - s[1] * y - s[2] * x + s[3];
    }
    void check(int x1, int y1, int x2, int y2) const {
        assert(0 <= x1 && x1 <= x2 && x2 <= n && 0 <= y1 && y1 <= y2 && y2 <= m);
    }

  public:
    // 建立 n×m 全零矩阵；任一维为 0 合法。
    RangeAddSum2D(int rows, int cols) : n(rows), m(cols) {
        assert(n >= 0 && m >= 0);
        bit.resize(n + 1, vector<array<i64, 4>>(m + 1));
    }
    // 给半开矩形每项加 v；空矩形无操作，无返回值。
    void add(int x1, int y1, int x2, int y2, i64 v) {
        check(x1, y1, x2, y2);
        if (x1 == x2 || y1 == y2)
            return;
        corner(x1, y1, v);
        corner(x2, y1, -v);
        corner(x1, y2, -v);
        corner(x2, y2, v);
    }
    // 返回半开矩形和，空矩形返回 0。
    i64 sum(int x1, int y1, int x2, int y2) const {
        check(x1, y1, x2, y2);
        return prefix(x2, y2) - prefix(x1, y2) - prefix(x2, y1) + prefix(x1, y1);
    }
};
```
