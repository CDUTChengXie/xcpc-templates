# 区间加区间和（双树状数组）

长度预先确定，v 不为 LLONG_MIN，所有乘积、中间和不溢出 i64。 下标从 0 开始，修改和查询区间均为半开 [l,r)（含左端，不含右端）。

区间加、区间和 O(log n)；空间 O(n)。

```cpp
class RangeAddSum {
    struct BIT {
        vector<i64> t;
        void init(int n) {
            t.assign(n + 1, 0);
        }
        void add(int p, i64 v) {
            for (int i = p + 1; i < int(t.size()); i += i & -i)
                t[i] += v;
        }
        i64 sum(int p) const {
            i64 z = 0;
            for (int i = p; i; i -= i & -i)
                z += t[i];
            return z;
        }
    } a, b;
    int n = 0;

  public:
    // 建立长度 n 的全零数组；n>=0。
    explicit RangeAddSum(int n_ = 0) {
        init(n_);
    }
    // 清空并重设长度；无返回值。
    void init(int n_) {
        assert(n_ >= 0);
        n = n_;
        a.init(n);
        b.init(n);
    }

  private:
    void addPoint(int p, i64 v) {
        if (p < n) {
            a.add(p, v);
            b.add(p, v * p);
        }
    }

  public:
    // 给半开区间 [l,r) 每项加 v；空区间无操作。
    void rangeAdd(int l, int r, i64 v) {
        assert(0 <= l && l <= r && r <= n);
        if (l == r)
            return;
        addPoint(l, v);
        addPoint(r, -v);
    }
    // 返回 [0,r) 的和；r 会被限制在 [0,n]。
    i64 prefixSum(int r) const {
        r = clamp(r, 0, n);
        return r * a.sum(r) - b.sum(r);
    }
    // 返回 [l,r) 的和；空区间为 0。
    i64 rangeSum(int l, int r) const {
        assert(0 <= l && l <= r && r <= n);
        return prefixSum(r) - prefixSum(l);
    }
};
```
