# 线性基（带时间戳）

值为 u64，时间戳>=0；只插入不删除；时间戳相同的数均可选，空子集可选。 lowerTime 是可用时间下界，包含等于它的数；要查位置区间 [l,r]，先仅插入到 r 再以 l 为下界。 第 k 小按不同异或结果计，k 从 0 起，含空子集 0；维数 d<64 时须 k<2^d，d=64 时任意 u64 k 合法。

交集返回固定子空间，不保留原时间戳；每组新数据可重新创建 Basis。

add/query/最值 O(L)，kth/rankOf/intersection O(L²)，L=64；空间 O(L)。

```cpp
struct Basis {
    u64 a[64]{};
    int t[64]{};

    // 创建空时间戳线性基；无返回值。
    Basis() {
        std::fill(t, t + 64, -1);
    }

    // 插入 u64 值 x 及非负时间戳 y；无返回值。
    void add(u64 x, int y = 1E9) {
        assert(y >= 0);
        for (int i = 63; i >= 0; i--) {
            if (x >> i & 1) {
                if (y > t[i]) {
                    std::swap(a[i], x);
                    std::swap(t[i], y);
                }
                x ^= a[i];
            }
        }
    }

    // 返回 target 能否由时间戳 >=y 的数异或得到。
    bool query(u64 x, int y = 0) const {
        for (int i = 63; i >= 0; i--) {
            if ((x >> i & 1) && t[i] >= y) {
                x ^= a[i];
            }
        }
        return x == 0;
    }
    // 返回 seed 与时间戳 >=lowerTime 的子集异或的最大值；可取空子集。
    u64 queryMax(u64 seed = 0, int lowerTime = 0) const {
        for (int i = 63; i >= 0; i--)
            if (t[i] >= lowerTime)
                seed = std::max(seed, seed ^ a[i]);
        return seed;
    }
    // 返回 seed 与时间戳 >=lowerTime 的子集异或的最小值；可取空子集。
    u64 queryMin(u64 seed = 0, int lowerTime = 0) const {
        for (int i = 63; i >= 0; i--)
            if (t[i] >= lowerTime)
                seed = std::min(seed, seed ^ a[i]);
        return seed;
    }
    // 合并另一基的向量及时间戳，不改变 other；无返回值。
    void merge(const Basis &other) {
        for (int i = 0; i < 64; i++)
            if (other.t[i] >= 0)
                add(other.a[i], other.t[i]);
    }

  private:
    array<u64, 64> reduced(int lowerTime) const {
        array<u64, 64> rows{};
        for (int i = 0; i < 64; i++)
            if (t[i] >= lowerTime) {
                rows[i] = a[i];
                for (int j = i - 1; j >= 0; j--)
                    if ((rows[i] >> j) & 1)
                        rows[i] ^= rows[j];
            }
        return rows;
    }

  public:
    // 返回时间戳 >=lowerTime 子空间的维数，范围 0..64。
    int dimension(int lowerTime = 0) const {
        int n = 0;
        for (int i = 0; i < 64; i++)
            n += t[i] >= lowerTime;
        return n;
    }
    // 返回时间戳 >=lowerTime 子空间中第 k 小不同异或值，k 从 0 起，含空子集 0。
    u64 kth(u64 k, int lowerTime = 0) const {
        int dim = dimension(lowerTime);
        assert(dim == 64 || k < (1ULL << dim));
        auto rows = reduced(lowerTime);
        u64 result = 0;
        int bit = 0;
        for (int i = 0; i < 64; i++)
            if (rows[i]) {
                if ((k >> bit) & 1)
                    result ^= rows[i];
                bit++;
            }
        return result;
    }
    // 返回 x 在可表示异或值升序集合中的 0-based 排名；不在子空间返回 nullopt。
    optional<u64> rankOf(u64 x, int lowerTime = 0) const {
        auto rows = reduced(lowerTime);
        array<int, 64> index{};
        int next = 0;
        for (int i = 0; i < 64; i++)
            if (rows[i])
                index[i] = next++;
        u64 result = 0;
        for (int i = 63; i >= 0; i--)
            if ((x >> i) & 1) {
                if (!rows[i])
                    return nullopt;
                x ^= rows[i];
                result |= 1ULL << index[i];
            }
        return result;
    }
    // 返回两个指定时间下界子空间的交；结果是固定空间，原时间戳不保留。
    Basis intersection(const Basis &other, int lowerThis = 0, int lowerOther = 0) const {
        array<u64, 64> value{}, fromThis{};
        for (int i = 0; i < 64; i++)
            if (t[i] >= lowerThis) {
                value[i] = a[i];
                fromThis[i] = a[i];
            }
        Basis result;
        for (int i = 0; i < 64; i++)
            if (other.t[i] >= lowerOther) {
                u64 x = other.a[i], track = 0;
                for (int j = 63; j >= 0; j--)
                    if ((x >> j) & 1) {
                        if (value[j]) {
                            x ^= value[j];
                            track ^= fromThis[j];
                        } else {
                            value[j] = x;
                            fromThis[j] = track;
                            break;
                        }
                    }
                if (!x)
                    result.add(track);
            }
        return result;
    }
};
```
