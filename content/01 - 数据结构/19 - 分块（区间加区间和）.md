# 分块（区间加区间和）

n>=0，a 为 `vector<i64>`，为空表示 n 个零，否则至少 n 项；值、和及 delta*块长不溢出 i64。 下标从 0 编号，区间 [l,r)（含左端，不含右端），0<=l<=r<=n；p 须在 [0,n)。 空区间和为 0、加法无操作；每组新数据用 init(n,a) 重建，全零数组传 {}。

每次操作 O(sqrt n)；空间 O(n)。

```cpp
// 序列分块：整块打 tag，散块暴力
struct Block {
    int n;
    int len;
    std::vector<i64> a, tag, sum;
    std::vector<int> bel;

    // 创建空对象；使用前调用 init
    Block() {}
    // 用数组 a 构建 n_ 长度的块；a 为空表示全零
    Block(int n_, std::vector<i64> init_ = {}) {
        init(n_, init_);
    }

    // 替换数组并重置所有块，丢弃旧状态
    void init(int n_, const std::vector<i64> &init_) {
        n = n_;
        len = std::max(1, (int)std::sqrt(n) + 1);
        int m = (n + len - 1) / len;
        a.assign(n, 0);
        tag.assign(m, 0);
        sum.assign(m, 0);
        bel.assign(n, 0);
        for (int i = 0; i < n; i++) {
            bel[i] = i / len;
            a[i] = init_.empty() ? 0 : init_[i];
            sum[bel[i]] += a[i];
        }
    }

    // 给 [l,r) 每项加 i64 x；空区间无操作
    void addRange(int l, int r, i64 x) { // [l, r)
        if (l >= r) {
            return;
        }
        int bl = bel[l], br = bel[r - 1];
        if (bl == br) {
            for (int i = l; i < r; i++) {
                a[i] += x;
                sum[bl] += x;
            }
            return;
        }
        for (int i = l; i < (bl + 1) * len; i++) {
            a[i] += x;
            sum[bl] += x;
        }
        for (int b = bl + 1; b < br; b++) {
            tag[b] += x;
            sum[b] += x * len;
        }
        for (int i = br * len; i < r; i++) {
            a[i] += x;
            sum[br] += x;
        }
    }

    // 返回 [l,r) 区间和；空区间为 0
    i64 query(int l, int r) { // [l, r)
        if (l >= r) {
            return 0;
        }
        int bl = bel[l], br = bel[r - 1];
        if (bl == br) {
            i64 res = 0;
            for (int i = l; i < r; i++) {
                res += a[i] + tag[bl];
            }
            return res;
        }
        i64 res = 0;
        for (int i = l; i < (bl + 1) * len; i++) {
            res += a[i] + tag[bl];
        }
        for (int b = bl + 1; b < br; b++) {
            res += sum[b];
        }
        for (int i = br * len; i < r; i++) {
            res += a[i] + tag[br];
        }
        return res;
    }

    // 返回位置 p 的真实值（含块标记）
    i64 get(int p) {
        return a[p] + tag[bel[p]];
    }
};
```
