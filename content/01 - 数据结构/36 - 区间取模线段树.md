# 区间取模线段树

值非负且总和在 i64 内；模数 mod>0；取模改变元素时至少把它减半。 下标从 0 编号，区间 [l,r)（含左端，不含右端），空区间和为 0。

建树 O(n)，赋值/和 O(log n)，取模单次最坏 O(n)；全程 O((n+赋值次数) log V log n+查询次数 log n)，V 为最大赋入值。

```cpp
class RangeModuloTree {
    int n;
    vector<i64> sums, maximum;
    void pull(int p) {
        sums[p] = sums[p * 2] + sums[p * 2 + 1];
        maximum[p] = max(maximum[p * 2], maximum[p * 2 + 1]);
    }
    void build(int p, int l, int r, const vector<i64> &a) {
        if (r - l == 1) {
            assert(a[l] >= 0);
            sums[p] = maximum[p] = a[l];
            return;
        }
        int m = (l + r) / 2;
        build(p * 2, l, m, a);
        build(p * 2 + 1, m, r, a);
        pull(p);
    }
    void assign(int p, int l, int r, int x, i64 v) {
        if (r - l == 1) {
            sums[p] = maximum[p] = v;
            return;
        }
        int m = (l + r) / 2;
        if (x < m)
            assign(p * 2, l, m, x, v);
        else
            assign(p * 2 + 1, m, r, x, v);
        pull(p);
    }
    void reduce(int p, int l, int r, int a, int b, i64 mod) {
        if (b <= l || r <= a || maximum[p] < mod)
            return;
        if (r - l == 1) {
            sums[p] %= mod;
            maximum[p] = sums[p];
            return;
        }
        int m = (l + r) / 2;
        reduce(p * 2, l, m, a, b, mod);
        reduce(p * 2 + 1, m, r, a, b, mod);
        pull(p);
    }
    i64 query(int p, int l, int r, int a, int b) const {
        if (b <= l || r <= a)
            return 0;
        if (a <= l && r <= b)
            return sums[p];
        int m = (l + r) / 2;
        return query(p * 2, l, m, a, b) + query(p * 2 + 1, m, r, a, b);
    }
    void check(int l, int r) const {
        assert(0 <= l && l <= r && r <= n);
    }

  public:
    // 建立非负数组线段树；空数组合法。
    explicit RangeModuloTree(const vector<i64> &a)
        : n(a.size()), sums(4 * max(1, n)), maximum(sums.size()) {
        if (n)
            build(1, 0, n, a);
    }
    // 把位置 p 赋为非负 x；无返回值。
    void set(int p, i64 x) {
        assert(0 <= p && p < n && x >= 0);
        assign(1, 0, n, p, x);
    }
    // 将 [l,r) 每项取模 mod；mod>0，空区间无操作。
    void modulo(int l, int r, i64 mod) {
        check(l, r);
        assert(mod > 0);
        if (l < r)
            reduce(1, 0, n, l, r, mod);
    }
    // 返回 [l,r) 的和；空区间为 0。
    i64 sum(int l, int r) const {
        check(l, r);
        return l == r ? 0 : query(1, 0, n, l, r);
    }
};
```
