# 扫描线（矩形面积并）

坐标为 i64，须 x1<x2、y1<y2；坐标差、面积乘积和累计面积均不能溢出 i64。 每项 {x1,y1,x2,y2}，矩形为 [x1,x2)×[y1,y2)；按连续几何面积计算，空集合返回 0。

O(n log n)；空间 O(n)。

```cpp
// 返回矩形面积并；空集合返回 0
i64 rectangleUnionArea(const std::vector<std::array<i64, 4>> &rects) {
    int n = rects.size();
    if (!n) {
        return 0;
    }
    struct Event {
        i64 x;
        i64 y1;
        i64 y2;
        int v;
    };
    std::vector<Event> ev;
    std::vector<i64> ys;
    for (auto [x1, y1, x2, y2] : rects) {
        ys.push_back(y1);
        ys.push_back(y2);
        ev.push_back({x1, y1, y2, 1});
        ev.push_back({x2, y1, y2, -1});
    }
    std::sort(ys.begin(), ys.end());
    ys.erase(std::unique(ys.begin(), ys.end()), ys.end());
    int m = ys.size() - 1;
    std::sort(ev.begin(), ev.end(), [](const Event &a, const Event &b) { return a.x < b.x; });

    std::vector<int> cnt(4 * m + 4);
    std::vector<i64> len(4 * m + 4);

    std::function<void(int, int, int, int, int, int)> upd = [&](int p, int l, int r, int ql, int qr,
                                                                int v) {
        if (ql >= r || qr <= l) {
            return;
        }
        if (ql <= l && r <= qr) {
            cnt[p] += v;
        } else {
            int mid = (l + r) / 2;
            upd(2 * p, l, mid, ql, qr, v);
            upd(2 * p + 1, mid, r, ql, qr, v);
        }
        len[p] = cnt[p] ? ys[r] - ys[l] : (r - l == 1 ? 0 : len[2 * p] + len[2 * p + 1]);
    };

    i64 ans = 0;
    for (int i = 0; i < 2 * n; i++) {
        if (i && ev[i].x > ev[i - 1].x) {
            ans += (ev[i].x - ev[i - 1].x) * len[1];
        }
        int l = std::lower_bound(ys.begin(), ys.end(), ev[i].y1) - ys.begin();
        int r = std::lower_bound(ys.begin(), ys.end(), ev[i].y2) - ys.begin();
        upd(1, 0, m, l, r, ev[i].v);
    }
    return ans;
}
```
