# 可持久化数组（单点赋值区间和）

长度固定，数组和、中间和不溢出 i64；节点池随版本数增长。 从 0 编号，区间 [l,r)（含左端，不含右端）；版本从 0 开始，0 为原数组，每次 set 返回新版本编号。

建树 O(n)，赋值/查询/split O(log n)，merge O(重叠节点数)；空间为建树、修改、合并时分配的节点总数。

```cpp
class PersistentArraySum {
    struct Node {
        int ch[2] = {};
        i64 sum = 0;
    };
    int n;
    vector<Node> t{Node{}};
    vector<int> roots;
    int build(int l, int r, const vector<i64> &a) {
        if (l == r)
            return 0;
        int p = t.size();
        t.push_back({});
        if (r - l == 1)
            t[p].sum = a[l];
        else {
            int m = (l + r) / 2;
            t[p].ch[0] = build(l, m, a);
            t[p].ch[1] = build(m, r, a);
            t[p].sum = t[t[p].ch[0]].sum + t[t[p].ch[1]].sum;
        }
        return p;
    }
    int make(int left, int right) {
        if (!left && !right)
            return 0;
        Node node;
        node.ch[0] = left;
        node.ch[1] = right;
        node.sum = t[left].sum + t[right].sum;
        t.push_back(node);
        return t.size() - 1;
    }
    int combine(int a, int b, int l, int r) {
        if (!a || !b)
            return a ? a : b;
        Node x = t[a], y = t[b];
        if (r - l == 1) {
            Node node;
            node.sum = x.sum + y.sum;
            t.push_back(node);
            return t.size() - 1;
        }
        int m = (l + r) / 2, left = combine(x.ch[0], y.ch[0], l, m),
            right = combine(x.ch[1], y.ch[1], m, r);
        return make(left, right);
    }
    pair<int, int> partition(int p, int l, int r, int a, int b) {
        if (!p || b <= l || r <= a)
            return {0, p};
        if (a <= l && r <= b)
            return {p, 0};
        Node x = t[p];
        int m = (l + r) / 2;
        auto [il, ol] = partition(x.ch[0], l, m, a, b);
        auto [ir, orr] = partition(x.ch[1], m, r, a, b);
        return {make(il, ir), make(ol, orr)};
    }
    int update(int p, int l, int r, int x, i64 v) {
        Node node = t[p];
        t.push_back(node);
        int q = t.size() - 1;
        if (r - l == 1)
            t[q].sum = v;
        else {
            int m = (l + r) / 2, d = x >= m;
            t[q].ch[d] = update(node.ch[d], d ? m : l, d ? r : m, x, v);
            t[q].sum = t[t[q].ch[0]].sum + t[t[q].ch[1]].sum;
        }
        return q;
    }
    i64 query(int p, int l, int r, int a, int b) const {
        if (!p || b <= l || r <= a)
            return 0;
        if (a <= l && r <= b)
            return t[p].sum;
        int m = (l + r) / 2;
        return query(t[p].ch[0], l, m, a, b) + query(t[p].ch[1], m, r, a, b);
    }

  public:
    // 从原数组建立版本 0；空数组合法。
    explicit PersistentArraySum(const vector<i64> &a) : n(a.size()) {
        roots.push_back(build(0, n, a));
    }
    // 从 old 版本修改位置 p 为 x，返回新版本编号；旧版本不变。
    int set(int old, int p, i64 x) {
        assert(0 <= old && old < int(roots.size()) && 0 <= p && p < n);
        int root = update(roots[old], 0, n, p, x);
        roots.push_back(root);
        return roots.size() - 1;
    }
    // 按位置将版本 a、b 的值相加，返回新版本编号，输入版本保持不变。
    int merge(int a, int b) {
        assert(0 <= a && a < int(roots.size()) && 0 <= b && b < int(roots.size()));
        int p = combine(roots[a], roots[b], 0, n);
        roots.push_back(p);
        return roots.size() - 1;
    }
    // 将版本 v 分成 [l,r) 内、外两个版本；区间外/内分别为零，旧版不变。
    pair<int, int> split(int v, int l, int r) {
        assert(0 <= v && v < int(roots.size()) && 0 <= l && l <= r && r <= n);
        auto [in, out] = partition(roots[v], 0, n, l, r);
        int a = roots.size();
        roots.push_back(in);
        roots.push_back(out);
        return {a, a + 1};
    }
    // 返回 version 版本 [l,r) 的和；空区间为 0。
    i64 sum(int version, int l, int r) const {
        assert(0 <= version && version < int(roots.size()) && 0 <= l && l <= r && r <= n);
        return query(roots[version], 0, n, l, r);
    }
};
```
