# 在线静态区间不同数

数组不修改；值为 i64，无需预先离散化。 下标从 0 开始，查询 [l,r)（含左端，不含右端），空区间返回 0；在线指查询顺序，无在线修改。

预处理 O(n log n)，查询 O(log n)，空间 O(n log n)。

```cpp
class StaticRangeDistinct {
    struct Node {
        int ch[2] = {}, sum = 0;
    };
    int n;
    vector<Node> t{Node{}};
    vector<int> roots;
    int add(int p, int l, int r, int x) {
        Node node = t[p];
        t.push_back(node);
        int q = t.size() - 1;
        t[q].sum++;
        if (r - l > 1) {
            int m = l + (r - l) / 2, d = x >= m;
            t[q].ch[d] = add(node.ch[d], d ? m : l, d ? r : m, x);
        }
        return q;
    }
    int prefix(int p, int l, int r, int x) const {
        if (!p || x <= l)
            return 0;
        if (r <= x)
            return t[p].sum;
        int m = l + (r - l) / 2;
        return prefix(t[p].ch[0], l, m, x) + prefix(t[p].ch[1], m, r, x);
    }

  public:
    // 建立前缀「前次出现位置」频次版本；空数组合法。
    explicit StaticRangeDistinct(const vector<i64> &a) : n(a.size()), roots(n + 1) {
        map<i64, int> last;
        for (int i = 0; i < n; i++) {
            int old = last.contains(a[i]) ? last[a[i]] : 0;
            roots[i + 1] = add(roots[i], 0, n + 1, old);
            last[a[i]] = i + 1;
        }
    }
    // 返回 [l,r) 内不同数数量；0<=l<=r<=n。
    int query(int l, int r) const {
        assert(0 <= l && l <= r && r <= n);
        return prefix(roots[r], 0, n + 1, l + 1) - prefix(roots[l], 0, n + 1, l + 1);
    }
};
```
