# 李超线段树

先 `LiChaoTree<i64>(xl,xr,minimum)`；`addLine(k,b)` 加整条直线，`addSegment(k,b,l,r)` 加线段，`query(x)` 查询。

x 为 i64 整数，查询域为闭区间 [xl,xr]，xl<=xr；斜率、截距、乘加结果均不溢出 i64。 求最大时所有真实值须>=-INF，求最小时须<=INF；有效线段须在查询域内。 构造第三参 false 求最大，true 求最小；空树或 x 处没有有效线时分别返回 -INF、INF。

线段作用域 [left,right] 含两端，查询 x 须在 [xl,xr]。

全域插线/查询 O(log C)，线段插入 O(log² C)；空间 O(插入数 × log C)，C 为整数域大小。

```cpp
template <class T> struct LiChaoTree {
    struct Line {
        T k;
        T b;
    };

    T xl;
    T xr;
    bool minimize;
    std::vector<Line> seg;
    std::vector<int> lc, rc;

    // 建区间 [xl_, xr_] 的空树；minimize_ 为 true 求最小值
    LiChaoTree(T xl_, T xr_, bool minimize_ = false) : xl(xl_), xr(xr_), minimize(minimize_) {
        // node 0 为哨兵，node 1 为根
        seg.push_back({0, minimize ? INF : -INF});
        seg.push_back({0, minimize ? INF : -INF});
        lc.push_back(0);
        lc.push_back(0);
        rc.push_back(0);
        rc.push_back(0);
    }

    // 返回直线 f 在 x 处的取值
    T eval(const Line &f, T x) const {
        return f.k * x + f.b;
    }

    // 返回 f 在 x 处是否比 g 更优
    bool better(const Line &a, const Line &b, T x) const {
        T va = eval(a, x), vb = eval(b, x);
        return minimize ? va < vb : va > vb;
    }

    // 把直线 f 递归插入节点 p 覆盖的 [l,r]
    void addLine(Line f, int p, T l, T r) {
        if (l == r) {
            // 单点保留较优的线，并结束递归。
            if (better(f, seg[p], l)) {
                seg[p] = f;
            }
            return;
        }
        T mid = T((i128(l) + i128(r)) >> 1);
        bool left = better(f, seg[p], l);
        bool m = better(f, seg[p], mid);
        if (m) {
            std::swap(f, seg[p]);
        }
        if (left != m) {
            if (!lc[p]) {
                lc[p] = newLine();
            }
            addLine(f, lc[p], l, mid);
        } else {
            if (!rc[p]) {
                rc[p] = newLine();
            }
            addLine(f, rc[p], mid + 1, r);
        }
    }

    // 插入全域直线 y = kx + b
    void addLine(T k, T b) {
        addLine({k, b}, 1, xl, xr);
    }

    // 在区间 [ql, qr] 上插入线段
    void addSegment(Line f, int p, T l, T r, T ql, T qr) {
        if (qr < l || r < ql) {
            return;
        }
        if (ql <= l && r <= qr) {
            addLine(f, p, l, r);
            return;
        }
        T mid = T((i128(l) + i128(r)) >> 1);
        // 只向与有效线段相交的子区间递归。
        if (ql <= mid) {
            if (!lc[p]) {
                lc[p] = newLine();
            }
            addSegment(f, lc[p], l, mid, ql, qr);
        }
        if (qr > mid) {
            if (!rc[p]) {
                rc[p] = newLine();
            }
            addSegment(f, rc[p], mid + 1, r, ql, qr);
        }
    }

    // 插入只在 [ql,qr] 生效的线段 y = kx + b
    void addSegment(T k, T b, T ql, T qr) {
        addSegment({k, b}, 1, xl, xr, ql, qr);
    }

    // 递归查询节点 p 覆盖的 [l,r] 中 x 处的最优值
    T query(T x, int p, T l, T r) const {
        T res = eval(seg[p], x);
        if (l == r) {
            return res;
        }
        T mid = T((i128(l) + i128(r)) >> 1);
        if (x <= mid) {
            if (lc[p]) {
                T t = query(x, lc[p], l, mid);
                if (minimize ? t < res : t > res) {
                    res = t;
                }
            }
        } else if (rc[p]) {
            T t = query(x, rc[p], mid + 1, r);
            if (minimize ? t < res : t > res) {
                res = t;
            }
        }
        return res;
    }

    // 返回 x 处的最值；无有效线为 ∓INF
    T query(T x) const {
        return query(x, 1, xl, xr);
    }

    // 新建一个空节点并返回编号
    int newLine() {
        seg.push_back({0, minimize ? INF : -INF});
        lc.push_back(0);
        rc.push_back(0);
        return seg.size() - 1;
    }
};
```
