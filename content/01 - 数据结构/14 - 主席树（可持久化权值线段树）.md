# 主席树（可持久化权值线段树）

先 `ChairmanTree`，调用 `init(a)` 建表后，再用 `kth(l,r,k)` 或 `countLE(l,r,value)` 查询。

a 为 `vector<i64>`；查询前必须 init(a)，换数组时重新 init；频次包含重复值。 下标从 0 编号，区间 [l,r)（含左端，不含右端），0<=l<=r<=a.size()；k 从 1 起。 kth 要求 1<=k<=r-l；空区间可 countLE，结果为 0。

init O(n log n)；kth / countLE O(log n)；空间 O(n log n)。

```cpp
// 可持久化权值线段树：前缀版本差分求区间第 k 小
struct ChairmanTree {
    int n;
    std::vector<i64> vals; // vals[id] 为压缩编号 id 对应的原值，升序且去重
    // root[i] 是前 i 个元素的版本根；root[0] 为空，查询 [l,r) 用 root[r] 减 root[l]
    std::vector<int> root, lc, rc, sum;

    // 用数组 a 建前缀版本并离散化；空数组可计数，不能求第 k 小
    void init(const std::vector<i64> &a) {
        n = a.size();
        vals = a;
        std::sort(vals.begin(), vals.end());
        vals.erase(std::unique(vals.begin(), vals.end()), vals.end());
        lc.assign(1, 0);
        rc.assign(1, 0);
        sum.assign(1, 0);
        root.assign(n + 1, 0);
        for (int i = 0; i < n; i++) {
            int r = std::lower_bound(vals.begin(), vals.end(), a[i]) - vals.begin();
            root[i + 1] = insert(root[i], 0, vals.size() - 1, r);
        }
    }

    // 在版本 pre 上插入权值 x，返回值域 [l,r] 内的新根
    int insert(int pre, int l, int r, int x) {
        int cur = lc.size();
        lc.push_back(lc[pre]);
        rc.push_back(rc[pre]);
        sum.push_back(sum[pre] + 1);
        if (l != r) {
            int mid = (l + r) / 2;
            if (x <= mid) {
                lc[cur] = insert(lc[pre], l, mid, x);
            } else {
                rc[cur] = insert(rc[pre], mid + 1, r, x);
            }
        }
        return cur;
    }

    // 返回 [l,r) 第 k 小原值；1<=k<=r-l
    i64 kth(int l, int r, int k) {
        return vals[kth(root[l], root[r], 0, vals.size() - 1, k)];
    }

    // 在版本差分中递归求第 k 小对应的值域下标
    int kth(int u, int v, int l, int r, int k) {
        if (l == r) {
            return l;
        }
        int mid = (l + r) / 2;
        int cnt = sum[lc[v]] - sum[lc[u]];
        if (k <= cnt) {
            return kth(lc[u], lc[v], l, mid, k);
        }
        return kth(rc[u], rc[v], mid + 1, r, k - cnt);
    }

    // 返回 [l,r) 中 <= x 的元素个数；空区间为 0
    int countLE(int l, int r, i64 x) {
        int pos = std::upper_bound(vals.begin(), vals.end(), x) - vals.begin() - 1;
        if (pos < 0) {
            return 0;
        }
        return countLE(root[l], root[r], 0, vals.size() - 1, pos);
    }

    // 递归统计版本差分中值域 [l,r] 内 <= x 的个数
    int countLE(int u, int v, int l, int r, int x) {
        if (x < l) {
            return 0;
        }
        if (r <= x) {
            return sum[v] - sum[u];
        }
        int mid = (l + r) / 2;
        return countLE(lc[u], lc[v], l, mid, x) + countLE(rc[u], rc[v], mid + 1, r, x);
    }
};
```
