# 子序列自动机（小写稠密版）

主串为小写 a..z；位置从 0 起，next(pos,c) 查找 [pos,n)，无解返回 n。空模式为 true，查询中的非小写字符不匹配。

建表 O(26n) 时间与空间，长度 m 的查询 O(m)。

```cpp
// 小写字母子序列自动机。
class DenseSubsequence {
    vector<array<int, 26>> to;
    int n;

  public:
    // 建立小写字母转移表；空主串合法。
    explicit DenseSubsequence(const string &s) : to(s.size() + 1), n(s.size()) {
        to[n].fill(n);
        for (int i = n - 1; i >= 0; i--) {
            assert('a' <= s[i] && s[i] <= 'z');
            to[i] = to[i + 1];
            to[i][s[i] - 'a'] = i;
        }
    }
    // 返回 [pos,n) 内首个 c 下标，无解 n；0<=pos<=n，c 为小写字母。
    int next(int pos, char c) const {
        assert(0 <= pos && pos <= n && 'a' <= c && c <= 'z');
        return to[pos][c - 'a'];
    }
    // 返回模式串是否为主串的子序列；模式串非小写字符直接不匹配。
    bool contains(const string &s) const {
        int p = 0;
        for (char c : s) {
            if (c < 'a' || c > 'z')
                return false;
            p = next(p, c);
            if (p == n)
                return false;
            p++;
        }
        return true;
    }
};
```

## 子序列自动机（持久化版）

输入 `vector<T>`，T 可排序，自动离散化。next(pos,x) 查询 [pos,n)，允许 pos=n，无解返回 n；空模式为 true。

建表 O(n log sigma) 时间与空间，长度 m 的查询 O(m log sigma)。

```cpp
// 可离散化值域的子序列自动机。
template <class T> class PersistentSubsequence {
    struct Node {
        int ch[2] = {}, pos = 0;
    };
    int n;
    vector<T> values;
    vector<Node> nodes{Node{}};
    vector<int> roots;
    int update(int p, int l, int r, int x, int pos) {
        Node node = nodes[p];
        nodes.push_back(node);
        int q = nodes.size() - 1;
        if (r - l == 1)
            nodes[q].pos = pos;
        else {
            int m = (l + r) / 2, d = x >= m;
            nodes[q].ch[d] = update(node.ch[d], d ? m : l, d ? r : m, x, pos);
        }
        return q;
    }

  public:
    // 从主序列建立持久化 next；自动离散化，空主序列合法。
    explicit PersistentSubsequence(const vector<T> &a) : n(a.size()), values(a), roots(n + 1) {
        sort(values.begin(), values.end());
        values.erase(unique(values.begin(), values.end()), values.end());
        for (int i = n - 1; i >= 0; i--) {
            int x = lower_bound(values.begin(), values.end(), a[i]) - values.begin();
            roots[i] = update(roots[i + 1], 0, values.size(), x, i);
        }
    }
    // 返回 [pos,n) 内首个 x 下标；无解 n，允许 pos=n。
    int next(int pos, const T &x) const {
        assert(0 <= pos && pos <= n);
        auto it = lower_bound(values.begin(), values.end(), x);
        if (it == values.end() || *it != x)
            return n;
        int v = it - values.begin(), p = roots[pos], l = 0, r = values.size();
        while (p && r - l > 1) {
            int m = (l + r) / 2, d = v >= m;
            p = nodes[p].ch[d];
            if (d)
                l = m;
            else
                r = m;
        }
        return p ? nodes[p].pos : n;
    }
    // 返回 query 是否为主序列子序列；空 query 返回 true。
    bool contains(const vector<T> &query) const {
        int p = 0;
        for (const T &x : query) {
            p = next(p, x);
            if (p == n)
                return false;
            p++;
        }
        return true;
    }
};
```
