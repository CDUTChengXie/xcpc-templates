# AC自动机

text 和 patterns 中的字符串仅含小写 a..z；patterns 为 `vector<string>`。 counts[i] 对应 patterns[i]，包含重叠；找不到为 0，重复模式分别返回次数；空模式出现 text.size()+1 次。

O(模式总长×26+|text|+模式数) 时间，空间 O(模式总长×26+模式数)。

```cpp
// 支持多模式串匹配的 AC 自动机。
struct AhoCorasick {
    static constexpr int ALPHABET = 26;

    // 自动机节点：深度、fail 指针与转移表。
    struct Node {
        int len;
        int link;
        std::array<int, ALPHABET> next;
        Node() : len{0}, link{0}, next{} {}
    };

    std::vector<Node> t;
    std::vector<int> order; // work() 记录的先父后子的遍历顺序

    // 创建自动机并初始化。
    AhoCorasick() {
        init();
    }

    // 清空到仅含虚拟根和真根。
    void init() {
        t.assign(2, Node());
        order.clear();
        t[0].next.fill(1);
        t[0].len = -1;
    }

    // 创建并返回一个空节点。
    int newNode() {
        t.emplace_back();
        return t.size() - 1;
    }

    // 插入模式串，返回终点状态。
    int add(const std::string &a) {
        int p = 1;
        for (auto c : a) {
            int x = c - 'a';
            if (t[p].next[x] == 0) {
                t[p].next[x] = newNode();
                t[t[p].next[x]].len = t[p].len + 1;
            }
            p = t[p].next[x];
        }
        return p;
    }

    // 建立 fail 指针并将 next 补全为完整转移表。
    void work() {
        order.clear();
        std::queue<int> q;
        q.push(1);
        while (!q.empty()) {
            int x = q.front();
            q.pop();
            order.push_back(x);
            for (int i = 0; i < ALPHABET; i++) {
                if (t[x].next[i] == 0) {
                    t[x].next[i] = t[t[x].link].next[i];
                } else {
                    t[t[x].next[i]].link = t[t[x].link].next[i];
                    q.push(t[x].next[i]);
                }
            }
        }
    }

    // 已 work 后，返回每个节点代表的模式串在 text 中的出现次数；重复查询安全。
    std::vector<i64> countOccurrences(const std::string &text) const {
        std::vector<i64> count(t.size());
        int state = 1;
        for (char c : text) {
            state = t[state].next[c - 'a'];
            ++count[state];
        }
        for (auto it = order.rbegin(); it != order.rend(); ++it)
            if (*it != 1)
                count[t[*it].link] += count[*it];
        count[1] = text.size() + 1;
        return count;
    }

    // 返回整数转移目标。
    int next(int p, int x) {
        return t[p].next[x];
    }

    // 返回字符转移目标。
    int next(int p, char c, char offset = 'a') {
        return next(p, c - offset);
    }

    // 返回 fail 状态编号。
    int link(int p) {
        return t[p].link;
    }

    // 返回状态 p 的 trie 深度。
    int len(int p) {
        return t[p].len;
    }

    // 返回状态总数。
    int size() {
        return t.size();
    }
};

// 返回每个模式串的出现次数，顺序与 patterns 一致，包含重叠与重复模式。
std::vector<i64> acMatchCounts(const std::string &text, const std::vector<std::string> &patterns) {
    AhoCorasick ac;
    std::vector<int> ends;
    for (const auto &pattern : patterns)
        ends.push_back(ac.add(pattern));
    ac.work();
    auto count = ac.countOccurrences(text);
    std::vector<i64> result;
    for (int end : ends)
        result.push_back(count[end]);
    return result;
}
```
