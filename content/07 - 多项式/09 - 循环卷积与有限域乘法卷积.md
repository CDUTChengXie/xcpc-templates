# 循环卷积与有限域乘法卷积

计数系数模 998244353 且规范；cyclic 的 period>=1，a、b 长度<=period，所需 NTT 长度<=2^23。 multiplicative 的 prime 是素数且 2<=prime<=2^21，a、b 长度均为 prime；primitive 是模 prime 的原根。

输入下标从 0 开始；prime 是下标的模数，不是系数模数；原根须调用者提供。 cyclic 返回 period 项，空输入返回全 0；乘法卷积返回 prime 项，包含任一输入下标为 0 的贡献。

cyclic O(M(period))；multiplicative O(prime + M(prime))。

```cpp
namespace poly {
#ifndef XCPC_TEMPLATE_NTT
#define XCPC_TEMPLATE_NTT

    constexpr int MOD = 998244353;
    constexpr int ROOT = 3;

    // 返回 a^b mod MOD，要求 b >= 0
    int powerMod(int a, i64 b) {
        i64 res = 1, x = a;
        for (; b; b >>= 1, x = x * x % MOD) {
            if (b & 1) {
                res = res * x % MOD;
            }
        }
        return int(res);
    }

    std::vector<int> rev, roots{0, 1};

    // 原地正变换；要求 a.size() 是大于 1 的 2 的幂
    void dft(std::vector<int> &a) {
        int n = a.size();
        if (int(rev.size()) != n) {
            int k = __builtin_ctz(unsigned(n)) - 1;
            rev.resize(n);
            for (int i = 0; i < n; i++) {
                rev[i] = rev[i >> 1] >> 1 | (i & 1) << k;
            }
        }
        for (int i = 0; i < n; i++) {
            if (rev[i] < i) {
                std::swap(a[i], a[rev[i]]);
            }
        }
        if (int(roots.size()) < n) {
            int k = __builtin_ctz(unsigned(roots.size()));
            roots.resize(n);
            while ((1 << k) < n) {
                // 必须用 (MOD - 1) >> (k + 1)；998244353 - 1 含因子 119
                int e = powerMod(ROOT, (MOD - 1) >> (k + 1));
                for (int i = 1 << (k - 1); i < (1 << k); i++) {
                    roots[2 * i] = roots[i];
                    roots[2 * i + 1] = int(1LL * roots[i] * e % MOD);
                }
                k++;
            }
        }

        for (int k = 1; k < n; k *= 2) {
            for (int i = 0; i < n; i += 2 * k) {
                for (int j = 0; j < k; j++) {
                    int u = a[i + j];
                    int v = int(1LL * a[i + j + k] * roots[k + j] % MOD);
                    a[i + j] = (u + v) % MOD;
                    a[i + j + k] = (u - v + MOD) % MOD;
                }
            }
        }
    }

    // 原地逆变换；要求 a.size() 是大于 1 的 2 的幂
    void idft(std::vector<int> &a) {
        int n = a.size();
        std::reverse(a.begin() + 1, a.end());
        dft(a);
        // 必须取正代表元，避免后续乘法产生负余数
        int inv = MOD - (MOD - 1) / n;
        for (int i = 0; i < n; i++) {
            a[i] = int(1LL * a[i] * inv % MOD);
        }
    }

    // 返回乘积系数，result[i] 对应 x^i；任一输入为空返回空
    std::vector<int> nttMul(std::vector<int> a, std::vector<int> b) {
        if (a.empty() || b.empty()) {
            return {};
        }
        int n = 1, tot = int(a.size() + b.size() - 1);
        while (n < tot && n <= (1 << 23)) {
            n *= 2;
        }
        assert(n >= tot && "NTT 长度超出 998244353 支持的 2^23，请改用 469762049 或拆分卷积");
        if (tot < 128) {
            std::vector<int> c(tot, 0);
            int na = a.size(), nb = b.size();
            for (int i = 0; i < na; i++) {
                for (int j = 0; j < nb; j++) {
                    c[i + j] = int((c[i + j] + 1LL * a[i] * b[j]) % MOD);
                }
            }
            return c;
        }
        a.resize(n);
        b.resize(n);
        dft(a);
        dft(b);
        for (int i = 0; i < n; i++) {
            a[i] = int(1LL * a[i] * b[i] % MOD);
        }
        idft(a);
        a.resize(tot);
        return a;
    }

    // 返回 1/f 的前 m 项系数，result[i] 对应 x^i；要求 f 非空且 f[0]!=0
    std::vector<int> polyInv(std::vector<int> f, int m) {
        assert(!f.empty() && f[0] != 0 && "polyInv 要求 f[0] != 0");
        if (m <= 0) {
            return {};
        }
        std::vector<int> g(1, powerMod(f[0], MOD - 2));
        for (int len = 1; len < m; len *= 2) {
            std::vector<int> t(f.begin(), f.begin() + std::min(2 * len, int(f.size())));
            auto h = nttMul(t, g);
            h.resize(2 * len, 0);
            for (int i = 0; i < 2 * len; i++) {
                h[i] = (MOD - h[i]) % MOD;
            }
            h[0] = (h[0] + 2) % MOD;
            g = nttMul(g, h);
            g.resize(2 * len);
        }
        g.resize(m);
        return g;
    }

#endif // XCPC_TEMPLATE_NTT
}

namespace index_convolution {
    using V = std::vector<int>;
    constexpr int mod = poly::MOD;
    // 返回 (i+j)%period 的循环卷积，长度 period
    V cyclic(const V &a, const V &b, int period) {
        assert(period >= 1 && a.size() <= period && b.size() <= period);
        V ans(period);
        auto c = poly::nttMul(a, b);
        for (int i = 0; i < (int)c.size(); ++i)
            ans[i % period] = (ans[i % period] + c[i]) % mod;
        return ans;
    }
    // 返回 (i*j)%prime 的乘法卷积；要求 prime 为素数、a、b 长度均为 prime
    V multiplicative(const V &a, const V &b, int prime, int primitive) {
        assert(prime >= 2 && prime <= (1 << 21) && a.size() == prime && b.size() == prime &&
               primitive >= 1 && primitive < prime);
        V powers(prime - 1), seen(prime), x(prime - 1), y(prime - 1);
        long long v = 1;
        for (int i = 0; i < prime - 1; ++i) {
            assert(!seen[v]);
            seen[v] = 1;
            powers[i] = v;
            x[i] = a[v];
            y[i] = b[v];
            v = v * primitive % prime;
        }
        assert(v == 1);
        auto c = cyclic(x, y, prime - 1);
        V h(prime);
        for (int i = 0; i < prime - 1; ++i)
            h[powers[i]] = c[i];
        long long sa = 0, sb = 0;
        for (int z : a)
            sa = (sa + z) % mod;
        for (int z : b)
            sb = (sb + z) % mod;
        h[0] = (a[0] * sb + b[0] * sa - 1LL * a[0] * b[0] % mod + mod) % mod;
        return h;
    }
}
```
