# 离线区间不同数

a 为 `vector<i64>`；queries 为 `vector<pair<int,int>>`，每项 {l,r} 满足 0<=l<=r<=a.size()。 查询用半开区间 [l,r)（含左端，不含右端）；返回顺序与输入 queries 一致，空区间为 0。

O((n+q)log(n+q)) 时间，O(n+q) 空间。

```cpp
namespace DistinctOffline {
    // 一次性处理全部半开区间 [l,r)；非法端点抛异常
    std::vector<int> solve(const std::vector<i64> &a,
                           const std::vector<std::pair<int, int>> &queries) {
        int n = a.size(), q = queries.size();
        for (auto [l, r] : queries)
            if (l < 0 || l > r || r > n)
                throw std::invalid_argument("range");
        auto values = a;
        std::sort(values.begin(), values.end());
        values.erase(std::unique(values.begin(), values.end()), values.end());
        std::vector<int> last(values.size(), -1), bit(n + 1), order(q), answer(q);
        std::iota(order.begin(), order.end(), 0);
        std::sort(order.begin(), order.end(),
                  [&](int i, int j) { return queries[i].second < queries[j].second; });
        auto add = [&](int p, int delta) {
            for (++p; p <= n; p += p & -p)
                bit[p] += delta;
        };
        auto prefix = [&](int r) {
            int sum = 0;
            for (; r > 0; r -= r & -r)
                sum += bit[r];
            return sum;
        };
        int processed = 0;
        for (int id : order) {
            auto [l, r] = queries[id];
            while (processed < r) {
                int p = processed++,
                    value = std::lower_bound(values.begin(), values.end(), a[p]) - values.begin();
                if (last[value] != -1)
                    add(last[value], -1);
                add(p, 1);
                last[value] = p;
            }
            answer[id] = prefix(r) - prefix(l);
        }
        return answer;
    }
}
```
