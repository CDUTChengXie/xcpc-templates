# 万能欧几里得（非交换路径聚合）

p>=0、q>0、0<=r<q、l>=0，floor((p*l+r)/q) 不超过 long long；T{} 为空结果，operator+ 按顺序合并且满足结合律（只改括号不改结果）。 第 i=1..l 步先补 U 到 floor((p*i+r)/q)，再放一个 R；初始高度 0。

O(log(max(p,q))·log(路径长度)) 次半群合并，递归 O(log(max(p,q)))。

```cpp
namespace euclidean_word {
    // 按顺序合并 a 共 k 次，k>=0；k=0 返回 T{} 的空结果。
    template <class T> T repeat(T a, long long k) {
        T r{};
        for (; k; k >>= 1) {
            if (k & 1)
                r = r + a;
            if (k > 1)
                a = a + a;
        }
        return r;
    }
    namespace detail {
        template <class T> T solve(long long p, long long q, long long r, long long l, T U, T R) {
            if (!l)
                return T{};
            if (p >= q)
                return solve(p % q, q, r, l, U, repeat(U, p / q) + R);
            long long m = ((__int128)p * l + r) / q;
            if (!m)
                return repeat(R, l);
            long long first = (q - r - 1) / p;
            long long tail = l - ((__int128)q * m - r - 1) / p;
            return repeat(R, first) + U + solve(q, p, (q - r - 1) % p, m - 1, R, U) +
                   repeat(R, tail);
        }
    }
    // 聚合 l 个横向步骤，返回路径元素；输入区间见头注释。
    template <class T> T solve(long long p, long long q, long long r, long long l, T U, T R) {
        assert(p >= 0 && q > 0 && r >= 0 && r < q && l >= 0);
        assert((__int128)p * l / q <= LLONG_MAX);
        return detail::solve(p, q, r, l, U, R);
    }
}
```
