# 数独字典序构造与求解

construct 需 N,M>=0、N+M<=10，宫高=2^N、宫宽=2^M；solve 需 h,w>=1、h*w<=16，输入为边长 h*w 的方阵。 下标 [0,边长)；数字 1..边长，0 为待填；保持提示不变并返回字典序最小解，无解返回 nullopt，非法提示抛异常。

construct O(边长²)；solve 最坏指数时间，空间 O(边长²)。

```cpp
namespace SudokuTools {
    // 构造宫高 2^N、宫宽 2^M 的空数独；限制 N+M<=10 控制内存
    std::vector<std::vector<int>> construct(int N, int M) {
        if (N < 0 || M < 0 || N > 10 || M > 10 || N + M > 10)
            throw std::invalid_argument("size");
        int h = 1 << N, w = 1 << M, n = h * w;
        std::vector a(n, std::vector<int>(n));
        for (int i = 0; i < n; ++i)
            for (int j = 0; j < n; ++j)
                a[i][j] = ((i % h) ^ (j / w)) * w + ((i / h) ^ (j % w)) + 1;
        return a;
    }
    // 返回行优先、数字升序下的第一组解，即完整矩阵的字典序最小解
    std::optional<std::vector<std::vector<int>>> solve(std::vector<std::vector<int>> a, int h,
                                                       int w) {
        if (h < 1 || w < 1 || h > 16 || w > 16 || h * w > 16)
            throw std::invalid_argument("size");
        int n = h * w;
        if (a.size() != size_t(n))
            throw std::invalid_argument("rows");
        std::vector<unsigned> row(n), col(n), box(n);
        auto block = [&](int i, int j) { return (i / h) * h + j / w; };
        for (int i = 0; i < n; ++i) {
            if (a[i].size() != size_t(n))
                throw std::invalid_argument("cols");
            for (int j = 0; j < n; ++j) {
                int v = a[i][j];
                if (v < 0 || v > n)
                    throw std::invalid_argument("digit");
                if (!v)
                    continue;
                unsigned b = 1u << (v - 1);
                int id = block(i, j);
                if ((row[i] | col[j] | box[id]) & b)
                    throw std::invalid_argument("duplicate");
                row[i] |= b;
                col[j] |= b;
                box[id] |= b;
            }
        }
        // ponytail: 按字典序逐格搜索最坏指数；大型提示盘改用精确覆盖可行性判定配合逐格定值。
        auto dfs = [&](auto &&self, int p) -> bool {
            if (p == n * n)
                return true;
            int i = p / n, j = p % n, id = block(i, j);
            if (a[i][j])
                return self(self, p + 1);
            unsigned available = ((1u << n) - 1) & ~(row[i] | col[j] | box[id]);
            while (available) {
                unsigned b = available & -available;
                available -= b;
                a[i][j] = std::countr_zero(b) + 1;
                row[i] |= b;
                col[j] |= b;
                box[id] |= b;
                if (self(self, p + 1))
                    return true;
                row[i] ^= b;
                col[j] ^= b;
                box[id] ^= b;
            }
            a[i][j] = 0;
            return false;
        };
        if (!dfs(dfs, 0))
            return std::nullopt;
        return a;
    }
}
```
