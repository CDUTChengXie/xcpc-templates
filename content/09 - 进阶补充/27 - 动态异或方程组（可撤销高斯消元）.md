# 动态异或方程组（可撤销高斯消元）

N 为变量个数的编译期上界；调用方把删除转为栈撤销或离线区间。 变量 [0,N)，coeff 为 `bitset<N>`，coeff[j]=1 表示方程包含变量 x[j]，所有项异或后等于 bool rhs。 解中自由变量取 0；时间为已加入方程数，矛盾/重复方程也占一个时间；先记录再添加，最后撤销。

add/solution O(N²/字长 + N)，rollback 每条 O(N/字长)，空间 O(N²/字长 + 操作数)。

```cpp
namespace xor_equations {
    // 可撤销异或方程组。
    template <size_t N> class System {
        std::array<std::bitset<N>, N> rows{};
        std::array<bool, N> rhs{};
        int bad = 0;
        struct Undo {
            int pivot, bad;
        };
        std::vector<Undo> history;

      public:
        // 返回当前已加入方程数，可交给rollback。
        int time() const {
            return history.size();
        }
        // 加入一条方程，返回所有当前方程是否存在解。
        bool add(std::bitset<N> a, bool b) {
            int old = bad;
            for (int p = (int)N - 1; p >= 0; --p)
                if (a[p]) {
                    if (rows[p][p])
                        a ^= rows[p], b ^= rhs[p];
                    else {
                        rows[p] = a;
                        rhs[p] = b;
                        history.push_back({p, old});
                        return !bad;
                    }
                }
            bad += b;
            history.push_back({-1, old});
            return !bad;
        }
        // 撤销至t，t∈[0,time()]。
        void rollback(int t) {
            assert(t >= 0 && t <= time());
            while (time() > t) {
                auto u = history.back();
                history.pop_back();
                bad = u.bad;
                if (u.pivot >= 0)
                    rows[u.pivot].reset(), rhs[u.pivot] = false;
            }
        }
        // 返回一个解，自由变量为0；无解时返回nullopt。
        std::optional<std::bitset<N>> solution() const {
            if (bad)
                return {};
            std::bitset<N> x;
            for (size_t p = 0; p < N; ++p)
                if (rows[p][p])
                    x[p] = rhs[p] ^ ((rows[p] & x).count() & 1);
            return x;
        }
    };
}
```
