# 组合键哈希（嵌套容器）

叶子元素需支持 std::hash；顺序和长度参与计算，哈希可能碰撞。

时间 O(叶子元素数)，递归栈 O(组合嵌套层数)。

```cpp
// 组合键哈希，支持 pair/tuple/vector 等有序组合键。
struct CompositeHash {
    // 返回单项或组合键哈希；叶子元素需支持 std::hash
    template <class T> size_t operator()(const T &x) const {
        return std::hash<T>{}(x);
    }
    // 返回有序 pair 哈希，first 与 second 的顺序保留
    template <class A, class B> size_t operator()(const std::pair<A, B> &x) const {
        size_t h = 0;
        combine(h, (*this)(x.first));
        combine(h, (*this)(x.second));
        return h;
    }
    // 返回有序 tuple 哈希，允许嵌套组合键
    template <class... T> size_t operator()(const std::tuple<T...> &x) const {
        size_t h = 0;
        std::apply([&](const auto &...v) { (combine(h, (*this)(v)), ...); }, x);
        return h;
    }
    // 返回有序 vector 哈希，元素顺序与长度参与计算
    template <class T> size_t operator()(const std::vector<T> &x) const {
        size_t h = x.size();
        for (const auto &v : x)
            combine(h, (*this)(v));
        return h;
    }

  private:
    static void combine(size_t &h, size_t v) {
        h ^= v + size_t(0x9e3779b97f4a7c15ULL) + (h << 6) + (h >> 2);
    }
};
```

## 可复现随机数据生成

使用 mt19937_64；l<=r，数组 n>=0；树 n>=1 且 root 在 [0,n)。图 1<=n<=2000、n-1<=m<=n(n-1)/2。

节点从 0 编号；同一标准库实现下种子及调用顺序相同可复现。生成树不是均匀标号树；图无自环重边，非法参数抛异常。

数组与树 O(n)；图 O(n² log n) 时间、O(n²) 空间。

```cpp
// 可复现随机数据生成器。
struct SampleTools {
    std::mt19937_64 rng;
    // 固定种子重现同一批数据
    explicit SampleTools(u64 seed = 1) : rng(seed) {}
    // 返回闭区间 [l,r] 的均匀 i64，支持完整有符号范围
    i64 integer(i64 l, i64 r) {
        if (l > r)
            throw std::invalid_argument("range");
        return std::uniform_int_distribution<i64>(l, r)(rng);
    }
    // 返回指定长度的数组；n>=0
    std::vector<i64> array(int n, i64 l, i64 r) {
        if (n < 0 || l > r)
            throw std::invalid_argument("array");
        std::vector<i64> a(n);
        for (auto &x : a)
            x = integer(l, r);
        return a;
    }
    // 生成以指定 root 为起点的一棵标号树；返回无向边，节点从 0 开始
    std::vector<std::pair<int, int>> tree(int n, int root = 0) {
        if (n < 1 || root < 0 || root >= n)
            throw std::invalid_argument("tree");
        std::vector<int> labels(n);
        std::iota(labels.begin(), labels.end(), 0);
        std::swap(labels[0], labels[root]);
        std::shuffle(labels.begin() + 1, labels.end(), rng);
        std::vector<std::pair<int, int>> edges;
        // ponytail: 随机前驱树并非均匀标号树；需要均匀分布时改用随机 Prüfer 序列。
        for (int i = 1; i < n; ++i)
            edges.emplace_back(labels[i], labels[integer(0, i - 1)]);
        std::shuffle(edges.begin(), edges.end(), rng);
        return edges;
    }
    // 返回连通简单图；要求 n-1<=m<=n(n-1)/2，n<=2000
    std::vector<std::pair<int, int>> connectedGraph(int n, int m) {
        if (n < 1 || n > 2000 || m < n - 1 || i64(m) > i64(n) * (n - 1) / 2)
            throw std::invalid_argument("graph");
        auto edges = tree(n);
        std::set<std::pair<int, int>> used;
        for (auto &[u, v] : edges) {
            if (u > v)
                std::swap(u, v);
            used.emplace(u, v);
        }
        // ponytail: 枚举全部未使用边占 O(n²) 内存；超大稀疏图改用拒绝采样，稠密图采样补图。
        std::vector<std::pair<int, int>> candidates;
        for (int u = 0; u < n; ++u)
            for (int v = u + 1; v < n; ++v)
                if (!used.count({u, v}))
                    candidates.emplace_back(u, v);
        std::shuffle(candidates.begin(), candidates.end(), rng);
        edges.insert(edges.end(), candidates.begin(), candidates.begin() + (m - (n - 1)));
        std::shuffle(edges.begin(), edges.end(), rng);
        return edges;
    }
};
```
