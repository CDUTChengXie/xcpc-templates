# 精确威佐夫与斐波那契取法

威佐夫两堆为 u64、均在 0..10^18；每步减一堆或两堆同减正数；斐波那契 n 为正 u64，首步不能取完，之后至多前步两倍。 Wythoff::move 返回两堆各取走的数量；FibonacciTake::first 返回首步数量；无必胜取法为 nullopt。

威佐夫判定 O(log V)，取法 O(log²V)；斐波那契 O(log n)。

```cpp
namespace Wythoff {
    using Wide = unsigned __int128;
    // 返回必败对 (floor(t*phi), floor(t*phi)+t)，t 在 0..10^18
    std::pair<u64, u64> cold(u64 t) {
        if (t > 1000000000000000000ULL)
            throw std::out_of_range("wythoff");
        Wide n = Wide(5) * t * t;
        u64 l = 0, r = 3000000000000000000ULL;
        while (l < r) {
            u64 m = l + (r - l + 1) / 2;
            if (Wide(m) * m <= n)
                l = m;
            else
                r = m - 1;
        }
        u64 a = (t + l) / 2;
        return {a, a + t};
    }
    // 精确判定必败，输入顺序任意
    bool losing(u64 a, u64 b) {
        if (a > 1000000000000000000ULL || b > 1000000000000000000ULL)
            throw std::out_of_range("heap");
        if (a > b)
            std::swap(a, b);
        return cold(b - a).first == a;
    }
    // 返回必胜取数，不改变输入；每个数对应原参数的石堆
    std::optional<std::pair<u64, u64>> move(u64 a, u64 b) {
        if (losing(a, b))
            return std::nullopt;
        bool reversed = a > b;
        if (reversed)
            std::swap(a, b);
        std::vector<u64> candidates{b - a};
        for (u64 v : {a, b})
            for (int side = 0; side < 2; ++side) {
                u64 l = 0, r = v;
                while (l < r) {
                    u64 m = l + (r - l) / 2;
                    auto p = cold(m);
                    if ((side ? p.second : p.first) < v)
                        l = m + 1;
                    else
                        r = m;
                }
                candidates.push_back(l);
            }
        for (u64 t : candidates) {
            auto [x, y] = cold(t);
            for (int swapTarget = 0; swapTarget < 2; ++swapTarget) {
                if (swapTarget)
                    std::swap(x, y);
                if (x > a || y > b)
                    continue;
                u64 da = a - x, db = b - y;
                if (da + db != 0 && (da == 0 || db == 0 || da == db)) {
                    if (reversed)
                        std::swap(da, db);
                    return std::pair{da, db};
                }
            }
        }
        assert(false);
        return std::nullopt;
    }
}
namespace FibonacciTake {
    // 返回 n 的 Zeckendorf 分解中最小 Fibonacci 项；n>=1
    u64 smallest(u64 n) {
        if (n == 0)
            throw std::invalid_argument("n");
        std::vector<u64> f{1, 2};
        while (f.back() <= n && f.back() <= ULLONG_MAX - f[f.size() - 2])
            f.push_back(f.back() + f[f.size() - 2]);
        u64 last = 0;
        for (auto it = f.rbegin(); it != f.rend(); ++it)
            if (*it <= n) {
                n -= *it;
                last = *it;
            }
        return last;
    }
    // 首步不能取完；返回最小必胜首步，没有时返回 nullopt
    std::optional<u64> first(u64 n) {
        u64 t = smallest(n);
        return t == n ? std::nullopt : std::optional<u64>(t);
    }
}
```
