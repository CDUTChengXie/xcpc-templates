# 德州扑克牌型比较

标准 52 张牌、无鬼牌、不得重复；Card.rank 为 2..14（A=14），suit 为 0/1/2/3 对应 C/D/H/S。 Key 是 `array<int,6>`，字典序越大越强；key[0]=0..9 对应高牌/对子/两对/三条/顺子/同花/葫芦/四条/同花顺/皇家同花顺。

顺子类 key[1] 为最高点数（A2345 为 5）；高牌/同花 key[1..5] 为降序点数；其余按组张数多优先、同张数点数大优先存各组点数，未用项为 0。 同点数不按花色决胜，非法牌/张数/重复牌抛 invalid_argument。

five O(1)，best 最多枚举 C(7,5)=21 组，空间 O(1)。

```cpp
namespace Poker {
    // 一张牌：点数 2..14、花色 0..3。
    struct Card {
        int rank, suit;
    };
    // key[0] 为牌型，后五项为依次比较的点数，未用项为 0；字段规则见头部约定。
    using Key = std::array<int, 6>;
    // 读取 2..9,T,J,Q,K,A 加 C,D,H,S；也允许 10，非法格式抛异常
    Card parse(std::string_view s) {
        if (s.size() < 2 || s.size() > 3)
            throw std::invalid_argument("card");
        std::string ranks = "23456789TJQKA", suits = "CDHS";
        int rank = -1;
        if (s.size() == 3 && s.substr(0, 2) == "10")
            rank = 10;
        if (s.size() == 2) {
            auto p = ranks.find(s[0]);
            if (p != std::string::npos)
                rank = int(p) + 2;
        }
        auto suit = suits.find(s.back());
        if (rank < 2 || suit == std::string::npos)
            throw std::invalid_argument("card");
        return {rank, int(suit)};
    }
    // 验证点数、花色与重复牌；不修改输入
    void validate(const std::vector<Card> &a) {
        std::set<std::pair<int, int>> seen;
        for (auto c : a)
            if (c.rank < 2 || c.rank > 14 || c.suit < 0 || c.suit >= 4 ||
                !seen.emplace(c.rank, c.suit).second)
                throw std::invalid_argument("cards");
    }
    // 五张比较键：0高牌、1对子、2两对、3三条、4顺子、5同花、6葫芦、7四条、8同花顺、9皇家同花顺
    Key five(const std::vector<Card> &a) {
        if (a.size() != 5)
            throw std::invalid_argument("five");
        validate(a);
        std::array<int, 15> count{};
        bool flush = true;
        for (auto c : a) {
            ++count[c.rank];
            flush &= c.suit == a[0].suit;
        }
        std::vector<int> descending;
        std::vector<std::pair<int, int>> groups;
        for (int r = 14; r >= 2; --r)
            if (count[r]) {
                descending.push_back(r);
                groups.emplace_back(count[r], r);
            }
        std::sort(groups.rbegin(), groups.rend());
        int straight = 0;
        if (descending.size() == 5) {
            if (descending.front() - descending.back() == 4)
                straight = descending.front();
            if (descending == std::vector<int>{14, 5, 4, 3, 2})
                straight = 5;
        }
        Key key{};
        if (straight && flush) {
            key[0] = straight == 14 ? 9 : 8;
            key[1] = straight;
            return key;
        }
        if (groups[0].first == 4)
            key[0] = 7;
        else if (groups[0].first == 3 && groups[1].first == 2)
            key[0] = 6;
        else if (flush)
            key[0] = 5;
        else if (straight) {
            key[0] = 4;
            key[1] = straight;
            return key;
        } else if (groups[0].first == 3)
            key[0] = 3;
        else if (groups[0].first == 2 && groups[1].first == 2)
            key[0] = 2;
        else if (groups[0].first == 2)
            key[0] = 1;
        if (key[0] == 0 || key[0] == 5)
            std::copy(descending.begin(), descending.end(), key.begin() + 1);
        else
            for (int i = 0; i < int(groups.size()); ++i)
                key[i + 1] = groups[i].second;
        return key;
    }
    // 返回五到七张牌中的最强五张键，拒绝重复牌
    Key best(const std::vector<Card> &a) {
        if (a.size() < 5 || a.size() > 7)
            throw std::invalid_argument("hand");
        validate(a);
        Key result{};
        int n = a.size();
        for (int i = 0; i < n; ++i)
            for (int j = i + 1; j < n; ++j)
                for (int k = j + 1; k < n; ++k)
                    for (int l = k + 1; l < n; ++l)
                        for (int m = l + 1; m < n; ++m)
                            result = std::max(result, five({a[i], a[j], a[k], a[l], a[m]}));
        return result;
    }
}
```
