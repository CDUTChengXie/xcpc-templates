# 线性规划（两阶段单纯形）

A 每行长 c.size、b.size==A.size；输入有限 long double；eps>0，数据应先缩放。 变量下标 [0,n)、约束 [0,m)，A[i][j] 为第 i 条约束中 x[j] 的系数；原输入不修改。

每个实例只 solve 一次；optimal 表示找到最优解，infeasible 表示无可行解，unbounded 表示目标可无限增大。

单次转轴 O(mn)，最坏指数次转轴，空间 O(mn)；浮点需合理 eps。

```cpp
namespace linear_program {
    using Real = long double;
    enum class Status { optimal, infeasible, unbounded };
    struct Result {
        Status status;
        Real value;
        std::vector<Real> x;
    }; // 仅 optimal 时 value 与 x[j] 为最优目标和变量值
    // 两阶段单纯形求解 max c·x，s.t. A·x<=b、x>=0。
    class Simplex {
        int m, n;
        Real eps;
        std::vector<int> B, N;
        std::vector<std::vector<Real>> D;
        void pivot(int r, int s) {
            Real inv = 1 / D[r][s];
            for (int i = 0; i < m + 2; ++i)
                if (i != r)
                    for (int j = 0; j < n + 2; ++j)
                        if (j != s)
                            D[i][j] -= D[r][j] * D[i][s] * inv;
            for (int j = 0; j < n + 2; ++j)
                if (j != s)
                    D[r][j] *= inv;
            for (int i = 0; i < m + 2; ++i)
                if (i != r)
                    D[i][s] *= -inv;
            D[r][s] = inv;
            std::swap(B[r], N[s]);
        }
        bool phase(int ph) {
            int obj = ph == 1 ? m + 1 : m;
            for (;;) {
                int s = -1;
                for (int j = 0; j <= n; ++j)
                    if (!(ph == 2 && N[j] == -1) && D[obj][j] < -eps && (s == -1 || N[j] < N[s]))
                        s = j;
                if (s == -1)
                    return true;
                int r = -1;
                for (int i = 0; i < m; ++i)
                    if (D[i][s] > eps) {
                        Real v = D[i][n + 1] / D[i][s];
                        if (r == -1 || v < D[r][n + 1] / D[r][s] - eps ||
                            (std::abs(v - D[r][n + 1] / D[r][s]) <= eps && B[i] < B[r]))
                            r = i;
                    }
                if (r == -1)
                    return false;
                pivot(r, s);
            }
        }

      public:
        // 建立 LP 表；输入 A 为 m×n，b 为 m，c 为 n。
        Simplex(const std::vector<std::vector<Real>> &A, const std::vector<Real> &b,
                const std::vector<Real> &c, Real eps = 1e-12L)
            : m(b.size()), n(c.size()), eps(eps), B(m), N(n + 1),
              D(m + 2, std::vector<Real>(n + 2)) {
            assert(A.size() == b.size() && eps > 0);
            for (int i = 0; i < m; ++i) {
                assert(A[i].size() == c.size() && std::isfinite(b[i]));
                for (int j = 0; j < n; ++j) {
                    assert(std::isfinite(A[i][j]));
                    D[i][j] = A[i][j];
                }
                B[i] = n + i;
                D[i][n] = -1;
                D[i][n + 1] = b[i];
            }
            for (int j = 0; j < n; ++j) {
                assert(std::isfinite(c[j]));
                N[j] = j;
                D[m][j] = -c[j];
            }
            N[n] = -1;
            D[m + 1][n] = 1;
        }
        // 消耗内部表并求解；每个实例调用一次，返回状态、目标值及最优变量。
        Result solve() {
            int r = -1;
            for (int i = 0; i < m; ++i)
                if (r == -1 || D[i][n + 1] < D[r][n + 1])
                    r = i;
            if (r != -1 && D[r][n + 1] < -eps) {
                pivot(r, n);
                if (!phase(1) || std::abs(D[m + 1][n + 1]) > eps)
                    return {Status::infeasible, NAN, {}};
                for (int i = 0; i < m; ++i)
                    if (B[i] == -1) {
                        int s = -1;
                        for (int j = 0; j <= n; ++j)
                            if (std::abs(D[i][j]) > eps && (s == -1 || N[j] < N[s]))
                                s = j;
                        if (s != -1)
                            pivot(i, s);
                    }
            }
            if (!phase(2))
                return {Status::unbounded, INFINITY, {}};
            std::vector<Real> x(n);
            for (int i = 0; i < m; ++i)
                if (B[i] >= 0 && B[i] < n)
                    x[B[i]] = D[i][n + 1];
            return {Status::optimal, D[m][n + 1], x};
        }
    };
}
```
