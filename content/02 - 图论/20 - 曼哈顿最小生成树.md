# 曼哈顿最小生成树

坐标为整数；距离和用 i64 表示；本实现适用于约 n<=2000。 点按下标 0..n-1 编号；n<2 时总权为 0、边集为空。

O(n^2 log n) 时间，O(n^2) 空间。

```cpp
namespace mstkruskal {
    // Kruskal 所需并查集
    struct DSU {
        std::vector<int> p, sz;
        // 创建 n 个独立集合
        explicit DSU(int n) : p(n), sz(n, 1) {
            for (int i = 0; i < n; i++) {
                p[i] = i;
            }
        }
        // 返回 x 所在集合的代表
        int find(int x) {
            while (p[x] != x) {
                p[x] = p[p[x]];
                x = p[x];
            }
            return x;
        }
        // 合并两个集合，成功返回 true
        bool unite(int a, int b) {
            a = find(a);
            b = find(b);
            if (a == b) {
                return false;
            }
            if (sz[a] < sz[b]) {
                std::swap(a, b);
            }
            p[b] = a;
            sz[a] += sz[b];
            return true;
        }
    };
} // namespace mstkruskal

// 返回最小生成树的所有边 {u, v, w}（w 为曼哈顿距离）
std::vector<std::tuple<int, int, i64>>
manhattanMSTEdges(const std::vector<std::pair<i64, i64>> &pts) {
    int n = (int)pts.size();
    std::vector<std::tuple<int, int, i64>> edges;
    if (n < 2) {
        return edges;
    }
    edges.reserve((size_t)n * (n - 1) / 2);
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            i64 w = std::abs(pts[i].first - pts[j].first) + std::abs(pts[i].second - pts[j].second);
            edges.emplace_back(i, j, w);
        }
    }
    std::sort(edges.begin(), edges.end(),
              [](const auto &a, const auto &b) { return std::get<2>(a) < std::get<2>(b); });
    mstkruskal::DSU dsu(n);
    std::vector<std::tuple<int, int, i64>> mst;
    for (auto &[u, v, w] : edges) {
        if (dsu.unite(u, v)) {
            mst.emplace_back(u, v, w);
            if ((int)mst.size() == n - 1) {
                break;
            }
        }
    }
    return mst;
}

// 返回最小生成树总边权
i64 manhattanMST(const std::vector<std::pair<i64, i64>> &pts) {
    i64 s = 0;
    for (auto &[u, v, w] : manhattanMSTEdges(pts)) {
        s += w;
    }
    return s;
}
```
