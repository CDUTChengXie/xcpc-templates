# 最短次短路与路径计数

有向或无向图，每条边 0<w<TwoShortestPaths::INF；有限距离须小于 INF；1<=mod<=INT_MAX。 点 0..n-1；s 为源点，t 为目标点；按边序列区分走法，允许重复点/边，重边各自计数。

dist[t][0/1] 为最短/严格次短距离，缺项为 INF；count[t][0/1] 为对应走法数模 mod。 count=0 也可能只因取模，须用距离判可达；无向自环只加一次；源点空路径长度 0、计数 1%mod。

O((n+m)log(n+m)) 时间，O(n+m) 空间。

```cpp
#ifndef XCPC_TWO_SHORTEST_PATHS
#define XCPC_TWO_SHORTEST_PATHS
namespace TwoShortestPaths {
    inline constexpr i64 INF = 4000000000000000000LL; // INF是哨兵，所有保留的有限距离严格小于它
    // 单源最短和严格次短路结果
    struct Result {
        std::vector<std::array<i64, 2>> dist; // dist[v][0/1]：两个不同长度，缺项为INF
        std::vector<std::array<int, 2>>
            count; // count[v][0/1]：相应边序列数mod，不能据此判断是否可达
    };
    // 单源最短/严格次短路及方案数
    class Graph {
        int n, mod;
        std::vector<std::vector<std::pair<int, i64>>> adj;
        void checkEdge(int u, int v, i64 w) const { // 私有输入校验；拒绝越界、零权、负权和哨兵边权
            if (u < 0 || u >= n || v < 0 || v >= n)
                throw std::out_of_range("edge vertex");
            if (w <= 0 || w >= INF)
                throw std::invalid_argument("edge weight must be in (0, INF)");
        }

      public:
        // 创建 n 个点并指定计数模数
        explicit Graph(int vertices, int modulus = 1000000007) : n(vertices), mod(modulus) {
            if (n < 0 || mod <= 0)
                throw std::invalid_argument("n >= 0 and mod > 0 required");
            adj.resize(n);
        }
        // 加入一条有向正权边
        void addEdge(int u, int v, i64 w) {
            checkEdge(u, v, w);
            adj[u].push_back({v, w});
        }
        // 加入一条无向正权边
        void addUndirectedEdge(int u, int v, i64 w) {
            checkEdge(u, v, w);
            adj[u].push_back({v, w});
            if (u != v)
                adj[v].push_back({u, w});
        }
        // 返回所有点的最短/严格次短距离与方案数
        Result work(int source) const {
            if (source < 0 || source >= n)
                throw std::out_of_range("source vertex");
            Result r{std::vector<std::array<i64, 2>>(n, {INF, INF}),
                     std::vector<std::array<int, 2>>(n, {0, 0})};
            using State =
                std::tuple<i64, int, int>; // 堆记录{距离,顶点,最短/次短层}，过期状态不转移
            std::priority_queue<State, std::vector<State>, std::greater<State>> q;
            r.dist[source][0] = 0;
            r.count[source][0] = 1 % mod;
            q.push({0, source, 0});
            while (!q.empty()) {
                auto [d, u, rank] = q.top();
                q.pop();
                if (d != r.dist[u][rank])
                    continue;
                int ways =
                    r.count[u][rank]; // 正权保证所有等长贡献都来自更小距离，在本次出堆前已到齐
                for (auto [v, w] : adj[u]) {
                    if (w >= INF - d)
                        continue; // 不计算达到哨兵的长度，先减再加避免溢出
                    i64 candidate = d + w;
                    if (candidate < r.dist[v][0]) {
                        r.dist[v][1] = r.dist[v][0];
                        r.count[v][1] = r.count[v][0]; // 原最短降为次短，旧次短舍弃
                        if (r.dist[v][1] < INF)
                            q.push({r.dist[v][1], v, 1});
                        r.dist[v][0] = candidate;
                        r.count[v][0] = ways;
                        q.push({candidate, v, 0});
                    } else if (candidate == r.dist[v][0]) {
                        r.count[v][0] =
                            int((i64(r.count[v][0]) + ways) % mod); // 同距离只合并计数，不重复出堆
                    } else if (candidate < r.dist[v][1]) {
                        r.dist[v][1] = candidate;
                        r.count[v][1] = ways;
                        q.push({candidate, v, 1});
                    } else if (candidate == r.dist[v][1]) {
                        r.count[v][1] = int((i64(r.count[v][1]) + ways) % mod);
                    }
                }
            }
            return r;
        }
    };
} // namespace TwoShortestPaths
#endif
```
