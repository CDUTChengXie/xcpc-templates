# 线段树优化建图（点区间与区间区间）

n>0；有向边权非负且小于 2^60；所需有限最短距离小于 2^60。 原点 0..n-1；区间 [l,r) 左闭右开，空区间不加边；w 为每条批量边的权值。 两区间连边方向为第一段到第二段；加完边后 distances(s)，只返回原 n 个点的距离。

建图 O(n+qlog n)；最短路 O((n+qlog n)log(n+q))；空间 O(n+qlog n)。

```cpp
// 线段树优化区间连边图
struct RangeGraph {
    int n;
    std::vector<std::vector<std::pair<int, i64>>> adj;
    std::vector<int> down, up;
    // 创建 n 个原始点，支持点与区间之间连边
    explicit RangeGraph(int n) : n(n), adj(n), down(4 * std::max(1, n)), up(4 * std::max(1, n)) {
        assert(n > 0);
        build(1, 0, n);
    }
    // 添加原始点 u 到 v 的有向边，无返回值
    void addEdge(int u, int v, i64 w) {
        checkPoint(u);
        checkPoint(v);
        checkWeight(w);
        adj[u].push_back({v, w});
    }
    // 添加 u 到 [l,r) 内每个点的有向边，无返回值
    void pointToRange(int u, int l, int r, i64 w) {
        checkPoint(u);
        checkRange(l, r);
        checkWeight(w);
        for (int v : cover(l, r, false))
            adj[u].push_back({v, w});
    }
    // 添加 [l,r) 内每个点到 v 的有向边，无返回值
    void rangeToPoint(int l, int r, int v, i64 w) {
        checkPoint(v);
        checkRange(l, r);
        checkWeight(w);
        for (int u : cover(l, r, true))
            adj[u].push_back({v, w});
    }
    // 添加 [l1,r1) 到 [l2,r2) 的所有有向边，边权均为 w，无返回值
    void rangeToRange(int l1, int r1, int l2, int r2, i64 w) {
        checkRange(l1, r1);
        checkRange(l2, r2);
        checkWeight(w);
        if (l1 == r1 || l2 == r2)
            return;
        int connector = int(adj.size());
        adj.emplace_back();
        for (int u : cover(l1, r1, true))
            adj[u].push_back({connector, w});
        for (int v : cover(l2, r2, false))
            adj[connector].push_back({v, 0});
    }
    // 返回 source 到原始 n 个点的最短距离；不可达为 2^60
    std::vector<i64> distances(int source) const {
        checkPoint(source);
        constexpr i64 BIG = 1LL << 60;
        std::vector<i64> d(adj.size(), BIG);
        d[source] = 0;
        using Node = std::pair<i64, int>;
        std::priority_queue<Node, std::vector<Node>, std::greater<Node>> q;
        q.push({0, source});
        while (!q.empty()) {
            auto [du, u] = q.top();
            q.pop();
            if (du != d[u])
                continue;
            for (auto [v, w] : adj[u])
                if (w < BIG - du && d[v] > du + w) {
                    d[v] = du + w;
                    q.push({d[v], v});
                }
        }
        d.resize(n);
        return d;
    }

  private:
    void checkPoint(int u) const {
        assert(u >= 0 && u < n);
    }
    void checkRange(int l, int r) const {
        assert(l >= 0 && l <= r && r <= n);
    }
    static void checkWeight(i64 w) {
        assert(w >= 0 && w < (1LL << 60));
    }
    void build(int p, int l, int r) {
        if (r - l == 1) {
            down[p] = up[p] = l;
            return;
        }
        int mid = (l + r) / 2;
        build(p * 2, l, mid);
        build(p * 2 + 1, mid, r);
        down[p] = int(adj.size());
        adj.emplace_back();
        up[p] = int(adj.size());
        adj.emplace_back();
        adj[down[p]].push_back({down[p * 2], 0});
        adj[down[p]].push_back({down[p * 2 + 1], 0});
        adj[up[p * 2]].push_back({up[p], 0});
        adj[up[p * 2 + 1]].push_back({up[p], 0});
    }
    std::vector<int> cover(int l, int r, bool upward) const {
        std::vector<int> result;
        auto dfs = [&](auto &&self, int p, int a, int b) -> void {
            if (r <= a || b <= l)
                return;
            if (l <= a && b <= r) {
                result.push_back(upward ? up[p] : down[p]);
                return;
            }
            int m = (a + b) / 2;
            self(self, p * 2, a, m);
            self(self, p * 2 + 1, m, b);
        };
        dfs(dfs, 1, 0, n);
        return result;
    }
};
```
