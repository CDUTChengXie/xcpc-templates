# 欧拉路径

n>=1；每个对象只调用一次 work；有解须满足度数条件且忽略孤立点后连通。 点下标 0..n-1；directed=true 为有向图，false 为无向图；重边和自环可用。 无解返回空，无边返回 {0}；有 m 条边时路径含 m+1 个顶点，回路首尾相同。

O(n+m log m) 时间，O(n+m) 空间。

```cpp
// 有向图或无向图的字典序最小欧拉路径
struct EulerPath {
    int n;
    bool directed;
    std::vector<std::vector<std::pair<int, int>>> adj;
    std::vector<int> it, used;
    std::vector<int> path;

    // 创建 n 个点的有向或无向图
    EulerPath(int n_, bool directed_) : n(n_), directed(directed_), adj(n_), it(n_), used(n_) {}

    // 加入一条独立边
    void addEdge(int u, int v) {
        int id = used.size();
        used.push_back(0);
        adj[u].push_back({v, id});
        if (!directed) {
            adj[v].push_back({u, id});
        }
    }

    // Hierholzer DFS；仅供 work 调用
    void dfs(int u) {
        std::vector<int> pending{u};
        while (!pending.empty()) {
            int vertex = pending.back();
            while (it[vertex] < int(adj[vertex].size()) && used[adj[vertex][it[vertex]].second])
                ++it[vertex];
            if (it[vertex] == int(adj[vertex].size())) {
                path.push_back(vertex);
                pending.pop_back();
            } else {
                auto [next, edge] = adj[vertex][it[vertex]++];
                used[edge] = 1;
                pending.push_back(next);
            }
        }
    }

    // 返回字典序最小的欧拉路径，无解为空
    std::vector<int> work() {
        int start = 0;
        bool found = false;
        for (int u = 0; u < n; u++) {
            std::sort(adj[u].begin(), adj[u].end());
            if (!adj[u].empty() && !found) {
                start = u;
                found = true;
            }
        }
        if (directed) {
            std::vector<int> in(n, 0), out(n, 0);
            for (int u = 0; u < n; u++) {
                out[u] = adj[u].size();
                for (auto [v, id] : adj[u]) {
                    in[v]++;
                }
            }
            int s = 0, t = 0;
            for (int u = 0; u < n; u++) {
                if (in[u] == out[u]) {
                    continue;
                }
                if (in[u] + 1 == out[u]) {
                    if (++s > 1) {
                        return {};
                    }
                    start = u;
                } else if (out[u] + 1 == in[u]) {
                    t++;
                } else {
                    return {};
                }
            }
            if (s != t) {
                return {};
            }
        } else {
            std::vector<int> deg(n, 0);
            for (int u = 0; u < n; u++) {
                for (auto [v, id] : adj[u]) {
                    deg[u]++;
                }
            }
            int odd = 0;
            for (int u = 0; u < n; u++) {
                if (deg[u] % 2) {
                    if (++odd > 2) {
                        return {};
                    }
                    if (odd == 1)
                        start = u;
                }
            }
        }
        dfs(start);
        std::reverse(path.begin(), path.end());
        if (path.size() != used.size() - n + 1)
            return {};
        return path;
    }
};
```
