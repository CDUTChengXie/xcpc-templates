# 点双连通分量（顶点与边集）

先 `VertexBiconnected(n)`，每条无向边 `addEdge(u,v)` 一次；加完后 `work()`，再读分量和割点标记。

每条无向边只加一次； 点、输入边号和分量号均从 0 开始；重边允许，自环单独成分量，孤点形成无边的单点分量。 加完边再 work，可重复计算；顶点可属于多个分量，但每条输入边恰好属于一个分量。

O(n+m) 时间和空间。

```cpp
// 点双连通分量与割点
class VertexBiconnected {
    int n, timer = 0;
    std::vector<std::pair<int, int>> edges;
    std::vector<std::vector<std::pair<int, int>>> a;
    std::vector<int> dfn, low, stack, mark;
    void emit(int last) {
        int cid = edgeComponents.size();
        std::vector<int> vs, es;
        while (!stack.empty()) {
            int id = stack.back();
            stack.pop_back();
            es.push_back(id);
            auto [u, v] = edges[id];
            for (int x : {u, v})
                if (mark[x] != cid) {
                    mark[x] = cid;
                    vs.push_back(x);
                }
            if (id == last)
                break;
        }
        vertexComponents.push_back(std::move(vs));
        edgeComponents.push_back(std::move(es));
    }
    void dfs(int u, int parentEdge) {
        struct Frame {
            int vertex, edge, children;
            size_t next;
        };
        std::vector<Frame> frames{{u, parentEdge, 0, 0}};
        dfn[u] = low[u] = ++timer;
        while (!frames.empty()) {
            auto &frame = frames.back();
            int vertex = frame.vertex;
            if (frame.next < a[vertex].size()) {
                auto [next, edge] = a[vertex][frame.next++];
                if (edge == frame.edge)
                    continue;
                if (!dfn[next]) {
                    ++frame.children;
                    stack.push_back(edge);
                    dfn[next] = low[next] = ++timer;
                    frames.push_back({next, edge, 0, 0});
                } else if (dfn[next] < dfn[vertex]) {
                    stack.push_back(edge);
                    low[vertex] = std::min(low[vertex], dfn[next]);
                }
            } else {
                int edge = frame.edge;
                frames.pop_back();
                if (frames.empty())
                    continue;
                auto &parent = frames.back();
                low[parent.vertex] = std::min(low[parent.vertex], low[vertex]);
                if (low[vertex] >= dfn[parent.vertex]) {
                    if (parent.edge != -1 || parent.children > 1)
                        isCut[parent.vertex] = true;
                    emit(edge);
                }
            }
        }
    }

  public:
    // work 后，vertexComponents[c] 和 edgeComponents[c] 是同一分量的顶点编号和输入边号
    std::vector<std::vector<int>> vertexComponents, edgeComponents;
    std::vector<bool> isCut; // isCut[u] 表示原顶点 u 是不是割点
    // 创建 n 个点，点编号0..n-1
    explicit VertexBiconnected(int n) : n(n), a(n) {
        if (n < 0)
            throw std::invalid_argument("negative n");
    }
    // 加一条无向边，返回原边编号；重边、自环分别计数
    int addEdge(int u, int v) {
        if (u < 0 || u >= n || v < 0 || v >= n)
            throw std::out_of_range("vertex");
        int id = edges.size();
        edges.push_back({u, v});
        if (u != v) {
            a[u].push_back({v, id});
            a[v].push_back({u, id});
        }
        return id;
    }
    // 重新计算全部分量和割点；结果保存在三个公开数组，无返回值
    void work() {
        timer = 0;
        dfn.assign(n, 0);
        low.assign(n, 0);
        mark.assign(n, -1);
        isCut.assign(n, false);
        stack.clear();
        vertexComponents.clear();
        edgeComponents.clear();
        std::vector<bool> hasEdge(n);
        for (int id = 0; id < int(edges.size()); ++id) {
            auto [u, v] = edges[id];
            hasEdge[u] = hasEdge[v] = true;
            if (u == v) {
                vertexComponents.push_back({u});
                edgeComponents.push_back({id});
            }
        }
        for (int u = 0; u < n; ++u)
            if (!dfn[u]) {
                dfs(u, -1);
                if (!hasEdge[u]) {
                    vertexComponents.push_back({u});
                    edgeComponents.push_back({});
                }
            }
    }
};
```
