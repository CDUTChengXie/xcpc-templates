# 正反向区间等价约束

区间必须等长，可重叠；所有操作只增加等价约束。 位置0..n-1，区间[l,r)左闭右开；空区间允许。

约束处理总 O((n log n+q log n)α(n))；components 单次 O(n log n)；空间 O(n log n)。

```cpp
namespace xcpc_range_equal {
    // 支持逐层区间合并的内部并查集
    class RangeDSU {
        vector<vector<int>> p, sz;

      public:
        // 初始化所有可容纳长度 2^k 区间的层；空数组合法。
        explicit RangeDSU(int n) {
            assert(n >= 0);
            for (int len = 1; len <= max(1, n); len <<= 1) {
                int m = n - len + 1;
                if (m < 0)
                    break;
                p.emplace_back(m);
                sz.emplace_back(m, 1);
                iota(p.back().begin(), p.back().end(), 0);
                if (len > (1 << 29))
                    break;
            }
        }
        // 返回第 k 层位置 x 所属块的代表。
        int find(int k, int x) {
            assert(0 <= k && k < int(p.size()) && 0 <= x && x < int(p[k].size()));
            return p[k][x] == x ? x : p[k][x] = find(k, p[k][x]);
        }

      private:
        // 合并第 k 层两个长度 2^k 区间的代表；成功返回 true。
        bool merge(int k, int x, int y) {
            x = find(k, x);
            y = find(k, y);
            if (x == y)
                return false;
            if (sz[k][x] < sz[k][y])
                swap(x, y);
            p[k][y] = x;
            sz[k][x] += sz[k][y];
            return true;
        }

      public:
        // 把两个长度 2^k 区间逐层对应合并；每次底层合并成功后调用 callback。
        template <class F> void mergeRange(int k, int x, int y, F callback) {
            assert(0 <= k && k < int(p.size()) && x >= 0 && y >= 0 &&
                   x + (1 << k) <= int(p[0].size()) && y + (1 << k) <= int(p[0].size()));
            if (find(k, x) == find(k, y))
                return;
            if (k == 0) {
                int a = find(0, x), b = find(0, y);
                if (merge(0, a, b))
                    callback(a, b);
                return;
            }
            mergeRange(k - 1, x, y, callback);
            mergeRange(k - 1, x + (1 << (k - 1)), y + (1 << (k - 1)), callback);
            merge(k, x, y);
        }
    };
}
// 正反向区间等价约束
class RangeEqualConstraints {
    int n;
    xcpc_range_equal::RangeDSU dsu;
    static int checkedSize(int n) {
        if (n < 0 || n > INT_MAX / 2)
            throw std::invalid_argument("invalid n");
        return 2 * n;
    }

  public:
    // 创建 n 个位置；额外镜像层用于反向约束
    explicit RangeEqualConstraints(int n) : n(n), dsu(checkedSize(n)) {
        for (int i = 0; i < n; ++i)
            dsu.mergeRange(0, i, 2 * n - 1 - i, [](int, int) {});
    }
    // 把两个等长半开区间逐位合并；reversed=true时反向对齐，无返回值
    void merge(int l, int r, int a, int b, bool reversed = false) {
        if (l < 0 || l > r || r > n || a < 0 || a > b || b > n || r - l != b - a)
            throw std::invalid_argument("invalid ranges");
        int len = r - l;
        if (!len)
            return;
        int k = std::bit_width(unsigned(len)) - 1, block = 1 << k;
        int start = reversed ? 2 * n - b : a;
        dsu.mergeRange(k, l, start, [](int, int) {});
        dsu.mergeRange(k, r - block, start + len - block, [](int, int) {});
    }
    // 两位置被约束相等返回true，位置范围0..n-1
    bool same(int i, int j) {
        if (i < 0 || i >= n || j < 0 || j >= n)
            throw std::out_of_range("position");
        return dsu.find(0, i) == dsu.find(0, j);
    }
    // 返回原 n 个位置的等价类数
    int components() {
        std::set<int> roots;
        for (int i = 0; i < n; ++i)
            roots.insert(dsu.find(0, i));
        return roots.size();
    }
};
```
