# 最小斯坦纳树

无向图非负边权；指定点去重后 k<=20；2^k*n<=1e8 个距离单元，且须装得下内存。 点下标 0..n-1；无向边加两个方向；有限答案小于 2^59；指定点不超过 1 个时返回 0。

O(3^k n+2^k(n+m)log n) 时间，O(2^k n) 空间。

```cpp
namespace SteinerTree {
    constexpr i64 UNREACHABLE = (1LL << 60);
    // 连接 terminals 中所有点，可经过其他点；返回最小边权和，不能连通返回 UNREACHABLE
    inline i64 work(const std::vector<std::vector<std::pair<int, i64>>> &adj,
                    std::vector<int> terminals) {
        int n = int(adj.size());
        std::sort(terminals.begin(), terminals.end());
        terminals.erase(std::unique(terminals.begin(), terminals.end()), terminals.end());
        int k = int(terminals.size());
        assert(k <= 20);
        for (int u : terminals)
            assert(u >= 0 && u < n);
        for (auto &row : adj)
            for (auto [v, w] : row)
                assert(v >= 0 && v < n && w >= 0 && w < UNREACHABLE);
        if (k <= 1)
            return 0;
        size_t cells = size_t(1 << k) * n;
        assert(cells <= size_t(100000000));
        std::vector<std::vector<i64>> dp(1 << k, std::vector<i64>(n, UNREACHABLE));
        for (int i = 0; i < k; ++i)
            dp[1 << i][terminals[i]] = 0;
        using Node = std::pair<i64, int>;
        for (int mask = 1; mask < (1 << k); ++mask) {
            for (int sub = (mask - 1) & mask; sub; sub = (sub - 1) & mask) {
                int other = mask ^ sub;
                if (sub > other)
                    continue;
                for (int u = 0; u < n; ++u)
                    if (dp[sub][u] != UNREACHABLE && dp[other][u] != UNREACHABLE)
                        dp[mask][u] = std::min(dp[mask][u], dp[sub][u] + dp[other][u]);
            }
            std::priority_queue<Node, std::vector<Node>, std::greater<Node>> q;
            for (int u = 0; u < n; ++u)
                if (dp[mask][u] != UNREACHABLE)
                    q.push({dp[mask][u], u});
            while (!q.empty()) {
                auto [d, u] = q.top();
                q.pop();
                if (d != dp[mask][u])
                    continue;
                for (auto [v, w] : adj[u])
                    if (w < UNREACHABLE - d && dp[mask][v] > d + w) {
                        dp[mask][v] = d + w;
                        q.push({d + w, v});
                    }
            }
        }
        return *std::min_element(dp.back().begin(), dp.back().end());
    }
}
```
