# 负边与全源最短路（Bellman-Ford / Floyd / Johnson / 01BFS）

有向边 {u,v,w}，w 可为负；无向边加两个方向；(n+m+1)²×最大边权绝对值小于 2^58。 zeroOneBFS 的邻接表权值只能为 0/1；s 为源点，t 为目标点。 点下标 0..n-1；无路径为 ShortestPaths::UNREACHABLE，受负环影响可无限变小为 NEGATIVE。

Bellman-Ford 只标记 s 可达负环；Floyd 标记受负环影响的点对；Johnson 任意负环即 nullopt。

Bellman-Ford O(nm)；Floyd O(n³)；Johnson O(nm+n(n+m)log n)；01BFS O(n+m)。

```cpp
namespace ShortestPaths {
    constexpr i64 UNREACHABLE = (1LL << 60);
    constexpr i64 NEGATIVE = -UNREACHABLE;
    // 一条带权有向边
    struct Edge {
        int u, v;
        i64 w;
    };
    // 单源最短路结果及负环标记
    struct BellmanResult {
        std::vector<i64> distance; // distance[u] 为源点到 u 的距离；不可达为 UNREACHABLE，受负环影响为 NEGATIVE
        bool negativeCycle;
    };

    // 从 source 求最短路；不可达为 UNREACHABLE，受可达负环影响为 NEGATIVE
    inline BellmanResult bellmanFord(int n, const std::vector<Edge> &edges, int source) {
        assert(n > 0 && source >= 0 && source < n);
        std::vector<i64> d(n, UNREACHABLE);
        d[source] = 0;
        std::vector<std::vector<int>> adj(n);
        for (auto [u, v, w] : edges) {
            assert(u >= 0 && u < n && v >= 0 && v < n);
            adj[u].push_back(v);
        }
        std::vector<int> affected;
        for (int step = 0; step < n; ++step) {
            bool changed = false;
            for (auto [u, v, w] : edges)
                if (d[u] != UNREACHABLE && d[v] > d[u] + w) {
                    d[v] = d[u] + w;
                    changed = true;
                    if (step == n - 1)
                        affected.push_back(v);
                }
            if (!changed)
                break;
        }
        std::vector<bool> bad(n);
        std::queue<int> q;
        for (int v : affected)
            if (!bad[v]) {
                bad[v] = true;
                q.push(v);
            }
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (int v : adj[u])
                if (!bad[v]) {
                    bad[v] = true;
                    q.push(v);
                }
        }
        for (int u = 0; u < n; ++u)
            if (bad[u])
                d[u] = NEGATIVE;
        return {d, !affected.empty()};
    }

    // 求所有点对最短路；受负环影响的点对为 NEGATIVE，不可达为 UNREACHABLE
    inline std::vector<std::vector<i64>> floyd(int n, const std::vector<Edge> &edges) {
        assert(n >= 0);
        std::vector<std::vector<i64>> d(n, std::vector<i64>(n, UNREACHABLE));
        for (int i = 0; i < n; ++i)
            d[i][i] = 0;
        for (auto [u, v, w] : edges) {
            assert(u >= 0 && u < n && v >= 0 && v < n);
            d[u][v] = std::min(d[u][v], w);
        }
        for (int k = 0; k < n; ++k)
            for (int i = 0; i < n; ++i)
                if (d[i][k] != UNREACHABLE)
                    for (int j = 0; j < n; ++j)
                        if (d[k][j] != UNREACHABLE) {
                            i128 candidate = i128(d[i][k]) + d[k][j];
                            d[i][j] =
                                std::min(d[i][j], i64(std::max(candidate, i128(NEGATIVE + 1))));
                        }
        std::vector<int> negative;
        for (int k = 0; k < n; ++k)
            if (d[k][k] < 0)
                negative.push_back(k);
        auto finite = d;
        for (int k : negative)
            for (int i = 0; i < n; ++i)
                if (finite[i][k] != UNREACHABLE)
                    for (int j = 0; j < n; ++j)
                        if (finite[k][j] != UNREACHABLE)
                            d[i][j] = NEGATIVE;
        return d;
    }

    // 稀疏图的所有点对最短路；任意负环返回 nullopt，不可达为 UNREACHABLE
    inline std::optional<std::vector<std::vector<i64>>> johnson(int n,
                                                                const std::vector<Edge> &edges) {
        assert(n >= 0);
        auto withSource = edges;
        for (int u = 0; u < n; ++u)
            withSource.push_back({n, u, 0});
        auto result = bellmanFord(n + 1, withSource, n);
        if (result.negativeCycle)
            return std::nullopt;
        auto &h = result.distance;
        std::vector<std::vector<std::pair<int, i64>>> adj(n);
        for (auto [u, v, w] : edges)
            adj[u].push_back({v, w + h[u] - h[v]});
        std::vector<std::vector<i64>> all(n, std::vector<i64>(n, UNREACHABLE));
        for (int s = 0; s < n; ++s) {
            auto &d = all[s];
            d[s] = 0;
            std::priority_queue<std::pair<i64, int>, std::vector<std::pair<i64, int>>,
                                std::greater<std::pair<i64, int>>>
                pq;
            pq.push({0, s});
            while (!pq.empty()) {
                auto [current, u] = pq.top();
                pq.pop();
                if (current != d[u])
                    continue;
                for (auto [v, w] : adj[u])
                    if (d[v] > current + w) {
                        d[v] = current + w;
                        pq.push({d[v], v});
                    }
            }
            for (int u = 0; u < n; ++u)
                if (d[u] != UNREACHABLE)
                    d[u] += h[u] - h[s];
        }
        return all;
    }

    // 边权仅为 0 或 1；返回源点到各点的距离，不可达为 UNREACHABLE
    inline std::vector<i64> zeroOneBFS(const std::vector<std::vector<std::pair<int, int>>> &adj,
                                       int source) {
        int n = int(adj.size());
        assert(source >= 0 && source < n);
        std::vector<i64> d(n, UNREACHABLE);
        d[source] = 0;
        std::deque<int> q{source};
        while (!q.empty()) {
            int u = q.front();
            q.pop_front();
            for (auto [v, w] : adj[u]) {
                assert(v >= 0 && v < n && (w == 0 || w == 1));
                if (d[v] > d[u] + w) {
                    d[v] = d[u] + w;
                    if (w == 0)
                        q.push_front(v);
                    else
                        q.push_back(v);
                }
            }
        }
        return d;
    }
}
```
