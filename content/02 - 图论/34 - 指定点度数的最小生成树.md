# 指定点度数的最小生成树

无向图，n>=1，整数边权可为负；最终总权须可用 i64 表示，否则抛异常。 点 0..n-1；s 为限制度数的点，k 为所需树边数；重边保留，自环忽略。 只返回总权，不返回边集；不连通或度数不可行返回 nullopt。

O(m log m log(n*最大绝对边权+1)) 时间，O(n+m) 空间。

```cpp
// 指定点度数约束下的最小生成树
class DegreeConstrainedMST {
    struct Edge {
        int u, v;
        i64 w;
        bool special;
    };
    int n, s;
    std::vector<Edge> e;
    struct Trial {
        bool connected;
        __int128 cost;
        int degree;
    };
    Trial run(__int128 penalty) const {
        std::vector<int> ids(e.size()), p(n), sz(n, 1);
        std::iota(ids.begin(), ids.end(), 0);
        std::iota(p.begin(), p.end(), 0);
        std::sort(ids.begin(), ids.end(), [&](int a, int b) {
            auto x = __int128(e[a].w) + penalty * e[a].special,
                 y = __int128(e[b].w) + penalty * e[b].special;
            return x != y ? x < y : e[a].special > e[b].special;
        });
        auto find = [&](auto &&self, int x) -> int {
            return p[x] == x ? x : p[x] = self(self, p[x]);
        };
        Trial r{n == 1, 0, 0};
        int count = 0;
        for (int id : ids) {
            auto x = e[id];
            int u = find(find, x.u), v = find(find, x.v);
            if (u == v)
                continue;
            if (sz[u] < sz[v])
                std::swap(u, v);
            p[v] = u;
            sz[u] += sz[v];
            r.cost += __int128(x.w) + penalty * x.special;
            r.degree += x.special;
            ++count;
        }
        r.connected = count == n - 1;
        return r;
    }

  public:
    // 创建图，限制点 s 的度数；点编号 0..n-1
    DegreeConstrainedMST(int n, int s) : n(n), s(s) {
        if (n < 1 || s < 0 || s >= n)
            throw std::invalid_argument("invalid n or s");
    }
    // 加无向边，自环忽略，重边保留，无返回值
    void addEdge(int u, int v, i64 w) {
        if (u < 0 || u >= n || v < 0 || v >= n)
            throw std::out_of_range("vertex");
        if (u != v)
            e.push_back({u, v, w, u == s || v == s});
    }
    // 返回指定点恰好 k 度的最小权重；不可行返回空 optional
    std::optional<i64> solve(int k) const {
        if (k < 0 || k >= n)
            return std::nullopt;
        if (n == 1)
            return k == 0 ? std::optional<i64>(0) : std::nullopt;
        __int128 maxWeight = 0;
        for (auto x : e) {
            __int128 w = x.w;
            if (w < 0)
                w = -w;
            maxWeight = std::max(maxWeight, w);
        }
        __int128 bound = 2 * __int128(n - 1) * maxWeight + 1, lo = -bound, hi = bound;
        auto maximum = run(lo), minimum = run(hi);
        if (!maximum.connected || maximum.degree < k || minimum.degree > k)
            return std::nullopt;
        while (lo < hi) {
            __int128 mid = lo + (hi - lo + 1) / 2;
            if (run(mid).degree >= k)
                lo = mid;
            else
                hi = mid - 1;
        }
        auto r = run(lo);
        __int128 cost = r.cost - lo * k;
        if (cost < std::numeric_limits<i64>::min() || cost > std::numeric_limits<i64>::max())
            throw std::overflow_error("MST sum exceeds i64");
        return i64(cost);
    }
};
```
