# 最小环与环见证

minimumCycle/findTriangle 要求无自环、无重边；最小环边权 0<w<CycleGraphs::INF，mod 为正 int。 anyCycle 允许重边和自环；backEdgeCount 只用于有向图，结果不是简单环总数。

点和输入边号从 0 开始；directed=true 为有向，false 为无向，无向边只输入一次。 最小环有限权和须小于 INF，count 对 mod 取模；默认计有向二元环，可用最后参数 false 排除。 无环 Minimum={INF,0}；count=0 也可能因取模；无向环旋转/反向同算一个，有向仅旋转同算。

Witness 的 vertices 不重复首点，edgeIds[i] 连接 vertices[i] 到下一个点（末项回首点）；无环两数组为空。

minimumCycle、findTriangle O(n^3+m) 时间/O(n^2) 空间；anyCycle、backEdgeCount O(n+m)。

```cpp
#ifndef XCPC_MINIMUM_CYCLE_AND_WITNESS
#define XCPC_MINIMUM_CYCLE_AND_WITNESS
namespace CycleGraphs {
    inline constexpr i64 INF = 4000000000000000000LL; // 有限答案严格小于 INF
    // 一条带权输入边
    struct Edge {
        int u, v;
        i64 weight;
    };
    // 最小环总权和方案数
    struct Minimum {
        i64 weight = INF;
        int count = 0;
    };
    // 一个环的顶点和输入边编号
    struct Witness {
        // vertices 按环排列，不重复首点；edgeIds[i] 为从该点到下一点的原边号，末项回到首点
        std::vector<int> vertices, edgeIds;
    };

    // 返回最小环权和方案数
    inline Minimum minimumCycle(int n, const std::vector<Edge> &edges, bool directed,
                                int mod = 1000000007, bool includeTwoCycles = true) {
        if (n < 0 || mod <= 0)
            throw std::invalid_argument("n >= 0 and mod > 0 required");
        std::vector<std::vector<i64>> a(n, std::vector<i64>(n, INF)); // 闭环必须使用原图边
        for (auto [u, v, w] : edges) {
            if (u < 0 || u >= n || v < 0 || v >= n)
                throw std::out_of_range("edge vertex");
            if (u == v || w <= 0 || w >= INF || a[u][v] != INF)
                throw std::invalid_argument("positive simple graph required");
            a[u][v] = w;
            if (!directed)
                a[v][u] = w;
        }
        auto dist = a;
        std::vector<std::vector<int>> count(n, std::vector<int>(n));
        for (int i = 0; i < n; ++i)
            for (int j = 0; j < n; ++j) {
                if (i == j)
                    dist[i][j] = 0; // 对角计数保持 0
                else if (a[i][j] < INF)
                    count[i][j] = 1 % mod;
            }
        Minimum answer;
        if (directed && includeTwoCycles)
            for (int u = 0; u < n; ++u)
                for (int v = u + 1; v < n; ++v) {
                    if (a[u][v] == INF || a[v][u] == INF)
                        continue;
                    __int128 candidate = __int128(a[u][v]) + a[v][u]; // 每对 u<v 只计一个二元环
                    if (candidate >= INF)
                        continue;
                    if (candidate < answer.weight)
                        answer = {i64(candidate), 1 % mod};
                    else if (candidate == answer.weight)
                        answer.count = int((i64(answer.count) + 1) % mod);
                }
        for (int k = 0; k < n; ++k) {
            // 闭环前距离只允许内部顶点编号小于 k
            for (int i = 0; i < k; ++i)
                if (a[k][i] < INF) {
                    for (int j = 0; j < k; ++j)
                        if (i != j && (directed || i < j) && dist[i][j] < INF && a[j][k] < INF) {
                            __int128 candidate = __int128(a[k][i]) + dist[i][j] + a[j][k];
                            if (candidate >= INF)
                                continue;
                            if (candidate < answer.weight)
                                answer = {i64(candidate), count[i][j]};
                            else if (candidate == answer.weight)
                                answer.count = int((i64(answer.count) + count[i][j]) % mod);
                        }
                }
            for (int i = 0; i < n; ++i)
                if (i != k && dist[i][k] < INF) {
                    for (int j = 0; j < n; ++j)
                        if (j != k && i != j && dist[k][j] < INF) {
                            __int128 candidate = __int128(dist[i][k]) + dist[k][j];
                            if (candidate >= INF)
                                continue;
                            int ways = int(i64(count[i][k]) * count[k][j] % mod);
                            if (candidate < dist[i][j]) {
                                dist[i][j] = i64(candidate);
                                count[i][j] = ways;
                            } else if (candidate == dist[i][j])
                                count[i][j] = int((i64(count[i][j]) + ways) % mod);
                        }
                }
        }
        return answer;
    }

    // 返回任意一个环，无环时两向量为空
    inline Witness anyCycle(int n, const std::vector<std::pair<int, int>> &edges, bool directed) {
        if (n < 0 || edges.size() > size_t(INT_MAX))
            throw std::invalid_argument("invalid graph size");
        std::vector<std::vector<std::pair<int, int>>> adj(n);
        for (int id = 0; id < int(edges.size()); ++id) {
            auto [u, v] = edges[id];
            if (u < 0 || u >= n || v < 0 || v >= n)
                throw std::out_of_range("edge vertex");
            adj[u].push_back({v, id});
            if (!directed && u != v)
                adj[v].push_back({u, id});
        }
        std::vector<int> color(n), parent(n, -1), parentEdge(n, -1), stack;
        std::vector<size_t> next(n); // 显式栈避免深链爆栈
        for (int root = 0; root < n; ++root)
            if (!color[root]) {
                stack.push_back(root);
                color[root] = 1;
                while (!stack.empty()) {
                    int u = stack.back();
                    if (next[u] == adj[u].size()) {
                        color[u] = 2;
                        stack.pop_back();
                        continue;
                    }
                    auto [v, id] = adj[u][next[u]++];
                    if (!directed && id == parentEdge[u])
                        continue; // 只跳父边 ID
                    if (!color[v]) {
                        parent[v] = u;
                        parentEdge[v] = id;
                        color[v] = 1;
                        stack.push_back(v);
                    } else if (color[v] == 1) {
                        Witness cycle;
                        for (int x = u; x != v; x = parent[x]) {
                            cycle.vertices.push_back(x);
                            cycle.edgeIds.push_back(parentEdge[x]);
                        }
                        cycle.vertices.push_back(v);
                        std::reverse(cycle.vertices.begin(), cycle.vertices.end());
                        std::reverse(cycle.edgeIds.begin(), cycle.edgeIds.end());
                        cycle.edgeIds.push_back(id);
                        return cycle;
                    }
                }
            }
        return {};
    }
    // 返回一个三元环见证，无三元环时为空
    inline Witness findTriangle(int n, const std::vector<std::pair<int, int>> &edges,
                                bool directed) {
        if (n < 0 || edges.size() > size_t(INT_MAX))
            throw std::invalid_argument("invalid graph size");
        std::vector<std::vector<int>> id(n, std::vector<int>(n, -1));
        for (int e = 0; e < int(edges.size()); ++e) {
            auto [u, v] = edges[e];
            if (u < 0 || u >= n || v < 0 || v >= n)
                throw std::out_of_range("edge vertex");
            if (u == v || id[u][v] != -1)
                throw std::invalid_argument("simple graph required");
            id[u][v] = e;
            if (!directed)
                id[v][u] = e;
        }
        for (int u = 0; u < n; ++u)
            for (int v = u + 1; v < n; ++v)
                for (int w = v + 1; w < n; ++w) {
                    if (id[u][v] != -1 && id[v][w] != -1 && id[w][u] != -1)
                        return {{u, v, w}, {id[u][v], id[v][w], id[w][u]}};
                    if (directed && id[u][w] != -1 && id[w][v] != -1 && id[v][u] != -1)
                        return {{u, w, v}, {id[u][w], id[w][v], id[v][u]}};
                }
        return {};
    }

    // 返回有向 DFS 指向灰色点的边数
    inline i64 backEdgeCount(int n, const std::vector<std::pair<int, int>> &edges) {
        if (n < 0 || edges.size() > size_t(INT_MAX))
            throw std::invalid_argument("invalid graph size");
        std::vector<std::vector<int>> adj(n);
        for (auto [u, v] : edges) {
            if (u < 0 || u >= n || v < 0 || v >= n)
                throw std::out_of_range("edge vertex");
            adj[u].push_back(v);
        }
        std::vector<int> color(n), stack;
        std::vector<size_t> next(n);
        i64 count = 0;
        for (int root = 0; root < n; ++root)
            if (!color[root]) {
                stack.push_back(root);
                color[root] = 1;
                while (!stack.empty()) {
                    int u = stack.back();
                    if (next[u] == adj[u].size()) {
                        color[u] = 2;
                        stack.pop_back();
                        continue;
                    }
                    int v = adj[u][next[u]++];
                    if (color[v] == 1)
                        ++count; // 只计灰色点
                    else if (!color[v]) {
                        color[v] = 1;
                        stack.push_back(v);
                    }
                }
            }
        return count;
    }
} // namespace CycleGraphs
#endif
```
