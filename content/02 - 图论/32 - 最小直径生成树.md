# 最小直径生成树

无向图，n>=1，边权非负；n*最大边权 < 2^58。 点 0..n-1；无向边只加一次，重边分别编号，自环忽略；不连通时 connected=false。 connected 为 true 时 diameter 是实际最小直径，edges 是组成该树的输入边号。

O(n³+n²log n+nm) 时间，O(n²+m) 空间。

```cpp
namespace xcpc_min_diameter {
    // 一条带权无向边
    struct Edge {
        int u, v;
        i64 w;
    };
    // 最小直径生成树结果
    struct Result {
        bool connected = false;
        i64 diameter = 0;
        std::vector<int> edges; // connected 为真时，存选中的输入边号；重边各有独立编号
    };
}
// 最小直径生成树
class MinimumDiameterTree {
    int n;
    std::vector<xcpc_min_diameter::Edge> edges;

  public:
    // 创建 n 个点，编号 0..n-1
    explicit MinimumDiameterTree(int n) : n(n) {
        if (n < 1)
            throw std::invalid_argument("n must be positive");
    }
    // 添加非负无向边，返回边编号；自环不会入树
    int addEdge(int u, int v, i64 w) {
        if (u < 0 || u >= n || v < 0 || v >= n || w < 0 || __int128(n) * w >= (__int128(1) << 58))
            throw std::invalid_argument("invalid edge or too large weight");
        edges.push_back({u, v, w});
        return int(edges.size()) - 1;
    }
    // 返回最小直径及生成树边编号；图不连通时 connected=false
    xcpc_min_diameter::Result solve() const {
        const i64 BIG = i64(1) << 60;
        std::vector<std::vector<i64>> d(n, std::vector<i64>(n, BIG));
        std::vector<std::vector<std::pair<int, int>>> adj(n);
        for (int u = 0; u < n; ++u)
            d[u][u] = 0;
        for (int id = 0; id < int(edges.size()); ++id) {
            auto [u, v, w] = edges[id];
            if (u == v)
                continue;
            d[u][v] = d[v][u] = std::min(d[u][v], w);
            adj[u].push_back({v, id});
            adj[v].push_back({u, id});
        }
        for (int k = 0; k < n; ++k)
            for (int u = 0; u < n; ++u)
                for (int v = 0; v < n; ++v)
                    d[u][v] = std::min(d[u][v], d[u][k] + d[k][v]);
        if (std::find(d[0].begin(), d[0].end(), BIG) != d[0].end())
            return {};
        std::vector<std::vector<int>> order(n, std::vector<int>(n));
        i64 best = BIG, offset = 0;
        int center = -1, centerEdge = -1;
        for (int u = 0; u < n; ++u) {
            std::iota(order[u].begin(), order[u].end(), 0);
            std::sort(order[u].begin(), order[u].end(),
                      [&](int a, int b) { return d[u][a] < d[u][b]; });
            i64 r = 2 * d[u][order[u].back()];
            if (r < best) {
                best = r;
                center = u;
            }
        }
        for (int id = 0; id < int(edges.size()); ++id) {
            auto [u, v, w] = edges[id];
            if (u == v)
                continue;
            std::vector<i64> suffix(n);
            suffix[n - 1] = d[v][order[u][n - 1]];
            for (int j = n - 2; j >= 0; --j)
                suffix[j] = std::max(suffix[j + 1], d[v][order[u][j]]);
            for (int j = 0; j + 1 < n; ++j) {
                i64 a = d[u][order[u][j]], b = suffix[j + 1];
                i64 t = std::clamp(b + w - a, i64(0), 2 * w); // 到 u 的距离乘 2
                i64 r = std::max(2 * a + t, 2 * b + 2 * w - t);
                if (r < best) {
                    best = r;
                    offset = t;
                    centerEdge = id;
                }
            }
        }
        // 从绝对中心出发求最短路树；边上中心相当于一个虚拟根
        std::vector<i64> dist(n, BIG);
        std::vector<int> parent(n, -1);
        using Item = std::pair<i64, int>;
        std::priority_queue<Item, std::vector<Item>, std::greater<Item>> q;
        if (centerEdge == -1) {
            dist[center] = 0;
            q.push({0, center});
        } else {
            auto [u, v, w] = edges[centerEdge];
            dist[u] = offset;
            dist[v] = 2 * w - offset;
            q.push({dist[u], u});
            q.push({dist[v], v});
        }
        while (!q.empty()) {
            auto [du, u] = q.top();
            q.pop();
            if (du != dist[u])
                continue;
            for (auto [v, id] : adj[u])
                if (dist[v] > du + 2 * edges[id].w) {
                    dist[v] = du + 2 * edges[id].w;
                    parent[v] = id;
                    q.push({dist[v], v});
                }
        }
        xcpc_min_diameter::Result result{true, best, {}};
        for (int id : parent)
            if (id != -1)
                result.edges.push_back(id);
        if (centerEdge != -1) {
            auto [u, v, w] = edges[centerEdge];
            if (parent[u] == -1 && parent[v] == -1)
                result.edges.push_back(centerEdge);
        }
        assert(int(result.edges.size()) == n - 1);
        return result;
    }
};
```
