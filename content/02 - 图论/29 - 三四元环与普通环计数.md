# 三四元环与普通环计数

无向简单图，无自环、无重边；环数和路径计数不得溢出 i64；全长度入口 n<=20。 点下标 0..n-1；无向边只传一次；旋转和反向算同一环，全长度结果只在 3..n 有意义。

三四元环 O(n+m√m) 时间/O(n+m) 空间；全长度 O(n²2^n) 时间/O(n2^n) 空间，只限 n<=20。

```cpp
namespace ShortCycles {
    // 无向简单图，edges 每条边只出现一次；返回 {三角形数,四边形数}
    inline std::array<i64, 2> count(int n, const std::vector<std::pair<int, int>> &edges) {
        assert(n >= 0);
        std::vector<std::vector<int>> adj(n);
        for (auto [u, v] : edges) {
            assert(u >= 0 && u < n && v >= 0 && v < n && u != v);
            adj[u].push_back(v);
            adj[v].push_back(u);
        }
        for (auto &row : adj) {
            std::sort(row.begin(), row.end());
            assert(std::adjacent_find(row.begin(), row.end()) == row.end());
        }
        auto lower = [&](int u, int v) {
            return std::pair{adj[u].size(), u} < std::pair{adj[v].size(), v};
        };
        std::vector<std::vector<int>> out(n);
        for (auto [u, v] : edges) {
            if (!lower(u, v))
                std::swap(u, v);
            out[u].push_back(v);
        }
        std::vector<int> mark(n, -1), cnt(n);
        i64 triangles = 0, quads = 0;
        for (int u = 0; u < n; ++u) {
            for (int v : out[u])
                mark[v] = u;
            for (int v : out[u])
                for (int w : out[v])
                    if (mark[w] == u)
                        ++triangles;
        }
        for (int u = 0; u < n; ++u) {
            std::vector<int> touched;
            for (int v : adj[u])
                if (lower(v, u))
                    for (int w : adj[v])
                        if (lower(w, u)) {
                            quads += cnt[w];
                            if (cnt[w]++ == 0)
                                touched.push_back(w);
                        }
            for (int w : touched)
                cnt[w] = 0;
        }
        return {triangles, quads};
    }
    // n<=20；返回长度3到n的简单环数，第len项即len长度；反向和旋转算同一个环
    inline std::vector<i64> allLengths(int n, const std::vector<std::pair<int, int>> &edges) {
        assert(n >= 0 && n <= 20);
        std::vector<unsigned> adj(n);
        for (auto [u, v] : edges) {
            assert(u >= 0 && u < n && v >= 0 && v < n && u != v);
            assert(!(adj[u] >> v & 1));
            adj[u] |= 1u << v;
            adj[v] |= 1u << u;
        }
        std::vector<i64> answer(n + 1);
        for (int s = 0; s < n; ++s) {
            int k = n - s;
            if (k < 3)
                break;
            std::vector<i64> dp(size_t(1 << k) * k);
            dp[k] = 1;
            for (int mask = 1; mask < (1 << k); mask += 2)
                for (int last = 0; last < k; ++last) {
                    i64 ways = dp[size_t(mask) * k + last];
                    if (!ways)
                        continue;
                    if (__builtin_popcount(unsigned(mask)) >= 3 && (adj[s + last] >> s & 1))
                        answer[__builtin_popcount(unsigned(mask))] += ways;
                    unsigned next = (adj[s + last] >> s) & ~unsigned(mask) & ((1u << k) - 1);
                    while (next) {
                        int v = __builtin_ctz(next);
                        next &= next - 1;
                        dp[size_t(mask | (1 << v)) * k + v] += ways;
                    }
                }
        }
        for (auto &x : answer)
            x /= 2;
        return answer;
    }
}
```
