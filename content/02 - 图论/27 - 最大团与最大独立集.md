# 最大团与最大独立集

0≤n≤64；运行时间随图结构变化很大，大图不能套用。 点下标从 0 开始；返回一组最优点编号；多组并列时返回其中一组。

最坏指数时间，空间 O(n²)。

```cpp
// 最大团与最大独立集搜索
struct MaximumClique {
    int n;
    std::vector<u64> adj;
    std::vector<int> best, current;
    // 创建不超过 64 个点的无向简单图
    explicit MaximumClique(int n) : n(n), adj(n) {
        assert(n >= 0 && n <= 64);
    }
    // 添加一条无向边，无返回值；自环忽略
    void addEdge(int u, int v) {
        assert(u >= 0 && u < n && v >= 0 && v < n);
        if (u != v) {
            adj[u] |= u64(1) << v;
            adj[v] |= u64(1) << u;
        }
    }
    // 返回一组最大团的点编号；任意两个被选点之间都有边
    std::vector<int> work() {
        best.clear();
        current.clear();
        expand(n == 64 ? ~u64(0) : (u64(1) << n) - 1);
        return best;
    }
    // 返回一组最大独立集的点编号；任意两个被选点之间都没有边
    std::vector<int> independentSet() const {
        MaximumClique complement(n);
        u64 all = n == 64 ? ~u64(0) : (u64(1) << n) - 1;
        for (int u = 0; u < n; ++u)
            complement.adj[u] = all & ~adj[u] & ~(u64(1) << u);
        return complement.work();
    }

  private:
    void expand(u64 candidates) {
        std::vector<int> order, color;
        u64 left = candidates;
        int c = 0;
        while (left) {
            ++c;
            u64 available = left;
            while (available) {
                int v = __builtin_ctzll(available);
                u64 bit = u64(1) << v;
                order.push_back(v);
                color.push_back(c);
                left &= ~bit;
                available &= ~bit;
                available &= ~adj[v];
            }
        }
        for (int i = int(order.size()) - 1; i >= 0; --i) {
            if (int(current.size()) + color[i] <= int(best.size()))
                return;
            int v = order[i];
            current.push_back(v);
            u64 next = candidates & adj[v];
            if (next)
                expand(next);
            else if (current.size() > best.size())
                best = current;
            current.pop_back();
            candidates &= ~(u64(1) << v);
        }
    }
};
```
