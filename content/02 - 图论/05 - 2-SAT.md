# 2-SAT

先 `TwoSat(n)`，用 `addClause(u,f,v,g)` 添加条件；`satisfiable()` 为 true 后才能读取 `answer()`。

每个子句形如 (u==f)||(v==g)；u/v 是变量编号，f/g 是要求的布尔值。 变量下标 0..n-1；加完子句再求解；answer() 仅在 satisfiable() 为 true 后有效。 forcedValues 返回 `optional<vector<int>>`；无解为 nullopt，-1 表示该变量两值均可取。

普通求解 O(n+m)，固定值查询 O(n(n+m)) 时间；空间 O(n+m)，m 为子句数。

```cpp
// 二值变量可满足性判定与构造
struct TwoSat {
    int n;
    std::vector<std::vector<int>> e;
    std::vector<bool> ans; // satisfiable() 返回 true 后，ans[u] 为变量 u 的一种可行取值

    // 创建n个布尔变量，编号0..n-1
    TwoSat(int n) : n(n), e(2 * n), ans(n) {}

    // 加入两个条件至少一个成立的子句，无返回值
    void addClause(int u, bool f, int v, bool g) {
        e[2 * u + !f].push_back(2 * v + g);
        e[2 * v + !g].push_back(2 * u + f);
    }

    // 可满足返回true，并保存一组取值；否则返回false
    bool satisfiable() {
        std::vector<int> id(2 * n, -1), dfn(2 * n, -1), low(2 * n, -1);
        std::vector<int> stk;
        int now = 0, cnt = 0;
        auto tarjan = [&](int u) {
            struct Frame {
                int vertex, parent;
                size_t next;
            };
            std::vector<Frame> frames{{u, -1, 0}};
            dfn[u] = low[u] = now++;
            stk.push_back(u);
            while (!frames.empty()) {
                auto &frame = frames.back();
                int vertex = frame.vertex;
                if (frame.next < e[vertex].size()) {
                    int next = e[vertex][frame.next++];
                    if (dfn[next] == -1) {
                        dfn[next] = low[next] = now++;
                        stk.push_back(next);
                        frames.push_back({next, vertex, 0});
                    } else if (id[next] == -1) {
                        low[vertex] = std::min(low[vertex], dfn[next]);
                    }
                } else {
                    int parent = frame.parent;
                    if (dfn[vertex] == low[vertex]) {
                        int member;
                        do {
                            member = stk.back();
                            stk.pop_back();
                            id[member] = cnt;
                        } while (member != vertex);
                        ++cnt;
                    }
                    frames.pop_back();
                    if (parent != -1)
                        low[parent] = std::min(low[parent], low[vertex]);
                }
            }
        };
        for (int i = 0; i < 2 * n; ++i) {
            if (dfn[i] == -1) {
                tarjan(i);
            }
        }
        for (int i = 0; i < n; ++i) {
            if (id[2 * i] == id[2 * i + 1]) {
                return false;
            }
            ans[i] = id[2 * i] > id[2 * i + 1];
        }
        return true;
    }

    // 返回上次求得的取值，须先确认satisfiable为true
    std::vector<bool> answer() {
        return ans;
    }
    // 返回文字编号x能否推出文字y；文字2*u表示false，2*u+1表示true
    bool check(int x, int y) const {
        if (x < 0 || x >= 2 * n || y < 0 || y >= 2 * n)
            throw std::out_of_range("literal");
        std::vector<bool> seen(2 * n);
        std::vector<int> q{x};
        seen[x] = true;
        for (size_t i = 0; i < q.size(); ++i)
            for (int v : e[q[i]])
                if (!seen[v]) {
                    seen[v] = true;
                    q.push_back(v);
                }
        return seen[y];
    }
    // 有解时返回各变量被强制的0/1，-1表示可取两值；无解返回空optional
    std::optional<std::vector<int>> forcedValues() {
        if (!satisfiable())
            return std::nullopt;
        std::vector<int> r(n, -1);
        for (int u = 0; u < n; ++u) {
            if (check(2 * u, 2 * u + 1))
                r[u] = 1;
            else if (check(2 * u + 1, 2 * u))
                r[u] = 0;
        }
        return r;
    }
};
```
