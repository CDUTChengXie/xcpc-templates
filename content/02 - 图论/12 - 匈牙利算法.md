# 匈牙利算法

无权二分图，左右两组点已区分，边只从左侧加入右侧。 左点 0..n-1，右点 0..m-1；加完边再 work，未匹配为 -1；重复 work 重新计算。

O(n(E+m)+n+m) 时间，O(n+m+E) 空间，E 为边数，m 为右点数。

```cpp
// 二分图最大匹配
struct Hungarian {
    int n;
    int m;
    std::vector<std::vector<int>> adj;
    // work 后，matchL[u] 是左点 u 配到的右点，matchR[v] 是右点 v 配到的左点；未匹配为 -1
    std::vector<int> matchL, matchR;
    std::vector<bool> vis;

    // 创建 n 个左点和 m 个右点
    Hungarian(int n_, int m_) : n(n_), m(m_), adj(n_), matchL(n_, -1), matchR(m_, -1) {}

    // 加入左点到右点的边
    void addEdge(int u, int v) {
        adj[u].push_back(v);
    }

    // 寻找一条增广路；仅供 work 调用
    bool dfs(int u) {
        struct Frame {
            int vertex, incoming;
            size_t next;
        };
        std::vector<Frame> frames{{u, -1, 0}};
        while (!frames.empty()) {
            auto &frame = frames.back();
            if (frame.next == adj[frame.vertex].size()) {
                frames.pop_back();
                continue;
            }
            int next = adj[frame.vertex][frame.next++];
            if (vis[next])
                continue;
            vis[next] = true;
            if (matchR[next] == -1) {
                for (int index = int(frames.size()) - 1; index >= 0; --index) {
                    matchR[next] = frames[index].vertex;
                    matchL[frames[index].vertex] = next;
                    next = frames[index].incoming;
                }
                return true;
            }
            frames.push_back({matchR[next], next, 0});
        }
        return false;
    }

    // 返回最大匹配对数
    int work() {
        std::fill(matchL.begin(), matchL.end(), -1);
        std::fill(matchR.begin(), matchR.end(), -1);
        int ans = 0;
        for (int u = 0; u < n; u++) {
            vis.assign(m, false);
            if (dfs(u)) {
                ans++;
            }
        }
        return ans;
    }
};
```
