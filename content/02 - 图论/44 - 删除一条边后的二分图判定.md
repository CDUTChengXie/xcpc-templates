# 删除一条边后的二分图判定

无向图；每条边添加一次，允许不连通、重边和自环。 点和输入边都从0编号；结果与加边顺序对应。

O(n+m) 时间和空间。

```cpp
// 逐边删除后的二分图判定
class DeleteEdgeBipartite {
    int n;
    std::vector<std::pair<int, int>> edges;
    std::vector<std::vector<std::pair<int, int>>> a;

  public:
    // 创建 n 个点，编号0..n-1
    explicit DeleteEdgeBipartite(int n) : n(n), a(n) {
        if (n < 0)
            throw std::invalid_argument("negative n");
    }
    // 添加无向边，返回输入编号；自环和重边照常保留
    int addEdge(int u, int v) {
        if (u < 0 || u >= n || v < 0 || v >= n)
            throw std::out_of_range("vertex");
        int id = edges.size();
        edges.push_back({u, v});
        a[u].push_back({v, id});
        a[v].push_back({u, id});
        return id;
    }
    // 返回每条边单独删除后的二分图判定，可重复调用
    std::vector<bool> work() const {
        int m = edges.size(), oddCycles = 0;
        std::vector<int> parent(n, -1), parentEdge(n, -1), depth(n), it(n), order, odd(n), even(n);
        std::vector<bool> used(m), tree(m), oddEdge(m);
        for (int root = 0; root < n; ++root)
            if (parent[root] == -1) {
                parent[root] = root;
                order.push_back(root);
                std::vector<int> stack{root};
                while (!stack.empty()) {
                    int u = stack.back();
                    if (it[u] == int(a[u].size())) {
                        stack.pop_back();
                        continue;
                    }
                    auto [v, id] = a[u][it[u]++];
                    if (used[id])
                        continue;
                    used[id] = true;
                    if (parent[v] == -1) {
                        parent[v] = u;
                        parentEdge[v] = id;
                        depth[v] = depth[u] + 1;
                        tree[id] = true;
                        order.push_back(v);
                        stack.push_back(v);
                    } else {
                        bool isOdd = ((depth[u] - depth[v] + 1) & 1);
                        oddEdge[id] = isOdd;
                        if (isOdd) {
                            ++oddCycles;
                            ++odd[u];
                            --odd[v];
                        } else {
                            ++even[u];
                            --even[v];
                        }
                    }
                }
            }
        for (int i = int(order.size()) - 1; i >= 0; --i) {
            int u = order[i];
            if (parent[u] != u) {
                odd[parent[u]] += odd[u];
                even[parent[u]] += even[u];
            }
        }
        std::vector<bool> answer(m, oddCycles == 0);
        if (!oddCycles)
            return answer;
        for (int u = 0; u < n; ++u)
            if (parentEdge[u] != -1)
                answer[parentEdge[u]] = odd[u] == oddCycles && even[u] == 0;
        for (int id = 0; id < m; ++id)
            if (!tree[id])
                answer[id] = oddEdge[id] && oddCycles == 1;
        return answer;
    }
};
```
