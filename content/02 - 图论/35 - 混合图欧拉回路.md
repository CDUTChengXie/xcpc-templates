# 混合图欧拉回路

n>=1；只求欧拉回路，有向边方向固定，无向边任选方向。 点、输入边号从 0 开始；无向边只加一次，重边和自环可用，孤点忽略；solve 可重复调用。 无解为 nullopt，无边时 vertices={0}；有 m 条边时 vertices 长 m+1，首尾相同。

edgeIds[i] 是 vertices[i]->vertices[i+1] 所走的输入边号；directions[id] 是该边最终起终点。

Dinic O(n²m)，回路 O(n+m)，空间 O(n+m)。

```cpp
namespace xcpc_mixed_euler {
    template <class T>
    // Dinic 最大流
    struct MaxFlow {
        // 内部残量边
        struct _Edge {
            int to;
            T cap;
            _Edge(int to_, T cap_) : to(to_), cap(cap_) {}
        };

        int n;
        std::vector<_Edge> e;
        std::vector<std::vector<int>> g;
        std::vector<int> cur, h;

        // 创建未初始化对象；使用前调用 init
        MaxFlow() {}
        // 创建 n 个点的空网络
        MaxFlow(int n) {
            init(n);
        }

        // 重置为 n 个点的空网络
        void init(int n) {
            this->n = n;
            e.clear();
            g.assign(n, {});
            cur.resize(n);
            h.resize(n);
        }

        // 建立分层图；仅供 flow 调用
        bool bfs(int s, int t) {
            h.assign(n, -1);
            std::queue<int> que;
            h[s] = 0;
            que.push(s);
            while (!que.empty()) {
                const int u = que.front();
                que.pop();
                for (int i : g[u]) {
                    auto [v, c] = e[i];
                    if (c > 0 && h[v] == -1) {
                        h[v] = h[u] + 1;
                        if (v == t) {
                            return true;
                        }
                        que.push(v);
                    }
                }
            }
            return false;
        }

        // 沿分层图增广；仅供 flow 调用
        T dfs(int u, int t, T f) {
            if (u == t) {
                return f;
            }
            auto r = f;
            for (int &i = cur[u]; i < int(g[u].size()); ++i) {
                const int j = g[u][i];
                auto [v, c] = e[j];
                if (c > 0 && h[v] == h[u] + 1) {
                    auto a = dfs(v, t, std::min(r, c));
                    e[j].cap -= a;
                    e[j ^ 1].cap += a;
                    r -= a;
                    if (r == 0) {
                        return f;
                    }
                }
            }
            return f - r;
        }

        // 加入有向容量边
        void addEdge(int u, int v, T c) {
            g[u].push_back(e.size());
            e.emplace_back(v, c);
            g[v].push_back(e.size());
            e.emplace_back(u, 0);
        }

        // 继续增广并返回新增流量
        T flow(int s, int t) {
            T ans = 0;
            while (bfs(s, t)) {
                cur.assign(n, 0);
                ans += dfs(s, t, std::numeric_limits<T>::max());
            }
            return ans;
        }

        // 返回最后一次分层搜索的 s 侧点集
        std::vector<bool> minCut() {
            std::vector<bool> c(n);
            for (int i = 0; i < n; i++) {
                c[i] = (h[i] != -1);
            }
            return c;
        }

        // 一条原始有向边及其流量
        struct Edge {
            int from;
            int to;
            T cap;
            T flow;
        };

        // 按加边顺序返回原始边及流量
        std::vector<Edge> edges() {
            std::vector<Edge> a;
            for (int i = 0; i < e.size(); i += 2) {
                Edge x;
                x.from = e[i + 1].to;
                x.to = e[i].to;
                x.cap = e[i].cap + e[i + 1].cap;
                x.flow = e[i + 1].cap;
                a.push_back(x);
            }
            return a;
        }
    };

}
// 混合图欧拉回路
class MixedEuler {
    struct Edge {
        int u, v;
        bool directed;
    };
    int n;
    std::vector<Edge> e;

  public:
    // 求解结果：顶点回路、边编号和每边方向
    struct Result {
        // vertices 按回路顺序列出顶点，首尾相同；edgeIds[i] 是从 vertices[i] 走到下一个顶点的输入边号
        std::vector<int> vertices, edgeIds;
        std::vector<std::pair<int, int>> directions; // directions[id] 为输入边 id 最终的 {起点, 终点}
    };
    // 创建 n 个点，编号 0..n-1
    explicit MixedEuler(int n) : n(n) {
        if (n < 1)
            throw std::invalid_argument("n must be positive");
    }
    // 加一条边；directed=true 固定 u->v，false 可定向，返回输入边编号
    int addEdge(int u, int v, bool directed) {
        if (u < 0 || u >= n || v < 0 || v >= n)
            throw std::out_of_range("vertex");
        e.push_back({u, v, directed});
        return int(e.size()) - 1;
    }
    // 返回定向方案和欧拉回路；不存在时返回空 optional，可重复调用
    std::optional<Result> solve() const {
        xcpc_mixed_euler::MaxFlow<int> flow(n + 2);
        int s = n, t = n + 1, need = 0;
        std::vector<int> deg(n), flowId(e.size(), -1);
        Result r;
        int nextId = 0;
        for (int id = 0; id < int(e.size()); ++id) {
            auto x = e[id];
            ++deg[x.u];
            --deg[x.v];
            r.directions.push_back({x.u, x.v});
            if (!x.directed && x.u != x.v) {
                flowId[id] = nextId++;
                flow.addEdge(x.u, x.v, 1);
            }
        }
        for (int u = 0; u < n; ++u) {
            if (deg[u] % 2)
                return std::nullopt;
            if (deg[u] > 0) {
                flow.addEdge(s, u, deg[u] / 2);
                need += deg[u] / 2;
            } else if (deg[u] < 0)
                flow.addEdge(u, t, -deg[u] / 2);
        }
        if (flow.flow(s, t) != need)
            return std::nullopt;
        auto residual = flow.edges();
        std::vector<std::vector<std::pair<int, int>>> adj(n);
        for (int id = 0; id < int(e.size()); ++id) {
            auto [u, v] = r.directions[id];
            if (flowId[id] != -1 && residual[flowId[id]].flow)
                std::swap(u, v);
            r.directions[id] = {u, v};
            adj[u].push_back({v, id});
        }
        int start = 0;
        for (int u = 0; u < n; ++u)
            if (!adj[u].empty()) {
                start = u;
                break;
            }
        std::vector<int> it(n);
        std::vector<std::pair<int, int>> stack{{start, -1}};
        while (!stack.empty()) {
            int u = stack.back().first;
            if (it[u] < int(adj[u].size())) {
                auto [v, id] = adj[u][it[u]++];
                stack.push_back({v, id});
            } else {
                auto [v, id] = stack.back();
                stack.pop_back();
                r.vertices.push_back(v);
                if (id != -1)
                    r.edgeIds.push_back(id);
            }
        }
        if (r.edgeIds.size() != e.size())
            return std::nullopt;
        std::reverse(r.vertices.begin(), r.vertices.end());
        std::reverse(r.edgeIds.begin(), r.edgeIds.end());
        return r;
    }
};
```
