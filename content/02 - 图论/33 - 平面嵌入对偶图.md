# 平面嵌入对偶图

n>=1 且图连通，坐标 |x|,|y|<=1e9；顶点互异；调用者保证线段只在公共端点相交。 无自环、无重合射线；对偶方向权 uv/vu 非负；非法嵌入抛异常，函数不检查任意交叉。 顶点、输入边、面、半边均从 0 编号；面 0 为外面；边 e 的 2e 半边沿 u->v，2e+1 反向。

halfEdges[h].face 为半边左侧面；faces[f].boundary 为边界半边号，area2 为 __int128 有向两倍面积。 拆外面时 0<=first<second<外面边界长度，[first,second) 属于新面 F=faces.size()，其余属于面 0。

build O(n log n+Σ(deg(v)log deg(v))+m) 时间/O(n+m) 空间；dualGraph O(m)。

```cpp
namespace PlanarDual {
    // 整数坐标点
    struct Point {
        i64 x, y;
    };
    // 带两个对偶方向权值的输入边
    struct Edge {
        int u, v;
        i64 uv, vu;
    };
    // 一条有向半边
    struct HalfEdge {
        int from, to, edge, next = -1, face = -1;
    };
    // 一个面的边界与有向两倍面积
    struct Face {
        std::vector<int> boundary;
        __int128 area2 = 0;
    };
    // 平面嵌入构建结果
    struct Result {
        int outerFace = 0;
        std::vector<Edge> edges;          // edges[id] 为第 id 条输入边
        std::vector<HalfEdge> halfEdges;  // 输入边 id 对应半边 2*id 和 2*id+1，分别正向和反向；face 在左侧
        std::vector<Face> faces;          // faces[f].boundary 存面 f 的边界半边号；面 0 是外面
        // 返回有向对偶邻接表；边权 uv 为左面到右面，vu 为反向，可按外面边界位置切分
        std::vector<std::vector<std::pair<int, i64>>> dualGraph(int first = -1,
                                                                int second = -1) const {
            if (faces.empty())
                throw std::invalid_argument("Result must come from build");
            bool split = first != -1 || second != -1;
            int F = int(faces.size());
            if (split && (first < 0 || second <= first || second >= int(faces[0].boundary.size())))
                throw std::invalid_argument("invalid outer boundary cut positions");
            std::vector<int> face(halfEdges.size());
            for (int h = 0; h < int(halfEdges.size()); ++h)
                face[h] = halfEdges[h].face;
            if (split)
                for (int i = first; i < second; ++i)
                    face[faces[0].boundary[i]] = F;
            std::vector<std::vector<std::pair<int, i64>>> g(F + int(split));
            for (int e = 0; e < int(edges.size()); ++e) {
                int a = face[2 * e], b = face[2 * e + 1];
                g[a].push_back({b, edges[e].uv});
                g[b].push_back({a, edges[e].vu});
            }
            return g;
        }
    };
    namespace detail {
        using Wide = __int128;
        Point difference(Point a, Point b) {
            return {a.x - b.x, a.y - b.y};
        }
        Wide cross(Point a, Point b) {
            return Wide(a.x) * b.y - Wide(a.y) * b.x;
        }
        int half(Point p) {
            return p.y < 0 || (p.y == 0 && p.x < 0);
        }
    }
    // 返回半边左面、外面 0 和面边界游走
    Result build(const std::vector<Point> &points, const std::vector<Edge> &edges) {
        using namespace detail;
        if (points.empty() || points.size() > size_t(INT_MAX) || edges.size() > size_t(INT_MAX / 2))
            throw std::invalid_argument("invalid graph size");
        int n = int(points.size()), m = int(edges.size());
        std::vector<std::pair<i64, i64>> locations;
        locations.reserve(n);
        for (Point p : points) {
            if (p.x < -1000000000LL || p.x > 1000000000LL || p.y < -1000000000LL ||
                p.y > 1000000000LL)
                throw std::invalid_argument("coordinate exceeds 1e9");
            locations.push_back({p.x, p.y});
        }
        std::sort(locations.begin(), locations.end());
        if (std::adjacent_find(locations.begin(), locations.end()) != locations.end())
            throw std::invalid_argument("duplicate vertex positions");
        Result r;
        r.edges = edges;
        r.halfEdges.reserve(2 * m);
        std::vector<std::vector<int>> out(n);
        for (int e = 0; e < m; ++e) {
            auto [u, v, uv, vu] = edges[e];
            if (u < 0 || u >= n || v < 0 || v >= n || u == v || uv < 0 || vu < 0)
                throw std::invalid_argument("invalid edge or weight");
            r.halfEdges.push_back({u, v, e});
            r.halfEdges.push_back({v, u, e});
            out[u].push_back(2 * e);
            out[v].push_back(2 * e + 1);
        }
        std::vector<char> seen(n);
        std::vector<int> queue{0};
        seen[0] = true;
        for (size_t k = 0; k < queue.size(); ++k)
            for (int h : out[queue[k]]) {
                int v = r.halfEdges[h].to;
                if (!seen[v]) {
                    seen[v] = true;
                    queue.push_back(v);
                }
            }
        if (int(queue.size()) != n)
            throw std::invalid_argument("embedding must be connected");
        std::vector<int> position(2 * m);
        for (int u = 0; u < n; ++u) {
            auto direction = [&](int h) {
                return difference(points[r.halfEdges[h].to], points[u]);
            };
            std::sort(out[u].begin(), out[u].end(), [&](int a, int b) {
                Point x = direction(a), y = direction(b);
                int hx = half(x), hy = half(y);
                if (hx != hy)
                    return hx < hy;
                Wide c = cross(x, y);
                return c != 0 ? c > 0 : a < b;
            });
            for (int i = 0; i < int(out[u].size()); ++i) {
                position[out[u][i]] = i;
                if (i) {
                    Point a = direction(out[u][i - 1]), b = direction(out[u][i]);
                    if (half(a) == half(b) && cross(a, b) == 0)
                        throw std::invalid_argument("parallel overlapping rays or repeated edge");
                }
            }
        }
        for (int h = 0; h < 2 * m; ++h) {
            int v = r.halfEdges[h].to, degree = int(out[v].size());
            r.halfEdges[h].next = out[v][(position[h ^ 1] + degree - 1) % degree];
        }
        if (m == 0) {
            r.faces.push_back({});
            return r;
        }
        for (int start = 0; start < 2 * m; ++start)
            if (r.halfEdges[start].face == -1) {
                int id = int(r.faces.size());
                Face face;
                int h = start;
                do {
                    if (r.halfEdges[h].face != -1)
                        throw std::invalid_argument("invalid face permutation");
                    auto &e = r.halfEdges[h];
                    e.face = id;
                    face.boundary.push_back(h);
                    face.area2 += cross(points[e.from], points[e.to]);
                    h = e.next;
                } while (h != start);
                r.faces.push_back(std::move(face));
            }
        if (int(r.faces.size()) != m - n + 2)
            throw std::invalid_argument(
                "Euler check failed: input not a connected planar embedding");
        int outer = -1;
        for (int f = 0; f < int(r.faces.size()); ++f) {
            if (r.faces[f].area2 < 0) {
                if (outer != -1)
                    throw std::invalid_argument("multiple outer walks");
                outer = f;
            } else if (r.faces[f].area2 == 0 && r.faces.size() > 1)
                throw std::invalid_argument("zero-area bounded face");
        }
        if (outer == -1) {
            if (r.faces.size() != 1 || r.faces[0].area2 != 0)
                throw std::invalid_argument("outer face not found");
            outer = 0;
        }
        if (outer != 0) {
            std::swap(r.faces[0], r.faces[outer]);
            for (auto &h : r.halfEdges) {
                if (h.face == 0)
                    h.face = outer;
                else if (h.face == outer)
                    h.face = 0;
            }
        }
        return r;
    }
}
```
