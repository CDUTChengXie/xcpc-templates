# 最小生成树计数

n>=1；模数 Mod 是 int 范围内的素数，编译期检查；边权为任意 i64。 节点下标 [0,n)；每条无向边只输入一次，重边按不同边计数，自环忽略。 不连通返回 0；单点空树返回 1；返回 0 也可能是树数恰为模数的倍数。

O(m log m + m alpha(n) + Σ(s³+s log Mod)) 时间，O(n+m+max s²) 空间。

```cpp
#ifndef XCPC_MST_COUNT
#define XCPC_MST_COUNT
namespace mst_count {
    // 一条带权无向输入边
    struct Edge {
        int u, v;
        i64 w;
    };

    namespace detail {
        // 内部并查集，维护连通块并支持按大小合并
        struct DSU {
            std::vector<int> parent, size;
            int components;
            // 创建 n 个独立集合，编号 [0,n)
            explicit DSU(int n) : parent(n), size(n, 1), components(n) {
                std::iota(parent.begin(), parent.end(), 0);
            }
            // 返回 x 所在集合的代表编号
            int find(int x) {
                while (parent[x] != x) {
                    parent[x] = parent[parent[x]];
                    x = parent[x];
                }
                return x;
            }
            // 合并两个点的集合；已有连接返回 false，否则返回 true
            bool merge(int u, int v) {
                u = find(u);
                v = find(v);
                if (u == v)
                    return false;
                if (size[u] < size[v])
                    std::swap(u, v);
                parent[v] = u;
                size[u] += size[v];
                --components;
                return true;
            }
        };
        // 返回非负整数幂模 Mod；用于非零高斯主元求逆
        template <int Mod> int power(int a, int exponent) {
            i64 answer = 1;
            for (; exponent; exponent >>= 1, a = i64(a) * a % Mod)
                if (exponent & 1)
                    answer = answer * a % Mod;
            return int(answer);
        }
        // 返回方阵的行列式模 Mod，按值接收；零主元换行，无非零主元返回 0
        template <int Mod> int determinant(std::vector<std::vector<int>> a) {
            int n = a.size(), answer = 1;
            for (int i = 0; i < n; ++i) {
                int pivot = i;
                while (pivot < n && a[pivot][i] == 0)
                    ++pivot;
                if (pivot == n)
                    return 0;
                if (pivot != i) {
                    std::swap(a[pivot], a[i]);
                    answer = answer ? Mod - answer : 0;
                }
                answer = i64(answer) * a[i][i] % Mod;
                int inverse = power<Mod>(a[i][i], Mod - 2);
                for (int j = i + 1; j < n; ++j) {
                    int factor = i64(a[j][i]) * inverse % Mod;
                    for (int k = i; k < n; ++k) {
                        a[j][k] = int((a[j][k] - i64(factor) * a[i][k]) % Mod);
                        if (a[j][k] < 0)
                            a[j][k] += Mod;
                    }
                }
            }
            return answer;
        }
    }

    // 返回 MST 数量模素数 Mod；节点非法或 n<1 抛 invalid_argument，原边数组不变
    template <int Mod = 1000000007> int count(int n, const std::vector<Edge> &edges) {
        static_assert(
            [] {
                if (Mod < 2)
                    return false;
                for (int d = 2; i64(d) * d <= Mod; ++d)
                    if (Mod % d == 0)
                        return false;
                return true;
            }(),
            "MST count requires a prime modulus");
        if (n < 1)
            throw std::invalid_argument("vertex count");
        for (auto [u, v, w] : edges)
            if (u < 0 || u >= n || v < 0 || v >= n)
                throw std::invalid_argument("edge endpoint");
        std::vector<size_t> order(edges.size());
        std::iota(order.begin(), order.end(), size_t(0));
        std::sort(order.begin(), order.end(),
                  [&](size_t i, size_t j) { return edges[i].w < edges[j].w; });
        detail::DSU global(n);
        std::vector<int> id(n, -1);
        int answer = 1;
        for (size_t begin = 0; begin < order.size();) {
            size_t end = begin + 1;
            while (end < order.size() && edges[order[end]].w == edges[order[begin]].w)
                ++end;
            std::vector<int> roots;
            std::vector<std::pair<int, int>> candidates;
            auto number = [&](int root) {
                if (id[root] == -1) {
                    id[root] = roots.size();
                    roots.push_back(root);
                }
                return id[root];
            };
            // 同权组全部读取旧连通关系；数完之后才合并，避免同权边互相遮掉
            for (size_t i = begin; i < end; ++i) {
                auto [u, v, w] = edges[order[i]];
                u = global.find(u);
                v = global.find(v);
                if (u != v) {
                    int a = number(u), b = number(v);
                    candidates.emplace_back(a, b);
                }
            }
            int k = roots.size();
            detail::DSU local(k);
            for (auto [u, v] : candidates)
                local.merge(u, v);
            std::vector<std::vector<int>> vertices(k);
            std::vector<std::vector<std::pair<int, int>>> links(k);
            for (int i = 0; i < k; ++i)
                vertices[local.find(i)].push_back(i);
            for (auto edge : candidates)
                links[local.find(edge.first)].push_back(edge);
            std::vector<int> position(k);
            for (int component = 0; component < k; ++component) {
                const auto &nodes = vertices[component];
                if (nodes.empty())
                    continue;
                int s = nodes.size(), dimension = s - 1;
                for (int i = 0; i < s; ++i)
                    position[nodes[i]] = i;
                std::vector lap(dimension, std::vector<int>(dimension));
                auto increment = [&](int &x) { x = x == Mod - 1 ? 0 : x + 1; };
                auto decrement = [&](int &x) { x = x == 0 ? Mod - 1 : x - 1; };
                // 直接建立去掉末行末列的余子式；每条平行边都贡献一次
                for (auto [u, v] : links[component]) {
                    u = position[u];
                    v = position[v];
                    if (u < dimension)
                        increment(lap[u][u]);
                    if (v < dimension)
                        increment(lap[v][v]);
                    if (u < dimension && v < dimension) {
                        decrement(lap[u][v]);
                        decrement(lap[v][u]);
                    }
                }
                answer = i64(answer) * detail::determinant<Mod>(std::move(lap)) % Mod;
            }
            for (size_t i = begin; i < end; ++i) {
                auto [u, v, w] = edges[order[i]];
                global.merge(u, v);
            }
            // 只清空本组访问的根编号，不为每个权值组扫描全部 n 个顶点
            for (int root : roots)
                id[root] = -1;
            begin = end;
        }
        return global.components == 1 ? answer : 0;
    }
}
#endif // XCPC_MST_COUNT
```
