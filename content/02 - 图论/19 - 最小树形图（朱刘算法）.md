# 最小树形图（朱刘算法）

n>=1，root 在 [0,n)；root 为出发根；允许重边和负权，自环忽略。 点下标 0..n-1；有限中间权值须小于 INF，权值运算不溢出；只返回总权，不返回边集。 directedMST 无解为 -1，与合法负权答案可能重合；directedMSTEx 失败时不改 answer。

O(nm) 时间，O(n+m) 空间。

```cpp
namespace dmst_detail {

    // 每个点的最小入边信息
    struct Node {
        i64 in;  // 该点的最小入边权
        int pre; // 最小入边的起点
    };

    // 返回最小树形图边权和；无解返回空 optional
    std::optional<i64> solve(int n, int root, std::vector<std::tuple<int, int, i64>> edges) {
        i64 total = 0;
        while (true) {
            std::vector<i64> in(n, INF);
            std::vector<int> pre(n, -1);
            // root 不需要入边，必须在选边时排除，否则会误把「指向 root 的边」当成 root 的入边
            for (auto &[u, v, w] : edges) {
                if (u != v && v != root && w < in[v]) {
                    in[v] = w;
                    pre[v] = u;
                }
            }
            in[root] = 0;
            for (int i = 0; i < n; i++) {
                if (i != root && in[i] == INF) {
                    return std::nullopt; // 有点没有入边 -> root 到不了它，无解
                }
            }
            total += std::accumulate(in.begin(), in.end(), (i64)0);

            // 找出所有环
            int cnt = 0;
            std::vector<int> id(n, -1), vis(n, -1);
            for (int i = 0; i < n; i++) {
                int v = i;
                while (v != root && vis[v] != i && id[v] == -1) {
                    vis[v] = i;
                    v = pre[v];
                }
                if (v != root && id[v] == -1) {
                    int u = pre[v];
                    while (u != v) {
                        id[u] = cnt;
                        u = pre[u];
                    }
                    id[v] = cnt++;
                }
            }
            if (cnt == 0) {
                return total; // 无环，当前选择就是最优
            }
            // 不在环里的点各自成一个新点
            for (int i = 0; i < n; i++) {
                if (id[i] == -1) {
                    id[i] = cnt++;
                }
            }
            // 缩点后的新边权要减去目标点原来的 in
            std::vector<std::tuple<int, int, i64>> ne;
            for (auto &[u, v, w] : edges) {
                if (id[u] != id[v]) {
                    ne.emplace_back(id[u], id[v], w - in[v]);
                }
            }
            root = id[root];
            n = cnt;
            edges.swap(ne);
        }
    }

} // namespace dmst_detail

// 返回最小树形图边权和，无解为 -1
i64 directedMST(int n, int root, const std::vector<std::tuple<int, int, i64>> &edges) {
    return dmst_detail::solve(n, root, edges).value_or(-1);
}

// 返回是否有解，有解时把最小总权写入 ans
bool directedMSTEx(int n, int root, const std::vector<std::tuple<int, int, i64>> &edges, i64 &ans) {
    auto r = dmst_detail::solve(n, root, edges);
    if (!r) {
        return false;
    }
    ans = *r;
    return true;
}
```
