# 拓扑排序

adj 为有向出边表；dagLongestPath 只接受无环图，调用前须先判环。 点下标 0..n-1；topoSort/topoSortLex 有环时返回空，n=0 的合法序列也为空。

topoSort、dagLongestPath O(n+m)；topoSortLex O((n+m)log n)；空间 O(n+m)。

```cpp
// 返回任意拓扑序，有环为空
std::vector<int> topoSort(int n, const std::vector<std::vector<int>> &adj) {
    std::vector<int> deg(n);
    for (int u = 0; u < n; u++) {
        for (int v : adj[u]) {
            deg[v]++;
        }
    }
    std::vector<int> ord;
    std::queue<int> que;
    for (int u = 0; u < n; u++) {
        if (deg[u] == 0) {
            que.push(u);
        }
    }
    while (!que.empty()) {
        int u = que.front();
        que.pop();
        ord.push_back(u);
        for (int v : adj[u]) {
            if (--deg[v] == 0) {
                que.push(v);
            }
        }
    }
    return ord.size() == n ? ord : std::vector<int>();
}

// 返回字典序最小拓扑序，有环为空
std::vector<int> topoSortLex(int n, const std::vector<std::vector<int>> &adj) {
    std::vector<int> deg(n);
    for (int u = 0; u < n; u++) {
        for (int v : adj[u]) {
            deg[v]++;
        }
    }
    std::priority_queue<int, std::vector<int>, std::greater<int>> que;
    for (int u = 0; u < n; u++) {
        if (deg[u] == 0) {
            que.push(u);
        }
    }
    std::vector<int> ord;
    while (!que.empty()) {
        int u = que.top();
        que.pop();
        ord.push_back(u);
        for (int v : adj[u]) {
            if (--deg[v] == 0) {
                que.push(v);
            }
        }
    }
    return ord.size() == n ? ord : std::vector<int>();
}

// 返回任意起点到每个点的最长路边数，要求无环
std::vector<int> dagLongestPath(int n, const std::vector<std::vector<int>> &adj) {
    std::vector<int> deg(n);
    for (int u = 0; u < n; u++) {
        for (int v : adj[u]) {
            deg[v]++;
        }
    }
    std::vector<int> dp(n, 0);
    std::queue<int> que;
    for (int u = 0; u < n; u++) {
        if (deg[u] == 0) {
            que.push(u);
        }
    }
    while (!que.empty()) {
        int u = que.front();
        que.pop();
        for (int v : adj[u]) {
            dp[v] = std::max(dp[v], dp[u] + 1);
            if (--deg[v] == 0) {
                que.push(v);
            }
        }
    }
    return dp;
}
```
