# 一般图最大匹配（带花树）

无向无权一般图，允许奇环和重边；调用者须排除自环。 点下标 0..n-1；match[u] 为匹配点，未匹配为 -1；每条无向边只加入一次。

O(n^3) 时间，O(n+m) 空间。

```cpp
// 一般图最大匹配
struct Graph {
    int n;
    std::vector<std::vector<int>> e;
    // 创建 n 个点的无向图
    Graph(int n) : n(n), e(n) {}
    // 加入一条无向边
    void addEdge(int u, int v) {
        e[u].push_back(v);
        e[v].push_back(u);
    }
    // 返回每个点的匹配点，-1 表示未匹配
    std::vector<int> findMatching() {
        std::vector<int> match(n, -1), vis(n), link(n), f(n), dep(n);
        auto find = [&](int u) {
            while (f[u] != u) {
                u = f[u] = f[f[u]];
            }
            return u;
        };
        auto lca = [&](int u, int v) {
            u = find(u);
            v = find(v);
            while (u != v) {
                if (dep[u] < dep[v]) {
                    std::swap(u, v);
                }
                u = find(link[match[u]]);
            }
            return u;
        };
        std::queue<int> que;
        auto blossom = [&](int u, int v, int p) {
            while (find(u) != p) {
                link[u] = v;
                v = match[u];
                if (vis[v] == 0) {
                    vis[v] = 1;
                    que.push(v);
                }
                f[u] = f[v] = p;
                u = link[v];
            }
        };
        auto augment = [&](int u) {
            while (!que.empty()) {
                que.pop();
            }
            std::iota(f.begin(), f.end(), 0);
            std::fill(vis.begin(), vis.end(), -1);
            que.push(u);
            vis[u] = 1;
            dep[u] = 0;
            while (!que.empty()) {
                int u = que.front();
                que.pop();
                for (int v : e[u]) {
                    if (vis[v] == -1) {
                        vis[v] = 0;
                        link[v] = u;
                        dep[v] = dep[u] + 1;
                        if (match[v] == -1) {
                            for (int x = v, y = u, temp; y != -1;
                                 x = temp, y = x == -1 ? -1 : link[x]) {
                                temp = match[y];
                                match[x] = y;
                                match[y] = x;
                            }
                            return;
                        }
                        vis[match[v]] = 1;
                        dep[match[v]] = dep[u] + 2;
                        que.push(match[v]);
                    } else if (vis[v] == 1 && find(v) != find(u)) {
                        int p = lca(u, v);
                        blossom(u, v, p);
                        blossom(v, u, p);
                    }
                }
            }
        };
        auto greedy = [&]() {
            for (int u = 0; u < n; ++u) {
                if (match[u] != -1) {
                    continue;
                }
                for (int v : e[u]) {
                    if (match[v] == -1) {
                        match[u] = v;
                        match[v] = u;
                        break;
                    }
                }
            }
        };
        greedy();
        for (int u = 0; u < n; ++u) {
            if (match[u] == -1) {
                augment(u);
            }
        }
        return match;
    }
};
```
