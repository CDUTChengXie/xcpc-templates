# 矩阵树定理

模数 P 必须是 int 范围内的素数，默认 1000000007；自环不贡献答案。 点下标 0..n-1；无向边只输入一次，重边作为不同边逐条输入；n<=1 返回 1。 不连通返回 0，但 0 也可能是树数恰为 P 的倍数。

O(n^3) 时间，O(n^2) 空间。

```cpp
#ifndef XCPC_MODINT
#define XCPC_MODINT

// 作用于取模类的快速幂：power(Z(2), 10) 得到 2^10
template <class T> constexpr T power(T a, i64 b) {
    T res = 1;
    for (; b; b /= 2, a *= a) {
        if (b % 2) {
            res *= a;
        }
    }
    return res;
}

// 模数 > 2^30 时使用：乘法用 __int128 精确计算，不依赖浮点精度
constexpr i64 mul(i64 a, i64 b, i64 p) {
    return i64(i128(a) * b % p);
}

template <i64 P>
// 64 位模数下的静态模整数
struct MLong {
    i64 x;

    // 构造值为 0 的取模数。
    constexpr MLong() : x{} {}
    // 用整数构造并规范到 [0,mod)。
    constexpr MLong(i64 x) : x{norm(x % getMod())} {}

    // 动态模数存储。
    static i64 Mod;
    // 返回当前模数。
    constexpr static i64 getMod() {
        return P > 0 ? P : Mod;
    }
    // 设置动态模数，仅 P=0 时生效。
    constexpr static void setMod(i64 Mod_) {
        Mod = Mod_;
    }
    // 将值规范到 [0,mod)。
    constexpr i64 norm(i64 x) const {
        if (x < 0) {
            x += getMod();
        }
        if (x >= getMod()) {
            x -= getMod();
        }
        return x;
    }
    // 返回底层值。
    constexpr i64 val() const {
        return x;
    }
    // 转换为 i64。
    explicit constexpr operator i64() const {
        return x;
    }
    // 返回相反数。
    constexpr MLong operator-() const {
        MLong res;
        res.x = norm(getMod() - x);
        return res;
    }
    // 返回模逆元，要求模数为素数且自身非零。
    constexpr MLong inv() const {
        assert(x != 0);
        return power(*this, getMod() - 2);
    }
    // 乘并赋值。
    constexpr MLong &operator*=(MLong rhs) & {
        x = mul(x, rhs.x, getMod());
        return *this;
    }
    // 加并赋值。
    constexpr MLong &operator+=(MLong rhs) & {
        x = i64((i128(x) + rhs.x) % getMod());
        return *this;
    }
    // 减并赋值。
    constexpr MLong &operator-=(MLong rhs) & {
        x = norm(x - rhs.x);
        return *this;
    }
    // 除并赋值。
    constexpr MLong &operator/=(MLong rhs) & {
        return *this *= rhs.inv();
    }
    // 返回两数乘积。
    friend constexpr MLong operator*(MLong lhs, MLong rhs) {
        MLong res = lhs;
        res *= rhs;
        return res;
    }
    // 返回两数之和。
    friend constexpr MLong operator+(MLong lhs, MLong rhs) {
        MLong res = lhs;
        res += rhs;
        return res;
    }
    // 返回两数之差。
    friend constexpr MLong operator-(MLong lhs, MLong rhs) {
        MLong res = lhs;
        res -= rhs;
        return res;
    }
    // 返回两数之商。
    friend constexpr MLong operator/(MLong lhs, MLong rhs) {
        MLong res = lhs;
        res /= rhs;
        return res;
    }
    // 从流中读入一个取模数。
    friend constexpr std::istream &operator>>(std::istream &is, MLong &a) {
        i64 v;
        if (is >> v)
            a = MLong(v);
        return is;
    }
    // 向流中输出规范值。
    friend constexpr std::ostream &operator<<(std::ostream &os, const MLong &a) {
        return os << a.val();
    }
    // 按规范值判断相等。
    friend constexpr bool operator==(MLong lhs, MLong rhs) {
        return lhs.val() == rhs.val();
    }
    // 按规范值判断不等。
    friend constexpr bool operator!=(MLong lhs, MLong rhs) {
        return lhs.val() != rhs.val();
    }
};

template <> i64 MLong<0LL>::Mod = 1000000007LL;

// 返回方阵行列式，按值接收矩阵
template <class T> T det(std::vector<std::vector<T>> a) {
    int n = a.size();
    T ans = 1;
    for (int i = 0; i < n; i++) {
        int p = i;
        for (int j = i + 1; j < n; j++) {
            if (a[j][i] != 0) {
                p = j;
                break;
            }
        }
        if (a[p][i] == 0) {
            return 0;
        }
        if (p != i) {
            std::swap(a[i], a[p]);
            ans = -ans;
        }
        ans *= a[i][i];
        T inv = a[i][i].inv();
        for (int j = i + 1; j < n; j++) {
            T t = a[j][i] * inv;
            for (int k = i; k < n; k++) {
                a[j][k] -= t * a[i][k];
            }
        }
    }
    return ans;
}

// 返回无向图生成树数量模 P
Z countSpanningTree(int n, const std::vector<std::pair<int, int>> &edges) {
    if (n <= 1) {
        return 1;
    }
    std::vector<std::vector<Z>> lap(n, std::vector<Z>(n));
    for (auto [u, v] : edges) {
        if (u == v) {
            continue;
        }
        lap[u][u] += 1;
        lap[v][v] += 1;
        lap[u][v] -= 1;
        lap[v][u] -= 1;
    }
    std::vector<std::vector<Z>> a(n - 1, std::vector<Z>(n - 1));
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1; j++) {
            a[i][j] = lap[i][j];
        }
    }
    return det(a);
}
```
