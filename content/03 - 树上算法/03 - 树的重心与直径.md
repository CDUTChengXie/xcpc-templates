# 树的重心与直径

静态无向连通树，节节点从 0 编号；构造时传入完整邻接表。 n >= 1；重心最多 2 个；直径返回边数和两个端点。

重心 O(n)；直径 O(n)；空间 O(n)

```cpp
// 求树的重心与直径
struct TreeUtil {
    int n;
    std::vector<std::vector<int>> adj;

    // 用 n 个点的邻接表构造查询器
    TreeUtil(int n_, const std::vector<std::vector<int>> &adj_) : n(n_), adj(adj_) {}

    // 求重心：删除重心后最大连通块尽量小
    std::pair<int, std::vector<int>> centroid() {
        std::vector<int> parent(n, -1), order{0}, sizes(n, 1), centers;
        for (size_t index = 0; index < order.size(); ++index) {
            int vertex = order[index];
            for (int next : adj[vertex])
                if (next != parent[vertex]) {
                    parent[next] = vertex;
                    order.push_back(next);
                }
        }
        int best = n;
        for (auto iterator = order.rbegin(); iterator != order.rend(); ++iterator) {
            int vertex = *iterator, largest = 0;
            for (int next : adj[vertex])
                if (next != parent[vertex]) {
                    sizes[vertex] += sizes[next];
                    largest = std::max(largest, sizes[next]);
                }
            largest = std::max(largest, n - sizes[vertex]);
            if (largest < best) {
                best = largest;
                centers = {vertex};
            } else if (largest == best)
                centers.push_back(vertex);
        }
        return {best, centers};
    }

    // 求直径：返回 {长度, 端点 a, 端点 b}
    std::tuple<int, int, int> diameter() {
        auto bfs = [&](int s) {
            std::vector<int> d(n, -1);
            std::queue<int> que;
            d[s] = 0;
            que.push(s);
            int far = s;
            while (!que.empty()) {
                int u = que.front();
                que.pop();
                if (d[u] > d[far]) {
                    far = u;
                }
                for (int v : adj[u]) {
                    if (d[v] == -1) {
                        d[v] = d[u] + 1;
                        que.push(v);
                    }
                }
            }
            return std::pair{d[far], far};
        };
        auto [_, a] = bfs(0);
        auto [d, b] = bfs(a);
        return {d, a, b};
    }
};
```
