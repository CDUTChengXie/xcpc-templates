# 点分治

先建立双向带权邻接表，构造 `CentroidDecomp(n,graph)` 后调用 `solve(K)`；同一棵树可重复求不同 K。

树连通且边权非负；距离与计数过程不能溢出 i64。 从 0 编号；solve(K) 统计 u != v 且 dist(u,v) <= K 的无序对数；n<=1 返回 0。

期望 / 最坏 O(n log^2 n)，空间 O(n)。

```cpp
// 统计树上距离不超过 K 的无序点对
struct CentroidDecomp {
    int n;
    std::vector<std::vector<std::pair<int, i64>>> g;
    std::vector<int> sz;
    std::vector<bool> del;
    i64 K = 0;
    i64 ans = 0;

    // 用 n 个点的带权邻接表构造
    CentroidDecomp(int n_, const std::vector<std::vector<std::pair<int, i64>>> &g_)
        : n(n_), g(g_), sz(n_), del(n_, false) {}

    // 返回当前未删连通块中以 u 为根的子树大小
    int getSize(int u, int fa) {
        std::vector<std::pair<int, int>> order{{u, fa}};
        for (size_t index = 0; index < order.size(); ++index) {
            auto [vertex, parent] = order[index];
            sz[vertex] = 1;
            for (auto [next, weight] : g[vertex]) {
                if (next != parent && !del[next])
                    order.push_back({next, vertex});
            }
        }
        for (auto iterator = order.rbegin(); iterator != order.rend(); ++iterator) {
            auto [vertex, parent] = *iterator;
            if (vertex != u)
                sz[parent] += sz[vertex];
        }
        return sz[u];
    }

    // 返回当前未删连通块的重心
    int findCentroid(int u, int fa, int total) {
        while (true) {
            int next = -1;
            for (auto [vertex, weight] : g[u]) {
                if (vertex != fa && !del[vertex] && sz[vertex] > total / 2) {
                    next = vertex;
                    break;
                }
            }
            if (next == -1)
                return u;
            fa = u;
            u = next;
        }
    }

    // 收集从 u 出发且未超过 K 的距离
    void dfsDist(int u, int fa, i64 d, std::vector<i64> &ds) {
        std::vector<std::tuple<int, int, i64>> pending{{u, fa, d}};
        while (!pending.empty()) {
            auto [vertex, parent, distance] = pending.back();
            pending.pop_back();
            if (distance > K)
                continue;
            ds.push_back(distance);
            for (auto [next, weight] : g[vertex]) {
                if (next != parent && !del[next] && weight <= K - distance) {
                    pending.push_back({next, vertex, distance + weight});
                }
            }
        }
    }

    // 返回有序 ds 中 i<j 且 ds[i]+ds[j]<=K 的下标对数
    i64 countPairs(const std::vector<i64> &ds) {
        i64 res = 0;
        int j = int(ds.size()) - 1;
        for (int i = 0; i < int(ds.size()); i++) {
            while (j > i && __int128(ds[i]) + ds[j] > K) {
                --j;
            }
            if (j > i) {
                res += j - i;
            }
        }
        return res;
    }

    // 递归拆分当前未删连通块
    void decompose(int u) {
        int total = getSize(u, -1);
        int c = findCentroid(u, -1, total);
        del[c] = true;

        std::vector<i64> all;
        all.push_back(0);
        for (auto [v, w] : g[c]) {
            if (!del[v]) {
                std::vector<i64> ds;
                dfsDist(v, c, w, ds);
                std::sort(ds.begin(), ds.end());
                ans -= countPairs(ds);
                for (i64 d : ds) {
                    all.push_back(d);
                }
            }
        }
        std::sort(all.begin(), all.end());
        ans += countPairs(all);

        for (auto [v, w] : g[c]) {
            if (!del[v]) {
                decompose(v);
            }
        }
    }

    // 返回距离不超过 K_ 的不同点无序对数；可重复调用
    i64 solve(i64 K_) {
        K = K_;
        ans = 0;
        std::fill(del.begin(), del.end(), false);
        if (n > 0) {
            decompose(0);
        }
        return ans;
    }
};
```
