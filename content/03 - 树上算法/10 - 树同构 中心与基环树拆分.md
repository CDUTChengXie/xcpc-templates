# 无根树同构与树中心

输入必须是双向无向树邻接表，点从 0 编号；中心最多两个。空树中心为空，两棵空树同构；有根类型编号仅在同一 Encoder 内可比较。

同构 O(n log n)，中心 O(n)，空间 O(n)。

```cpp
namespace TreeShape {
// 为有根树生成可比较的类型编号
    struct Encoder {
        std::map<std::vector<int>, int> ids;
        // 返回以 root 为根的树形编号；只有同一 Encoder 内的编号可互相比较
        int rooted(const std::vector<std::vector<int>> &adj, int root) {
            int n = int(adj.size());
            assert(root >= 0 && root < n);
            std::vector<int> parent(n, -1), order{root}, value(n);
            parent[root] = root;
            size_t edges = 0;
            for (size_t i = 0; i < order.size(); ++i) {
                int u = order[i];
                edges += adj[u].size();
                for (int v : adj[u]) {
                    assert(v >= 0 && v < n);
                    if (v == parent[u])
                        continue;
                    assert(parent[v] == -1);
                    parent[v] = u;
                    order.push_back(v);
                }
            }
            assert(order.size() == size_t(n) && edges == size_t(2 * (n - 1)));
            for (int i = n - 1; i >= 0; --i) {
                int u = order[i];
                std::vector<int> children;
                for (int v : adj[u])
                    if (v != parent[u])
                        children.push_back(value[v]);
                std::sort(children.begin(), children.end());
                auto [it, inserted] = ids.try_emplace(children, int(ids.size()) + 1);
                value[u] = it->second;
            }
            return value[root];
        }
    };
    // 返回树的中心节点（直径中点），最多两个
    inline std::vector<int> centers(const std::vector<std::vector<int>> &adj) {
        int n = int(adj.size());
        if (!n)
            return {};
        std::vector<int> degree(n);
        std::queue<int> q;
        for (int u = 0; u < n; ++u) {
            degree[u] = int(adj[u].size());
            if (degree[u] <= 1)
                q.push(u);
        }
        int remaining = n;
        while (remaining > 2) {
            int count = int(q.size());
            assert(count > 0);
            remaining -= count;
            while (count--) {
                int u = q.front();
                q.pop();
                for (int v : adj[u])
                    if (--degree[v] == 1)
                        q.push(v);
            }
        }
        std::vector<int> answer;
        while (!q.empty()) {
            answer.push_back(q.front());
            q.pop();
        }
        return answer;
    }
    // 比较两棵无标号无根树是否同构
    inline bool isomorphic(const std::vector<std::vector<int>> &a,
                           const std::vector<std::vector<int>> &b) {
        if (a.size() != b.size())
            return false;
        if (a.empty())
            return true;
        auto ca = centers(a), cb = centers(b);
        Encoder enc;
        int x = enc.rooted(a, ca[0]);
        for (int r : cb)
            if (enc.rooted(b, r) == x)
                return true;
        return false;
    }
}
```

## 基环图拆分

每个连通块必须恰有一个环；点 0..n-1，每条无向边 {u,v} 只传一次，允许自环和重边。空图返回空字段。

O(n+m) 时间与空间。

```cpp
namespace TreeShape {
// 基环树拆分结果
    struct UnicyclicResult {
        std::vector<std::vector<int>> cycles; // 每个连通分量的环，按环上顺序列顶点，不重复首点
        // cycleRoot[u] 是 u 挂到的环上顶点，不是环编号；环上点就是自身
        // parent[u] 是 u 朝环走的下一点；u 在环上时为 -1
        std::vector<int> cycleRoot, parent;
    };
    // 拆分每个连通分量恰好一个环的基环树
    inline UnicyclicResult decompose(int n, const std::vector<std::pair<int, int>> &edges) {
        assert(n >= 0);
        std::vector<std::vector<std::pair<int, int>>> adj(n);
        std::vector<int> degree(n);
        for (int i = 0; i < int(edges.size()); ++i) {
            auto [u, v] = edges[i];
            assert(u >= 0 && u < n && v >= 0 && v < n);
            adj[u].push_back({v, i});
            adj[v].push_back({u, i});
            ++degree[u];
            ++degree[v];
        }
        std::queue<int> q;
        std::vector<bool> onCycle(n, true);
        for (int u = 0; u < n; ++u)
            if (degree[u] <= 1)
                q.push(u);
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            if (!onCycle[u])
                continue;
            onCycle[u] = false;
            for (auto [v, id] : adj[u])
                if (onCycle[v] && --degree[v] == 1)
                    q.push(v);
        }
        UnicyclicResult result;
        result.cycleRoot.assign(n, -1);
        result.parent.assign(n, -1);
        std::vector<bool> seen(n);
        for (int s = 0; s < n; ++s)
            if (onCycle[s] && !seen[s]) {
                std::vector<int> cycle;
                int u = s, previousEdge = -1;
                do {
                    seen[u] = true;
                    cycle.push_back(u);
                    result.cycleRoot[u] = u;
                    int next = -1, idNext = -1;
                    for (auto [v, id] : adj[u])
                        if (onCycle[v] && id != previousEdge) {
                            next = v;
                            idNext = id;
                            break;
                        }
                    assert(next != -1 && (!seen[next] || next == s));
                    u = next;
                    previousEdge = idNext;
                } while (u != s);
                result.cycles.push_back(cycle);
            }
        for (int u = 0; u < n; ++u)
            if (onCycle[u])
                q.push(u);
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (auto [v, id] : adj[u])
                if (!onCycle[v] && result.cycleRoot[v] == -1) {
                    result.cycleRoot[v] = result.cycleRoot[u];
                    result.parent[v] = u;
                    q.push(v);
                }
        }
        for (int u = 0; u < n; ++u)
            assert(result.cycleRoot[u] != -1);
        assert(edges.size() == size_t(n));
        return result;
    }
}
```
