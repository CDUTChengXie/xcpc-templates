# 树上差分

先给 LCA 加边并 `work(root)`，再构造 `TreeDiff<T>(lca)`；完成全部路径加操作后，只调用一次 `collect()` 再读结果。

n>=1 的连通无向树且 LCA 已 work；点差分与边差分使用两套独立数组。 从 0 编号；edge[u] 表示边 (u,parent[u])，根的 edge 无意义；全部 add 后 collect 一次。 绑定后不能改变 LCA 的树或根；collect 后不能再 add，下一批重新创建 TreeDiff。

每次 addPoint / addEdge O(log n)；collect O(n)；空间 O(n)。

```cpp
#ifndef XCPC_TEMPLATE_LCA
#define XCPC_TEMPLATE_LCA

// 倍增预处理 LCA
struct LCA {
    int n, lg, root, timer;
    std::vector<std::vector<int>> adj;
    std::vector<std::vector<int>> up;
    std::vector<int> dep, dfn;

    // 创建空对象；使用前调用 init(n)
    LCA() {}
    // 创建 n 个点的空树
    explicit LCA(int n) {
        init(n);
    }
    // 重置为 n 个点的空树，需重新加边
    void init(int n) {
        this->n = n;
        lg = std::__lg(n) + 1;
        adj.assign(n, {});
    }
    // 加入无向边 u-v
    void addEdge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    // 从 u 开始遍历，记录父亲、深度和 DFS 序
    void dfs(int u, int p) {
        std::vector<std::pair<int, int>> pending{{u, p}};
        while (!pending.empty()) {
            auto [vertex, parent] = pending.back();
            pending.pop_back();
            up[0][vertex] = parent;
            dfn[vertex] = timer++;
            for (auto iterator = adj[vertex].rbegin(); iterator != adj[vertex].rend(); ++iterator) {
                int next = *iterator;
                if (next == parent)
                    continue;
                dep[next] = dep[vertex] + 1;
                pending.push_back({next, vertex});
            }
        }
    }
    // 以 root 预处理倍增表
    void work(int root_ = 0) {
        root = root_;
        up.assign(lg, std::vector<int>(n, root));
        dep.assign(n, 0);
        dfn.assign(n, 0);
        timer = 0;
        dfs(root, root);
        for (int j = 1; j < lg; j++) {
            for (int i = 0; i < n; i++) {
                up[j][i] = up[j - 1][up[j - 1][i]];
            }
        }
    }
    // 返回 u、v 的最近公共祖先编号
    int queryLca(int u, int v) const {
        if (dep[u] < dep[v]) {
            std::swap(u, v);
        }
        int d = dep[u] - dep[v];
        for (int j = 0; j < lg; j++) {
            if (d >> j & 1) {
                u = up[j][u];
            }
        }
        if (u == v) {
            return u;
        }
        for (int j = lg - 1; j >= 0; j--) {
            if (up[j][u] != up[j][v]) {
                u = up[j][u];
                v = up[j][v];
            }
        }
        return up[0][u];
    }
};

#endif // XCPC_TEMPLATE_LCA

// 树上点差分与边差分
template <class T = int> struct TreeDiff {
    LCA &lca;
    // collect 后，point[u] 为点 u 的累加值，edge[u] 为 u 到父节点那条边的累加值；根的 edge 不读
    std::vector<T> point, edge;

    // 绑定已预处理的 LCA，并初始化两套计数数组
    explicit TreeDiff(LCA &lca_) : lca(lca_), point(lca_.n, T(0)), edge(lca_.n, T(0)) {}

    // 给 u-v 路径上每个点加 val，含两个端点
    void addPoint(int u, int v, T val = 1) {
        int w = lca.queryLca(u, v);
        point[u] += val;
        point[v] += val;
        point[w] -= val;
        if (w != lca.root) {
            point[lca.up[0][w]] -= val;
        }
    }

    // 给 u-v 路径上每条边加 val，结果记在较深端
    void addEdge(int u, int v, T val = 1) {
        int w = lca.queryLca(u, v);
        edge[u] += val;
        edge[v] += val;
        edge[w] -= 2 * val;
    }

    // 自底向上累积；只能在全部 add 后调用一次
    void collect() {
        std::vector<int> order(lca.n);
        for (int vertex = 0; vertex < lca.n; ++vertex)
            order[lca.dfn[vertex]] = vertex;
        for (int index = lca.n - 1; index >= 0; --index) {
            int vertex = order[index];
            int parent = lca.up[0][vertex];
            if (parent == vertex)
                continue;
            point[parent] += point[vertex];
            edge[parent] += edge[vertex];
        }
    }
};
```
