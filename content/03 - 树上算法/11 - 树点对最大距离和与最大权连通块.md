# 树上配对与最优连通子树

n>=1，edges 为 {u,v,边权} 数组，每条无向树边传一次；配对边权非负，连通块允许负权。 节点从 0 编号；marked 为互异的偶数个点，空列表返回 0，省略则全点配对（n 必须偶数）。

point 长度 n；连通块必须非空，全负时也选点；非法树/参数抛 invalid_argument，结果溢出 i64 抛 overflow_error。

建树和每次求解均 O(n) 时间与空间

```cpp
#ifndef XCPC_TREE_PAIR_CONNECTED
#define XCPC_TREE_PAIR_CONNECTED
// 求点对最大距离和与最大权连通块
class TreePairConnected {
    int n;
    std::vector<int> parent, order;
    std::vector<i64> parentWeight;
    bool negativeEdge = false;

  public:
    // 检查 n>=1、端点合法、n-1 条边且连通；预处理父节点和 DFS 顺序，非法树抛 invalid_argument。
    TreePairConnected(int n, const std::vector<std::tuple<int, int, i64>> &edges) : n(n) {
        if (n < 1 || edges.size() != size_t(n - 1))
            throw std::invalid_argument("invalid tree size");
        std::vector<std::vector<std::pair<int, i64>>> adj(n);
        for (auto [u, v, w] : edges) {
            if (u < 0 || u >= n || v < 0 || v >= n || u == v)
                throw std::invalid_argument("invalid edge");
            adj[u].push_back({v, w});
            adj[v].push_back({u, w});
            negativeEdge |= w < 0;
        }
        parent.assign(n, -1);
        parentWeight.resize(n);
        std::vector<int> stack{0};
        parent[0] = 0;
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            order.push_back(u);
            for (auto [v, w] : adj[u])
                if (parent[v] == -1) {
                    parent[v] = u;
                    parentWeight[v] = w;
                    stack.push_back(v);
                }
        }
        // 无向图有 n-1 条边且连通，足以保证是树（也拒绝导致不连通的重边/环）。
        if (order.size() != size_t(n))
            throw std::invalid_argument("disconnected tree");
    }

    // 将全体顶点两两配对，返回最大距离和；奇数点数或任意负边权抛 invalid_argument。
    i64 maxPairDistance() const {
        std::vector<int> marked(n);
        std::iota(marked.begin(), marked.end(), 0);
        return maxPairDistance(marked);
    }

    // 将 marked 中互异的偶数个顶点配对，返回最大距离和；空集合返回0；非法集合/负边权拒绝。
    i64 maxPairDistance(const std::vector<int> &marked) const {
        if (negativeEdge || marked.size() > size_t(n) || marked.size() % 2)
            throw std::invalid_argument("negative edge or odd/invalid marked set");
        std::vector<int> count(n);
        for (int u : marked) {
            if (u < 0 || u >= n || count[u])
                throw std::invalid_argument("invalid or repeated marked vertex");
            count[u] = 1;
        }
        int total = int(marked.size());
        __int128 answer = 0;
        // 每条边最多跨 min(两侧选中点数) 个点对；树上存在同时达到所有这些上界的配对。
        for (int i = n - 1; i > 0; --i) {
            int u = order[i];
            answer += __int128(parentWeight[u]) * std::min(count[u], total - count[u]);
            count[parent[u]] += count[u];
        }
        if (answer > std::numeric_limits<i64>::max())
            throw std::overflow_error("pair distance sum overflow");
        return i64(answer);
    }

    // 返回 {非空连通子树的最大点权边权总和, 顶点列表}；wantVertices=false 时列表为空，零贡献分支舍弃。
    std::pair<i64, std::vector<int>> maxConnected(const std::vector<i64> &point,
                                                  bool wantVertices = false) const {
        if (point.size() != size_t(n))
            throw std::invalid_argument("invalid point weights");
        std::vector<__int128> dp(point.begin(), point.end());
        int bestRoot = 0;
        __int128 best = point[0];
        for (int i = n - 1; i >= 0; --i) {
            int u = order[i];
            if (dp[u] > best) {
                best = dp[u];
                bestRoot = u;
            }
            if (u != 0)
                dp[parent[u]] += std::max(__int128(0), dp[u] + parentWeight[u]);
        }
        if (best > std::numeric_limits<i64>::max())
            throw std::overflow_error("connected weight sum overflow");
        std::vector<int> vertices;
        if (wantVertices) {
            std::vector<bool> chosen(n);
            chosen[bestRoot] = true;
            for (int u : order) {
                if (u != bestRoot)
                    chosen[u] = chosen[parent[u]] && dp[u] + parentWeight[u] > 0;
                if (chosen[u])
                    vertices.push_back(u);
            }
        }
        return {i64(best), std::move(vertices)};
    }
};

#endif
```
