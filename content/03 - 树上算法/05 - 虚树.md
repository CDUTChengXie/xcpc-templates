# 虚树

先完成 LCA 的 `work(root)`，每批关键点调用 `buildVirtualTree(lca,keys)`。

n>=1 的连通无向树且 LCA 已 work；关键点编号在 [0,n)；虚树上的 DP/统计规则由用户提供。 从 0 编号；虚树使用局部编号，通过 orig 映射回原点；空 keys 时 n=0、root=-1。

O(k log k + k log n)，空间 O(k)；多批查询总代价只与 Σk 有关。

```cpp
#ifndef XCPC_TEMPLATE_LCA
#define XCPC_TEMPLATE_LCA

// 倍增预处理 LCA
struct LCA {
    int n, lg, root, timer;
    std::vector<std::vector<int>> adj;
    std::vector<std::vector<int>> up;
    std::vector<int> dep, dfn;

    // 创建空对象；使用前调用 init(n)
    LCA() {}
    // 创建 n 个点的空树
    explicit LCA(int n) {
        init(n);
    }
    // 重置为 n 个点的空树，需重新加边
    void init(int n) {
        this->n = n;
        lg = std::__lg(n) + 1;
        adj.assign(n, {});
    }
    // 加入无向边 u-v
    void addEdge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    // 从 u 开始遍历，记录父亲、深度和 DFS 序
    void dfs(int u, int p) {
        std::vector<std::pair<int, int>> pending{{u, p}};
        while (!pending.empty()) {
            auto [vertex, parent] = pending.back();
            pending.pop_back();
            up[0][vertex] = parent;
            dfn[vertex] = timer++;
            for (auto iterator = adj[vertex].rbegin(); iterator != adj[vertex].rend(); ++iterator) {
                int next = *iterator;
                if (next == parent)
                    continue;
                dep[next] = dep[vertex] + 1;
                pending.push_back({next, vertex});
            }
        }
    }
    // 以 root 预处理倍增表
    void work(int root_ = 0) {
        root = root_;
        up.assign(lg, std::vector<int>(n, root));
        dep.assign(n, 0);
        dfn.assign(n, 0);
        timer = 0;
        dfs(root, root);
        for (int j = 1; j < lg; j++) {
            for (int i = 0; i < n; i++) {
                up[j][i] = up[j - 1][up[j - 1][i]];
            }
        }
    }
    // 返回 u、v 的最近公共祖先编号
    int queryLca(int u, int v) const {
        if (dep[u] < dep[v]) {
            std::swap(u, v);
        }
        int d = dep[u] - dep[v];
        for (int j = 0; j < lg; j++) {
            if (d >> j & 1) {
                u = up[j][u];
            }
        }
        if (u == v) {
            return u;
        }
        for (int j = lg - 1; j >= 0; j--) {
            if (up[j][u] != up[j][v]) {
                u = up[j][u];
                v = up[j][v];
            }
        }
        return up[0][u];
    }
    // 返回 u、v 间的非加权边数
    int dist(int u, int v) const {
        return dep[u] + dep[v] - 2 * dep[queryLca(u, v)];
    }
};

#endif // XCPC_TEMPLATE_LCA

// 虚树查询结构
struct VirtualTree {
    int n = 0;
    std::vector<std::vector<std::pair<int, int>>> adj; // 局部编号；{子编号, 原树距离}
    std::vector<int> orig;                             // 局部编号 -> 原树节点
    int root = -1;

    // 返回局部编号的先序遍历；逆序即为自底向上
    std::vector<int> preorder() const {
        std::vector<int> res;
        if (root < 0) {
            return res;
        }
        std::vector<int> st{root};
        while (!st.empty()) {
            int u = st.back();
            st.pop_back();
            res.push_back(u);
            for (int i = int(adj[u].size()) - 1; i >= 0; i--) {
                st.push_back(adj[u][i].first);
            }
        }
        return res;
    }
};

// 按 LCA 构建局部编号虚树；keys 可重复，空集返回空树
VirtualTree buildVirtualTree(LCA &lca, std::vector<int> keys) {
    VirtualTree vt;
    sort(keys.begin(), keys.end(), [&](int a, int b) { return lca.dfn[a] < lca.dfn[b]; });
    keys.erase(std::unique(keys.begin(), keys.end()), keys.end());
    int k = int(keys.size());
    if (k == 0) {
        return vt;
    }
    if (k == 1) {
        vt.n = 1;
        vt.adj.resize(1);
        vt.orig = {keys[0]};
        vt.root = 0;
        return vt;
    }

    std::vector<int> nodes = keys;
    for (int i = 1; i < k; i++) {
        nodes.push_back(lca.queryLca(keys[i - 1], keys[i]));
    }
    std::sort(nodes.begin(), nodes.end(), [&](int a, int b) { return lca.dfn[a] < lca.dfn[b]; });
    nodes.erase(std::unique(nodes.begin(), nodes.end()), nodes.end());

    vt.n = int(nodes.size());
    vt.adj.assign(vt.n, {});
    vt.orig = nodes;
    vt.root = -1;

    // 相邻 LCA 必在 nodes 内，二分可定位局部编号
    auto idx = [&](int u) {
        int i = int(std::lower_bound(nodes.begin(), nodes.end(), u,
                                     [&](int a, int b) { return lca.dfn[a] < lca.dfn[b]; }) -
                    nodes.begin());
        assert(i < vt.n && nodes[i] == u);
        return i;
    };
    auto link = [&](int parentIdx, int childIdx) {
        int len = lca.dep[nodes[childIdx]] - lca.dep[nodes[parentIdx]];
        vt.adj[parentIdx].push_back({childIdx, len});
    };

    std::vector<int> st;
    for (int i = 0; i < vt.n; i++) {
        if (st.empty()) {
            st.push_back(i);
            continue;
        }
        int l = idx(lca.queryLca(nodes[st.back()], nodes[i]));
        while (st.size() >= 2 && lca.dep[nodes[st.back()]] > lca.dep[nodes[l]]) {
            int v = st.back();
            st.pop_back();
            link(st.back(), v);
        }
        if (st.back() != l) {
            int v = st.back();
            st.pop_back();
            link(l, v);
            st.push_back(l);
        }
        st.push_back(i);
    }
    while (st.size() >= 2) {
        int v = st.back();
        st.pop_back();
        link(st.back(), v);
    }
    vt.root = st.front();
    return vt;
}
```
