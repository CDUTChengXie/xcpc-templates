# 动态上凸壳与凸包包含

|x|,|y|<=1e9，越界抛 invalid_argument；无删除、面积/周长查询；附 O(log h) 支撑点与直线交查询。 Upper 的同 x 只留最高点，共线中间点删去；below 仅在壳 x 范围内判定“在壳上或下方”。

Hull 用上下两条链判闭包含，支持空、点、竖直段和一般线段；__int128 叉积精确，无 eps。

below/contains O(log n)；单次 add 最坏 O(n log n)，总 m 次插入 O(m log m)；support/intersectsLine O(log n)；空间 O(n)。

```cpp
namespace DynamicHull {
    using I = long long;
    namespace detail {
        void valid(I x, I y) {
            if (x < -1000000000LL || x > 1000000000LL || y < -1000000000LL || y > 1000000000LL)
                throw std::invalid_argument("DynamicHull: coordinate exceeds 1e9");
        }
        __int128 turn(std::pair<I, I> a, std::pair<I, I> b, std::pair<I, I> c) {
            return __int128(b.first - a.first) * (c.second - a.second) -
                   __int128(b.second - a.second) * (c.first - a.first);
        }
        using Int = __int128;
        using Point = std::pair<I, I>;
        struct Edge {
            Point a, b;
        };
        struct Direction {
            Int A, B;
        };
        struct SlopeOrder {
            using is_transparent = void;
            bool operator()(const Edge &a, const Edge &b) const {
                Int lhs = Int(a.b.second - a.a.second) * (b.b.first - b.a.first),
                    rhs = Int(b.b.second - b.a.second) * (a.b.first - a.a.first);
                return lhs != rhs ? lhs > rhs : a.a.first < b.a.first;
            }
            bool operator()(const Edge &e, Direction q) const {
                return q.A * (e.b.first - e.a.first) + q.B * (e.b.second - e.a.second) > 0;
            }
            bool operator()(Direction q, const Edge &e) const {
                return q.A * (e.b.first - e.a.first) + q.B * (e.b.second - e.a.second) < 0;
            }
        };

    }
    class Upper {
        std::map<I, I> q;
        std::set<detail::Edge, detail::SlopeOrder> slopes;
        using Iterator = std::map<I, I>::iterator;
        detail::Edge edge(Iterator it) const {
            return {*it, *std::next(it)};
        }
        void eraseVertex(Iterator it) {
            auto right = std::next(it);
            bool hasLeft = it != q.begin();
            auto left = hasLeft ? std::prev(it) : q.end();
            if (hasLeft)
                slopes.erase(edge(left));
            if (right != q.end())
                slopes.erase(edge(it));
            q.erase(it);
            if (hasLeft && right != q.end())
                slopes.insert(edge(left));
        }
        std::optional<detail::Point> supportWide(detail::Int A, detail::Int B) const {
            if (q.empty())
                return std::nullopt;
            detail::Point first = *q.begin(), last = *q.rbegin();
            if (B <= 0) {
                auto score = [&](detail::Point p) { return A * p.first + B * p.second; };
                return score(first) >= score(last) ? first : last;
            }
            auto it = slopes.lower_bound(detail::Direction{A, B});
            return it == slopes.end() ? last : it->a;
        }
        friend class Hull;

      public:
        // 返回是否为空，O(1)。
        bool empty() const {
            return q.empty();
        }
        // 查询 x 在壳横坐标范围内且 (x,y) 位于上壳或下方；空壳返回 false。
        bool below(I x, I y) const {
            detail::valid(x, y);
            auto r = q.lower_bound(x);
            if (r == q.end())
                return false;
            if (r->first == x)
                return y <= r->second;
            if (r == q.begin())
                return false;
            return detail::turn(*std::prev(r), *r, {x, y}) <= 0;
        }
        // 插入点，重复点及已在上壳下方的点忽略；同 x 保留更高点。
        void add(I x, I y) {
            detail::valid(x, y);
            if (below(x, y))
                return;
            auto old = q.find(x);
            if (old != q.end())
                eraseVertex(old);
            auto right = q.lower_bound(x);
            if (right != q.begin() && right != q.end())
                slopes.erase(edge(std::prev(right)));
            auto it = q.emplace(x, y).first;
            if (it != q.begin())
                slopes.insert(edge(std::prev(it)));
            if (std::next(it) != q.end())
                slopes.insert(edge(it));
            while (it != q.begin()) {
                auto b = std::prev(it);
                if (b == q.begin())
                    break;
                auto a = std::prev(b);
                if (detail::turn(*a, *b, *it) < 0)
                    break;
                eraseVertex(b);
            }
            while (std::next(it) != q.end()) {
                auto b = std::next(it), c = std::next(b);
                if (c == q.end())
                    break;
                if (detail::turn(*it, *b, *c) < 0)
                    break;
                eraseVertex(b);
            }
        }
        // 最大化 A*x+B*y 的上壳顶点，O(log h)，同极值任选；B<0 时凸函数极值在链端点，空壳为空。
        std::optional<detail::Point> support(I A, I B) const {
            return supportWide(A, B);
        }
        // 导出按 x 递增的上壳顶点，O(n)，不包含共线中间点。
        std::vector<std::pair<I, I>> vertices() const {
            return {q.begin(), q.end()};
        }
    };
    class Hull {
        Upper up, low;
        std::optional<detail::Point> supportWide(detail::Int A, detail::Int B) const {
            if (B >= 0)
                return up.supportWide(A, B);
            auto p = low.supportWide(A, -B);
            if (p)
                p->second = -p->second;
            return p;
        }

      public:
        // 插入点；lower 用 y 取负的上壳，参数约束同 Upper。
        void add(I x, I y) {
            detail::valid(x, y);
            up.add(x, y);
            low.add(x, -y);
        }
        // 返回点是否在当前完整凸包闭区域，含边界和低维退化。
        bool contains(I x, I y) const {
            detail::valid(x, y);
            return up.below(x, y) && low.below(x, -y);
        }
        // 返回最大化 A*x+B*y 的原坐标凸包顶点，O(log h)；空包返回空，允许全long long系数及零方向。
        std::optional<detail::Point> support(I A, I B) const {
            return supportWide(A, B);
        }
        // 直线 A*x+B*y=C 是否碰到闭凸包，O(log h)；包含切线，系数全零仅 C==0 且包非空为true。
        bool intersectsLine(I A, I B, I C) const {
            auto hi = supportWide(A, B), lo = supportWide(-detail::Int(A), -detail::Int(B));
            if (!hi)
                return false;
            auto value = [&](detail::Point p) {
                return detail::Int(A) * p.first + detail::Int(B) * p.second - C;
            };
            return value(*lo) <= 0 && value(*hi) >= 0;
        }
        // 返回当前是否没有任何点。
        bool empty() const {
            return up.empty();
        }
    };
}
```
