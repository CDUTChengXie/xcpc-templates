# 多边形几何（面积 包含判定 耳切三角剖分）

i64 坐标，叉积及面积累加须不溢出；顶点沿边界，不自交/重复首点；pointInConvexPolygon 须凸且逆时针。 耳切需非退化简单多边形，宜先去连续重复点与共线中间点；扇形剖分只用于凸多边形。 面积为两倍值，顶点下标从 0 开始；ensureCcw 会重排输入，之后下标对应重排后的数组。

耳切停滞可能返回不足 n-2 项，应核对数量；pointInPolygonRay 的边界结果不保证算内。

面积/凸包含/扇形 O(n)；耳切最坏 O(n^3)。

```cpp
#ifndef XCPC_TEMPLATE_POLY
#define XCPC_TEMPLATE_POLY

// 简化的二维点/向量：仅支持加减、原地加减与严格相等。
template <class T> struct Point {
    T x, y;
    // 构造点/向量，省略坐标得到原点。
    Point(const T &x_ = 0, const T &y_ = 0) : x(x_), y(y_) {}
    // 原地加向量。
    Point &operator+=(const Point &p) & {
        x += p.x;
        y += p.y;
        return *this;
    }
    // 原地减向量。
    Point &operator-=(const Point &p) & {
        x -= p.x;
        y -= p.y;
        return *this;
    }
    // 返回两向量之和。
    friend Point operator+(Point a, const Point &b) {
        return a += b;
    }
    // 返回两向量之差。
    friend Point operator-(Point a, const Point &b) {
        return a -= b;
    }
    // 返回坐标严格相等。
    friend bool operator==(const Point &a, const Point &b) {
        return a.x == b.x && a.y == b.y;
    }
};

// 返回两向量二维叉积。
template <class T> T cross(const Point<T> &a, const Point<T> &b) {
    return a.x * b.y - a.y * b.x;
}

// 返回 o->a 与 o->b 的叉积。
template <class T> T cross(const Point<T> &o, const Point<T> &a, const Point<T> &b) {
    return cross(a - o, b - o);
}

#endif // XCPC_TEMPLATE_POLY

// 返回两倍有向面积（逆时针为正，顺时针为负）。
template <class T> T signedArea2(const std::vector<Point<T>> &p) {
    T s = 0;
    int n = (int)p.size();
    for (int i = 0; i < n; i++) {
        s += cross(p[i], p[(i + 1) % n]);
    }
    return s;
}

// 返回两倍无向面积（取绝对值）。
template <class T> T polygonArea2(const std::vector<Point<T>> &p) {
    T s = signedArea2(p);
    return s < 0 ? (T)-s : s;
}

// 原地把顶点顺序统一成逆时针，退化零面积不变。
template <class T> void ensureCcw(std::vector<Point<T>> &p) {
    if (signedArea2(p) < 0) {
        std::reverse(p.begin(), p.end());
    }
}

// 返回点是否在逆时针凸多边形闭区域内（含边界）；凹多边形改用 pointInPolygonRay。
template <class T> bool pointInConvexPolygon(const std::vector<Point<T>> &poly, const Point<T> &q) {
    int n = (int)poly.size();
    if (n < 3) {
        return false;
    }
    for (int i = 0; i < n; i++) {
        if (cross(poly[i], poly[(i + 1) % n], q) < 0) {
            return false; // 在某条边的右侧 -> 在外面
        }
    }
    return true;
}

// 返回点是否在任意简单多边形内（射线法）；边界结果依赖射线方向，不保证算内。
template <class T> bool pointInPolygonRay(const std::vector<Point<T>> &poly, const Point<T> &q) {
    int n = (int)poly.size();
    bool in = false;
    for (int i = 0, j = n - 1; i < n; j = i++) {
        // 只在跨过 q 所在水平线的边上计数；水平边要跳过，否则除零
        if ((poly[i].y > q.y) == (poly[j].y > q.y)) {
            continue;
        }
        T side = cross(poly[j], poly[i], q);
        if ((poly[i].y > poly[j].y && side > 0) || (poly[i].y < poly[j].y && side < 0)) {
            in = !in;
        }
    }
    return in;
}

// 返回简单多边形的耳切三角剖分，输出 n-2 个逆时针下标三元组；n<3 返回空。
template <class T>
std::vector<std::array<int, 3>> earClipTriangulate(const std::vector<Point<T>> &poly) {
    int n = (int)poly.size();
    std::vector<std::array<int, 3>> tris;
    if (n < 3) {
        return tris;
    }
    std::vector<int> idx(n);
    std::iota(idx.begin(), idx.end(), 0);
    if (signedArea2(poly) < 0) {
        std::reverse(idx.begin(), idx.end()); // 统一成逆时针
    }
    // 退化保护：防止病态输入死循环
    int guard = 20 * n * n + 50;
    while (idx.size() > 3 && guard-- > 0) {
        int m = (int)idx.size();
        bool found = false;
        for (int k = 0; k < m; k++) {
            int i0 = idx[(k - 1 + m) % m], i1 = idx[k], i2 = idx[(k + 1) % m];
            if (cross(poly[i0], poly[i1], poly[i2]) <= 0) {
                continue; // 逆时针下必须左转（凸角）才是耳
            }
            // 三角形内部（含边界）不能有其他顶点
            bool ok = true;
            for (int j : idx) {
                if (j == i0 || j == i1 || j == i2) {
                    continue;
                }
                const Point<T> &q = poly[j];
                T d1 = cross(poly[i0], poly[i1], q);
                T d2 = cross(poly[i1], poly[i2], q);
                T d3 = cross(poly[i2], poly[i0], q);
                bool neg = (d1 < 0) || (d2 < 0) || (d3 < 0);
                bool pos = (d1 > 0) || (d2 > 0) || (d3 > 0);
                if (!(neg && pos)) {
                    ok = false;
                    break;
                }
            }
            if (!ok) {
                continue;
            }
            tris.push_back({i0, i1, i2});
            idx.erase(idx.begin() + k);
            found = true;
            break;
        }
        if (!found) {
            break;
        }
    }
    if (idx.size() == 3) {
        tris.push_back({idx[0], idx[1], idx[2]});
    }
    // 按实际叉积统一定成逆时针
    for (auto &t : tris) {
        if (cross(poly[t[0]], poly[t[1]], poly[t[2]]) < 0) {
            std::swap(t[0], t[2]);
        }
    }
    return tris;
}

// 返回凸多边形的扇形剖分（逆时针下标三元组）；n<3 返回空。
template <class T>
std::vector<std::array<int, 3>> fanTriangulate(const std::vector<Point<T>> &convex) {
    std::vector<std::array<int, 3>> tris;
    int n = (int)convex.size();
    if (n < 3) {
        return tris;
    }
    for (int i = 1; i + 1 < n; i++) {
        tris.push_back({0, i, i + 1});
    }
    for (auto &t : tris) {
        if (cross(convex[t[0]], convex[t[1]], convex[t[2]]) < 0) {
            std::swap(t[0], t[2]);
        }
    }
    return tris;
}
```
