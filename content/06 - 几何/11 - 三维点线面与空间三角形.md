# 三维点线面与空间三角形

坐标有限；直线方向与平面法向非零；空间三角形非共线，非法对象抛 invalid_argument。 Geo3D::Point 的 x/y/z 为 long double 坐标；Plane 的 n 是法向、d 是 n·x=d 的常数；Line 的 o 是线上点、d 是方向，线段另传两端点。

eps 为距离容差；近平行角度采用 eps 相对阈值，近退化输入可能归为平行/共面，非精确判断函数。

所有接口 O(1)，空间 O(1)。

```cpp
namespace Geo3D {
    using Real = long double;
    struct Point {
        Real x = 0, y = 0, z = 0;
        // 向量加减和乘除标量；除数须非零。
        Point operator+(Point b) const {
            return {x + b.x, y + b.y, z + b.z};
        }
        // 返回向量差。
        Point operator-(Point b) const {
            return {x - b.x, y - b.y, z - b.z};
        }
        // 返回标量倍向量。
        Point operator*(Real t) const {
            return {x * t, y * t, z * t};
        }
        // 返回除以非零标量后的向量。
        Point operator/(Real t) const {
            return {x / t, y / t, z / t};
        }
    };
    struct Line {
        Point o, d;
    };
    struct Plane {
        Point n;
        Real d;
    };
    // 返回点积。
    Real dot(Point a, Point b) {
        return a.x * b.x + a.y * b.y + a.z * b.z;
    }
    // 返回右手系叉积。
    Point cross(Point a, Point b) {
        return {a.y * b.z - a.z * b.y, a.z * b.x - a.x * b.z, a.x * b.y - a.y * b.x};
    }
    // 返回向量长度。
    Real norm(Point a) {
        return std::sqrt(dot(a, a));
    }
    namespace detail {
        void line(Line l) {
            if (dot(l.d, l.d) == 0)
                throw std::invalid_argument("zero line direction");
        }
        void plane(Plane p) {
            if (dot(p.n, p.n) == 0)
                throw std::invalid_argument("zero plane normal");
        }
    }
    // 由不共线三点建 Plane；法向按 a,b,c 的右手顺序。
    Plane plane(Point a, Point b, Point c) {
        Point n = cross(b - a, c - a);
        Plane p{n, dot(n, a)};
        detail::plane(p);
        return p;
    }
    // 点投影到直线；方向为零抛异常。
    Point project(Point q, Line l) {
        detail::line(l);
        return l.o + l.d * (dot(q - l.o, l.d) / dot(l.d, l.d));
    }
    // 点投影到平面；法向为零抛异常。
    Point project(Point q, Plane p) {
        detail::plane(p);
        return q - p.n * ((dot(p.n, q) - p.d) / dot(p.n, p.n));
    }
    // 返回点到直线的无符号距离。
    Real distance(Point q, Line l) {
        return norm(q - project(q, l));
    }
    // 返回点到平面的无符号距离。
    Real distance(Point q, Plane p) {
        detail::plane(p);
        return std::abs(dot(p.n, q) - p.d) / norm(p.n);
    }
    // 返回唯一线面交点；近平行（含整线在面内）返回空。
    std::optional<Point> intersect(Line l, Plane p, Real eps = 1e-12L) {
        detail::line(l);
        detail::plane(p);
        Real den = dot(p.n, l.d);
        if (std::abs(den) <= eps * norm(p.n) * norm(l.d))
            return std::nullopt;
        return l.o + l.d * ((p.d - dot(p.n, l.o)) / den);
    }
    // 返回唯一面面交线；近平行或重合返回空，方向未归一化。
    std::optional<Line> intersect(Plane a, Plane b, Real eps = 1e-12L) {
        detail::plane(a);
        detail::plane(b);
        Point d = cross(a.n, b.n);
        if (norm(d) <= eps * norm(a.n) * norm(b.n))
            return std::nullopt;
        return Line{cross(b.n * a.d - a.n * b.d, d) / dot(d, d), d};
    }
    // 返回两条无限直线最近点对；平行时取第二条原点及其投影。
    std::pair<Point, Point> closest(Line a, Line b, Real eps = 1e-12L) {
        detail::line(a);
        detail::line(b);
        Point w = a.o - b.o, n = cross(a.d, b.d);
        Real den = dot(n, n), A = dot(a.d, a.d), B = dot(a.d, b.d), C = dot(b.d, b.d);
        if (den <= eps * eps * A * C)
            return {project(b.o, a), b.o};
        Real D = dot(a.d, w), E = dot(b.d, w);
        return {a.o + a.d * ((B * E - C * D) / den), b.o + b.d * ((A * E - B * D) / den)};
    }
    // 返回两闭线段最近点对；零长段按点处理，近平行可调整 eps。
    std::pair<Point, Point> closestSegments(Point a, Point b, Point c, Point d, Real eps = 1e-12L) {
        Point u = b - a, v = d - c, w = a - c;
        Real A = dot(u, u), B = dot(u, v), C = dot(v, v), D = dot(u, w), E = dot(v, w), s = 0,
             t = 0;
        if (A <= eps * eps && C <= eps * eps)
            return {a, c};
        if (A <= eps * eps)
            t = std::clamp(E / C, Real(0), Real(1));
        else if (C <= eps * eps)
            s = std::clamp(-D / A, Real(0), Real(1));
        else {
            Real den = dot(cross(u, v), cross(u, v));
            if (den > eps * eps * A * C)
                s = std::clamp((B * E - C * D) / den, Real(0), Real(1));
            t = (B * s + E) / C;
            if (t < 0) {
                t = 0;
                s = std::clamp(-D / A, Real(0), Real(1));
            } else if (t > 1) {
                t = 1;
                s = std::clamp((B - D) / A, Real(0), Real(1));
            }
        }
        return {a + u * s, c + v * t};
    }
    // 两闭线段距离<=eps 返回 true；允许点段，不返回重合交集。
    bool segmentsIntersect(Point a, Point b, Point c, Point d, Real eps = 1e-12L) {
        auto [p, q] = closestSegments(a, b, c, d, eps);
        return norm(p - q) <= eps;
    }
    // 点是否在非退化空间三角形闭区域，含边界；不在三角形平面上返回 false。
    bool onTriangle(Point p, Point a, Point b, Point c, Real eps = 1e-12L) {
        Plane f = plane(a, b, c);
        Real len = norm(f.n);
        if (distance(p, f) > eps)
            return false;
        for (auto [u, v] : std::array<std::pair<Point, Point>, 3>{{{a, b}, {b, c}, {c, a}}})
            if (dot(f.n, cross(v - u, p - u)) < -eps * len * norm(v - u))
                return false;
        return true;
    }
    // 线段与非退化三角形求交；共面时返回重叠段端点，空/点/段分别 0/1/2 个点。
    std::vector<Point> segmentTriangle(Point a, Point b, Point u, Point v, Point w,
                                       Real eps = 1e-12L) {
        Plane p = plane(u, v, w);
        Real len = norm(p.n), sa = (dot(p.n, a) - p.d) / len, sb = (dot(p.n, b) - p.d) / len;
        if ((sa > eps && sb > eps) || (sa < -eps && sb < -eps))
            return {};
        if (std::abs(sa) <= eps && std::abs(sb) <= eps) {
            Real lo = 0, hi = 1;
            for (auto [x, y] : std::array<std::pair<Point, Point>, 3>{{{u, v}, {v, w}, {w, u}}}) {
                Real A = dot(p.n, cross(y - x, a - x)), B = dot(p.n, cross(y - x, b - x)),
                     tol = eps * len * norm(y - x);
                if (A < -tol && B < -tol)
                    return {};
                if (A < 0 && B >= 0)
                    lo = std::max(lo, A / (A - B));
                if (B < 0 && A >= 0)
                    hi = std::min(hi, A / (A - B));
            }
            if (lo > hi + eps)
                return {};
            Point x = a + (b - a) * lo, y = a + (b - a) * hi;
            return norm(x - y) <= eps ? std::vector<Point>{x} : std::vector<Point>{x, y};
        }
        Real t = sa / (sa - sb);
        if (t < -eps || t > 1 + eps)
            return {};
        Point q = a + (b - a) * std::clamp(t, Real(0), Real(1));
        return onTriangle(q, u, v, w, eps) ? std::vector<Point>{q} : std::vector<Point>{};
    }
    // 两个非退化空间三角形是否有公共点；允许完全包含、共面重合及擦边。
    bool trianglesIntersect(std::array<Point, 3> a, std::array<Point, 3> b, Real eps = 1e-12L) {
        plane(a[0], a[1], a[2]);
        plane(b[0], b[1], b[2]);
        for (int i = 0; i < 3; ++i)
            if (!segmentTriangle(a[i], a[(i + 1) % 3], b[0], b[1], b[2], eps).empty() ||
                !segmentTriangle(b[i], b[(i + 1) % 3], a[0], a[1], a[2], eps).empty())
                return true;
        return false;
    }
    // 三点共线判定，允许重复点；eps 为点到非零方向的距离阈值。
    bool collinear(Point a, Point b, Point c, Real eps = 1e-12L) {
        return norm(b - a) <= eps || norm(cross(b - a, c - a)) <= eps * norm(b - a);
    }
    // 四点共面；前三点共线时四点总能落在某个平面内，返回 true。
    bool coplanar(Point a, Point b, Point c, Point d, Real eps = 1e-12L) {
        Point n = cross(b - a, c - a);
        return norm(n) <= eps || std::abs(dot(n, d - a)) <= eps * norm(n);
    }
    // 点是否在闭线段；零长线段按点处理。
    bool onSegment(Point p, Point a, Point b, Real eps = 1e-12L) {
        auto [u, v] = closestSegments(p, p, a, b, eps);
        return norm(u - v) <= eps;
    }
    // 两点是否在直线同侧；要求 p,q 与直线共面，边界点返回 false，非共面抛异常。
    bool sameSideOfLine(Point p, Point q, Line l, Real eps = 1e-12L) {
        detail::line(l);
        if (!coplanar(l.o, l.o + l.d, p, q, eps))
            throw std::invalid_argument("points not coplanar with line");
        Point a = cross(l.d, p - l.o), b = cross(l.d, q - l.o);
        return norm(a) > eps * norm(l.d) && norm(b) > eps * norm(l.d) && dot(a, b) > 0;
    }
    // 两点是否在平面严格同侧，面上的点返回 false。
    bool sameSideOfPlane(Point a, Point b, Plane p, Real eps = 1e-12L) {
        detail::plane(p);
        Real x = (dot(p.n, a) - p.d) / norm(p.n), y = (dot(p.n, b) - p.d) / norm(p.n);
        return (x > eps && y > eps) || (x < -eps && y < -eps);
    }
    // 两直线方向近平行（含重合），按相对角度阈值判定。
    bool parallel(Line a, Line b, Real eps = 1e-12L) {
        detail::line(a);
        detail::line(b);
        return norm(cross(a.d, b.d)) <= eps * norm(a.d) * norm(b.d);
    }
    // 两平面法向近平行（含重合）。
    bool parallel(Plane a, Plane b, Real eps = 1e-12L) {
        detail::plane(a);
        detail::plane(b);
        return norm(cross(a.n, b.n)) <= eps * norm(a.n) * norm(b.n);
    }
    // 两直线方向近垂直；不要求它们相交，异面也可方向垂直。
    bool perpendicular(Line a, Line b, Real eps = 1e-12L) {
        detail::line(a);
        detail::line(b);
        return std::abs(dot(a.d, b.d)) <= eps * norm(a.d) * norm(b.d);
    }
    // 两平面近垂直。
    bool perpendicular(Plane a, Plane b, Real eps = 1e-12L) {
        detail::plane(a);
        detail::plane(b);
        return std::abs(dot(a.n, b.n)) <= eps * norm(a.n) * norm(b.n);
    }
    // 直线方向是否平行平面，包含整线在面内的情况。
    bool parallel(Line l, Plane p, Real eps = 1e-12L) {
        detail::line(l);
        detail::plane(p);
        return std::abs(dot(l.d, p.n)) <= eps * norm(l.d) * norm(p.n);
    }
    // 两直线是否重合。
    bool same(Line a, Line b, Real eps = 1e-12L) {
        return parallel(a, b, eps) && distance(a.o, b) <= eps;
    }
    // 两平面是否重合，法向可以反向或缩放。
    bool same(Plane a, Plane b, Real eps = 1e-12L) {
        detail::plane(a);
        detail::plane(b);
        return parallel(a, b, eps) && distance(a.n * (a.d / dot(a.n, a.n)), b) <= eps;
    }
    // 两直线唯一交点；异面/平行/重合返回空，重合的无穷交点不冒充单点。
    std::optional<Point> intersect(Line a, Line b, Real eps = 1e-12L) {
        if (parallel(a, b, eps))
            return std::nullopt;
        auto [u, v] = closest(a, b, eps);
        return norm(u - v) <= eps ? std::optional<Point>((u + v) / 2) : std::nullopt;
    }
    // 两条无向直线较小夹角的 cos，返回 [0,1]。
    Real angleCos(Line a, Line b) {
        detail::line(a);
        detail::line(b);
        return std::clamp(std::abs(dot(a.d, b.d)) / (norm(a.d) * norm(b.d)), Real(0), Real(1));
    }
    // 两平面较小夹角的 cos，返回 [0,1]。
    Real angleCos(Plane a, Plane b) {
        detail::plane(a);
        detail::plane(b);
        return std::clamp(std::abs(dot(a.n, b.n)) / (norm(a.n) * norm(b.n)), Real(0), Real(1));
    }
    // 直线与平面较小夹角的 sin，返回 [0,1]。
    Real angleSin(Line l, Plane p) {
        detail::line(l);
        detail::plane(p);
        return std::clamp(std::abs(dot(l.d, p.n)) / (norm(l.d) * norm(p.n)), Real(0), Real(1));
    }
    // 四面体有向体积的六倍，实际体积取 abs(value)/6，共面为零。
    Real signedVolume6(Point a, Point b, Point c, Point d) {
        return dot(cross(b - a, c - a), d - a);
    }

}
```
