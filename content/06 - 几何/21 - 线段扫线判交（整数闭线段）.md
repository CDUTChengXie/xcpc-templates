# 线段扫线判交（整数闭线段）

整数坐标 |x|,|y|<=1e6，超出抛 invalid_argument；__int128 做全部判定，无 eps。 端点触碰/共线重叠也算交；点段和竖直段支持；返回下标 in [0,n)，无交返回 nullopt。 扫线只存非竖直段并在相邻段发现交时立即终止；该不变量保证活动段次序不交换，不能复用本树枚举所有交点（需交点事件）。

O(n log n) 时间，O(n) 空间；不枚举交点。

```cpp
namespace SegmentSweep {
    using I = long long;
    using Int = __int128;
    struct Point {
        I x, y;
    };
    struct Segment {
        Point a, b;
    };
    namespace detail {
        Int turn(Point a, Point b, Point c) {
            return Int(b.x - a.x) * (c.y - a.y) - Int(b.y - a.y) * (c.x - a.x);
        }
        bool on(Point p, Point a, Point b) {
            return turn(a, b, p) == 0 && std::min(a.x, b.x) <= p.x && p.x <= std::max(a.x, b.x) &&
                   std::min(a.y, b.y) <= p.y && p.y <= std::max(a.y, b.y);
        }
        bool intersect(Segment a, Segment b) {
            Int x = turn(a.a, a.b, b.a), y = turn(a.a, a.b, b.b), z = turn(b.a, b.b, a.a),
                w = turn(b.a, b.b, a.b);
            if (x == 0 && on(b.a, a.a, a.b))
                return true;
            if (y == 0 && on(b.b, a.a, a.b))
                return true;
            if (z == 0 && on(a.a, b.a, b.b))
                return true;
            if (w == 0 && on(a.b, b.a, b.b))
                return true;
            return ((x < 0 && y > 0) || (x > 0 && y < 0)) && ((z < 0 && w > 0) || (z > 0 && w < 0));
        }
        struct Query {
            I y;
        };
        struct Compare {
            using is_transparent = void;
            const std::vector<Segment> *p;
            const I *x;
            Int numerator(int id) const {
                auto s = (*p)[id];
                return Int(s.a.y) * (s.b.x - s.a.x) + Int(s.b.y - s.a.y) * (*x - s.a.x);
            }
            I denominator(int id) const {
                return (*p)[id].b.x - (*p)[id].a.x;
            }
            bool operator()(int a, int b) const {
                Int lhs = numerator(a) * denominator(b), rhs = numerator(b) * denominator(a);
                if (lhs != rhs)
                    return lhs < rhs;
                return a < b;
            }
            bool operator()(int a, Query b) const {
                return numerator(a) < Int(b.y) * denominator(a);
            }
            bool operator()(Query a, int b) const {
                return Int(a.y) * denominator(b) < numerator(b);
            }
        };
    }
    // 检测闭线段相交，返回第一对原输入编号；竖直/点段单独处理，输入可为空。
    std::optional<std::array<int, 2>> findIntersection(std::vector<Segment> p) {
        int n = int(p.size());
        struct Event {
            I x;
            int id, type;
        };
        std::vector<Event> events;
        for (int i = 0; i < n; ++i) {
            for (Point q : std::array<Point, 2>{p[i].a, p[i].b})
                if (q.x < -1000000 || q.x > 1000000 || q.y < -1000000 || q.y > 1000000)
                    throw std::invalid_argument("SegmentSweep: coordinate exceeds 1e6");
            if (p[i].a.x > p[i].b.x)
                std::swap(p[i].a, p[i].b);
            if (p[i].a.x == p[i].b.x) {
                if (p[i].a.y > p[i].b.y)
                    std::swap(p[i].a, p[i].b);
                events.push_back({p[i].a.x, i, 1});
            } else {
                events.push_back({p[i].a.x, i, 0});
                events.push_back({p[i].b.x, i, 2});
            }
        }
        std::sort(events.begin(), events.end(), [](Event a, Event b) {
            return std::tie(a.x, a.type, a.id) < std::tie(b.x, b.type, b.id);
        });
        I x = 0;
        detail::Compare cmp{&p, &x};
        std::set<int, detail::Compare> active(cmp);
        std::vector<decltype(active)::iterator> handles(n);
        auto hit = [&](int a, int b) { return detail::intersect(p[a], p[b]); };
        for (int l = 0; l < int(events.size());) {
            int r = l + 1;
            while (r < int(events.size()) && events[r].x == events[l].x)
                ++r;
            x = events[l].x;
            std::vector<int> vertical;
            for (int k = l; k < r; ++k) {
                auto e = events[k];
                if (e.type == 1) {
                    vertical.push_back(e.id);
                    continue;
                }
                if (e.type != 0)
                    continue;
                auto it = active.insert(e.id).first;
                handles[e.id] = it;
                if (it != active.begin() && hit(*std::prev(it), *it))
                    return std::array<int, 2>{*std::prev(it), *it};
                if (std::next(it) != active.end() && hit(*it, *std::next(it)))
                    return std::array<int, 2>{*it, *std::next(it)};
            }
            std::sort(vertical.begin(), vertical.end(), [&](int a, int b) {
                return std::tie(p[a].a.y, p[a].b.y) < std::tie(p[b].a.y, p[b].b.y);
            });
            for (int k = 0; k < int(vertical.size()); ++k) {
                int id = vertical[k];
                if (k && p[vertical[k - 1]].b.y >= p[id].a.y)
                    return std::array<int, 2>{vertical[k - 1], id};
                auto it = active.lower_bound(detail::Query{p[id].a.y});
                if (it != active.end() &&
                    cmp.numerator(*it) <= Int(p[id].b.y) * cmp.denominator(*it))
                    return std::array<int, 2>{id, *it};
            }
            for (int k = l; k < r; ++k)
                if (events[k].type == 2) {
                    auto it = handles[events[k].id], next = std::next(it);
                    if (it != active.begin() && next != active.end() && hit(*std::prev(it), *next))
                        return std::array<int, 2>{*std::prev(it), *next};
                    active.erase(it);
                }
            l = r;
        }
        return std::nullopt;
    }
}
```
