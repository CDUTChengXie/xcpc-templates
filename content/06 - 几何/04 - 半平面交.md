# 半平面交

仅用于已保证有界的交集；坐标有限、每条直线两端不同；用 `Line<double>`(a,b) 保留 a→b 左侧，平行同向约束自动去重。 有界非退化输入顶点按逆时针排列；空向量与 bounded 标志都不能区分空集与无界（无界可能误标 true），调用方须保证有界。

O(n log n)，空间 O(n)。

```cpp
#ifndef XCPC_TEMPLATE_POINT
#define XCPC_TEMPLATE_POINT

template <class T> struct Point {
    T x;
    T y;

    // 构造点/向量，省略坐标得到原点。
    Point(const T &x_ = 0, const T &y_ = 0) : x(x_), y(y_) {}

    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator+=(const Point &p) & {
        x += p.x;
        y += p.y;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator-=(const Point &p) & {
        x -= p.x;
        y -= p.y;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator*=(const T &v) & {
        x *= v;
        y *= v;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator/=(const T &v) & {
        x /= v;
        y /= v;
        return *this;
    }
    // 返回反向向量。
    Point operator-() const {
        return Point(-x, -y);
    }
    // 返回两向量之和。
    friend Point operator+(Point a, const Point &b) {
        return a += b;
    }
    // 返回两向量之差。
    friend Point operator-(Point a, const Point &b) {
        return a -= b;
    }
    // 返回标量倍向量。
    friend Point operator*(Point a, const T &b) {
        return a *= b;
    }
    // 返回向量除以非零标量。
    friend Point operator/(Point a, const T &b) {
        return a /= b;
    }
    // 返回标量左乘向量。
    friend Point operator*(const T &a, Point b) {
        return b *= a;
    }
    // 返回坐标严格相等，不含 eps。
    friend bool operator==(const Point &a, const Point &b) {
        return a.x == b.x && a.y == b.y;
    }
    // 返回字典序比较（先 x 后 y），不含 eps。
    friend bool operator<(const Point &a, const Point &b) {
        return a.x < b.x || (a.x == b.x && a.y < b.y);
    }
    // 读入 x、y，返回输入流。
    friend std::istream &operator>>(std::istream &is, Point &p) {
        return is >> p.x >> p.y;
    }
    // 输出 x、y，返回输出流。
    friend std::ostream &operator<<(std::ostream &os, const Point &p) {
        return os << p.x << " " << p.y;
    }
};

template <class T> struct Line {
    Point<T> a;
    Point<T> b;

    // 构造有向线段端点；直线运算须保证两端点不同。
    Line(const Point<T> &a_ = Point<T>(), const Point<T> &b_ = Point<T>()) : a(a_), b(b_) {}
};

template <class T>
// 返回两向量点积。
T dot(const Point<T> &a, const Point<T> &b) {
    return a.x * b.x + a.y * b.y;
}

template <class T>
// 返回两向量二维叉积（a 到 b 逆时针为正）。
T cross(const Point<T> &a, const Point<T> &b) {
    return a.x * b.y - a.y * b.x;
}

template <class T>
// 返回向量长度平方。
T square(const Point<T> &p) {
    return dot(p, p);
}

template <class T>
// 返回向量长度，结果为 double。
double length(const Point<T> &p) {
    return std::sqrt(square(p));
}

template <class T>
// 返回两点距离，结果为 double。
double distance(const Point<T> &a, const Point<T> &b) {
    return length(a - b);
}

template <class T>
// 返回点到直线距离；两端点须不同。
double distancePL(const Point<T> &p, const Line<T> &l) {
    return std::abs(cross(l.a - l.b, l.a - p)) / length(l.b - l.a);
}

template <class T>
// 返回点到闭线段距离；端点须不同，零长段会除零。
double distancePS(const Point<T> &p, const Line<T> &l) {
    if (dot(p - l.a, l.b - l.a) < 0) {
        return distance(p, l.a);
    }
    if (dot(p - l.b, l.a - l.b) < 0) {
        return distance(p, l.b);
    }
    return distancePL(p, l);
}

template <class T>
// 返回两非平行直线交点；整数类型用 long double 计算后截断，精度场景改用浮点坐标。
Point<T> lineIntersection(const Line<T> &l1, const Line<T> &l2) {
    long double c1 = (long double)cross(l2.b - l2.a, l1.a - l2.a);
    long double c2 = (long double)cross(l2.b - l2.a, l1.a - l1.b);
    long double t = c1 / c2;
    return Point<T>(l1.a.x + (long double)(l1.b.x - l1.a.x) * t,
                    l1.a.y + (long double)(l1.b.y - l1.a.y) * t);
}

template <class T>
// 判断点是否在闭线段上（含端点；浮点需自行加 eps）。
bool pointOnSegment(const Point<T> &p, const Line<T> &l) {
    return cross(p - l.a, l.b - l.a) == 0 && std::min(l.a.x, l.b.x) <= p.x &&
           p.x <= std::max(l.a.x, l.b.x) && std::min(l.a.y, l.b.y) <= p.y &&
           p.y <= std::max(l.a.y, l.b.y);
}

template <class T>
// 判断点是否在简单多边形闭区域内（边界算内，与绕向无关，O(n)）。
bool pointInPolygon(const Point<T> &a, const std::vector<Point<T>> &p) {
    int n = p.size();
    for (int i = 0; i < n; i++) {
        if (pointOnSegment(a, Line(p[i], p[(i + 1) % n]))) {
            return true;
        }
    }
    int t = 0;
    for (int i = 0; i < n; i++) {
        auto u = p[i];
        auto v = p[(i + 1) % n];
        if (u.x < a.x && v.x >= a.x && cross(v - u, a - u) > 0) {
            t ^= 1;
        }
        if (u.x >= a.x && v.x < a.x && cross(u - v, a - v) > 0) {
            t ^= 1;
        }
    }
    return t == 1;
}

template <class T>
// 返回两闭线段相交分类：0 不相交、1 严格相交、2 重叠（返回两端点）、3 端点相交。
std::tuple<int, Point<T>, Point<T>> segmentIntersection(const Line<T> &l1, const Line<T> &l2) {
    if (l1.a == l1.b)
        return {pointOnSegment(l1.a, l2) ? 3 : 0, l1.a, l1.a};
    if (l2.a == l2.b)
        return {pointOnSegment(l2.a, l1) ? 3 : 0, l2.a, l2.a};
    if (std::max(l1.a.x, l1.b.x) < std::min(l2.a.x, l2.b.x) ||
        std::min(l1.a.x, l1.b.x) > std::max(l2.a.x, l2.b.x) ||
        std::max(l1.a.y, l1.b.y) < std::min(l2.a.y, l2.b.y) ||
        std::min(l1.a.y, l1.b.y) > std::max(l2.a.y, l2.b.y)) {
        return {0, Point<T>(), Point<T>()};
    }
    if (cross(l1.b - l1.a, l2.b - l2.a) == 0) {
        if (cross(l1.b - l1.a, l2.a - l1.a) != 0) {
            return {0, Point<T>(), Point<T>()};
        }
        Point<T> p1(std::max(std::min(l1.a.x, l1.b.x), std::min(l2.a.x, l2.b.x)),
                    std::max(std::min(l1.a.y, l1.b.y), std::min(l2.a.y, l2.b.y)));
        Point<T> p2(std::min(std::max(l1.a.x, l1.b.x), std::max(l2.a.x, l2.b.x)),
                    std::min(std::max(l1.a.y, l1.b.y), std::max(l2.a.y, l2.b.y)));
        if (!pointOnSegment(p1, l1)) {
            std::swap(p1.y, p2.y);
        }
        if (p1 == p2) {
            return {3, p1, p2};
        }
        return {2, p1, p2};
    }
    auto cp1 = cross(l2.a - l1.a, l2.b - l1.a);
    auto cp2 = cross(l2.a - l1.b, l2.b - l1.b);
    auto cp3 = cross(l1.a - l2.a, l1.b - l2.a);
    auto cp4 = cross(l1.a - l2.b, l1.b - l2.b);
    if ((cp1 > 0 && cp2 > 0) || (cp1 < 0 && cp2 < 0) || (cp3 > 0 && cp4 > 0) ||
        (cp3 < 0 && cp4 < 0)) {
        return {0, Point<T>(), Point<T>()};
    }
    Point p = lineIntersection(l1, l2);
    if (cp1 != 0 && cp2 != 0 && cp3 != 0 && cp4 != 0) {
        return {1, p, p};
    }
    return {3, p, p};
}

template <class T>
// 返回两闭线段最短距离，相交或重叠为 0。
double distanceSS(const Line<T> &l1, const Line<T> &l2) {
    if (std::get<0>(segmentIntersection(l1, l2)) != 0) {
        return 0.0;
    }
    return std::min(
        {distancePS(l1.a, l2), distancePS(l1.b, l2), distancePS(l2.a, l1), distancePS(l2.b, l1)});
}

template <class T>
// 返回简单多边形两倍无向面积，顶点须按边界顺序；空集或单点返回 0。
T polygonArea2(const std::vector<Point<T>> &p) {
    T s = 0;
    int n = p.size();
    for (int i = 0; i < n; i++) {
        s += cross(p[i], p[(i + 1) % n]);
    }
    return std::abs(s);
}

// 返回已保证有界的非退化交集按逆时针排列的顶点；空结果不能区分空集/无界/退化。
std::vector<Point<double>> halfPlaneIntersection(std::vector<Line<double>> ls) {
    using L = Line<double>;
    using Pt = Point<double>;
    std::sort(ls.begin(), ls.end(), [&](const L &a, const L &b) {
        auto angle = [](const Pt &v) { return std::atan2((double)v.y, (double)v.x); };
        double ta = angle(a.b - a.a), tb = angle(b.b - b.a);
        if (std::abs(ta - tb) > 1e-18) {
            return ta < tb;
        }
        // 同方向：取更靠左（截距更小）的，保证等价去重
        return cross(b.b - b.a, a.a - b.a) > 0;
    });
    auto inter = [](const L &a, const L &b) {
        Pt u = a.a, v = a.b, p = b.a, q = b.b;
        Pt du = v - u, dv = q - p;
        double t = cross(p - u, dv) / cross(du, dv);
        return u + du * t;
    };
    auto onRight = [](const Pt &p, const L &l) { return cross(l.b - l.a, p - l.a) < -1e-12; };
    int n = ls.size();
    std::vector<L> q(n);
    int h = 0, t = 0;
    for (auto &l : ls) {
        while (t - h >= 2 && onRight(inter(q[t - 1], q[t - 2]), l)) {
            t--;
        }
        while (t - h >= 2 && onRight(inter(q[h], q[h + 1]), l)) {
            h++;
        }
        // 有界非退化交集的相邻有效方向跨度须小于半周；不能对平行约束求交。
        if (t - h) {
            double turn = cross(q[t - 1].b - q[t - 1].a, l.b - l.a);
            if (turn < 0 || (turn == 0 && dot(l.b - l.a, q[t - 1].b - q[t - 1].a) < 0))
                return {};
        }
        if (t - h && std::abs(std::atan2((double)(l.b - l.a).y, (double)(l.b - l.a).x) -
                              std::atan2((double)(q[t - 1].b - q[t - 1].a).y,
                                         (double)(q[t - 1].b - q[t - 1].a).x)) < 1e-15) {
            // 平行且反向：无界为空
            if (cross(l.b - l.a, q[t - 1].a - l.a) < 0) {
                return {};
            }
            continue;
        }
        q[t++] = l;
    }
    while (t - h >= 3 && onRight(inter(q[t - 1], q[t - 2]), q[h])) {
        t--;
    }
    if (t - h < 3) {
        return {};
    }
    std::vector<Pt> res;
    for (int i = h; i < t - 1; i++) {
        res.push_back(inter(q[i], q[i + 1]));
    }
    res.push_back(inter(q[t - 1], q[h]));
    return res;
}

// poly 为有界非退化交集的逆时针顶点；bounded 仅为实现的闭合检查，不作无界输入判据。
struct HalfPlaneResult {
    std::vector<Point<double>> poly; // 顶点，逆时针
    bool bounded = false;            // 闭合检查标志；无界输入不能依赖它
};

// 返回顶点与闭合标志；少于 3 条约束返回空 poly/false，不能可靠判无界。
HalfPlaneResult halfPlaneIntersectionResult(std::vector<Line<double>> ls) {
    using L = Line<double>;
    using Pt = Point<double>;
    HalfPlaneResult out;
    if (ls.size() < 3) {
        return out; // 少于 3 条半平面不可能围出有界区域
    }
    std::sort(ls.begin(), ls.end(), [&](const L &a, const L &b) {
        auto angle = [](const Pt &v) { return std::atan2((double)v.y, (double)v.x); };
        double ta = angle(a.b - a.a), tb = angle(b.b - b.a);
        if (std::abs(ta - tb) > 1e-18) {
            return ta < tb;
        }
        return cross(b.b - b.a, a.a - b.a) > 0; // 同方向保留更严格的
    });
    auto inter = [](const L &a, const L &b) {
        Pt u = a.a, du = a.b - a.a, dv = b.b - b.a;
        return u + du * (cross(b.a - u, dv) / cross(du, dv));
    };
    auto onRight = [](const Pt &p, const L &l) { return cross(l.b - l.a, p - l.a) < -1e-12; };
    auto dir = [](const L &l) { return std::atan2((double)(l.b - l.a).y, (double)(l.b - l.a).x); };

    int n = int(ls.size());
    std::vector<L> q(n);
    int h = 0, t = 0;
    for (auto &l : ls) {
        while (t - h >= 2 && onRight(inter(q[t - 1], q[t - 2]), l)) {
            --t;
        }
        while (t - h >= 2 && onRight(inter(q[h], q[h + 1]), l)) {
            ++h;
        }
        if (t - h) {
            double turn = cross(q[t - 1].b - q[t - 1].a, l.b - l.a);
            if (turn < 0 || (turn == 0 && dot(l.b - l.a, q[t - 1].b - q[t - 1].a) < 0))
                return {};
        }
        if (t - h && std::abs(dir(l) - dir(q[t - 1])) < 1e-15) {
            // 反向平行且截距冲突即无解
            if (cross(l.b - l.a, q[t - 1].a - l.a) < 0) {
                return out; // poly 为空、bounded 为 false：交集为空
            }
            continue;
        }
        q[t++] = l;
    }
    while (t - h >= 3 && onRight(inter(q[t - 1], q[t - 2]), q[h])) {
        --t;
    }
    if (t - h < 3) {
        return out; // 有效约束不足 3 条：必然无界（不是无解）
    }
    // 首尾约束平行时无闭合交点（会除零），按无界处理
    auto dirv = [](const L &l) { return l.b - l.a; };
    if (std::abs(cross(dirv(q[t - 1]), dirv(q[h]))) < 1e-18) {
        return out;
    }
    out.poly.reserve(t - h);
    for (int i = h; i < t - 1; i++) {
        out.poly.push_back(inter(q[i], q[i + 1]));
    }
    out.poly.push_back(inter(q[t - 1], q[h]));

    // 合并重合点并去掉共线中间点，使顶点数等于真正的边数
    std::vector<Pt> uniq;
    auto samePt = [](const Pt &p, const Pt &q) {
        return std::abs(p.x - q.x) <= 1e-9 && std::abs(p.y - q.y) <= 1e-9;
    };
    for (const Pt &p : out.poly) {
        if (uniq.empty() || !samePt(uniq.back(), p)) {
            uniq.push_back(p);
        }
    }
    while (uniq.size() > 1 && samePt(uniq.front(), uniq.back())) {
        uniq.pop_back();
    }
    out.poly.clear();
    int m = int(uniq.size());
    for (int i = 0; i < m; i++) {
        Pt prev = uniq[(i + m - 1) % m], cur = uniq[i], next = uniq[(i + 1) % m];
        if (std::abs(cross(next - prev, cur - prev)) > 1e-9) {
            out.poly.push_back(cur); // 只保留真正拐弯的顶点
        }
    }

    // 约束前后交点重合即为冗余约束
    int active = 0, qsz = t - h;
    for (int i = 0; i < qsz; i++) {
        Pt from = inter(q[h + (i + qsz - 1) % qsz], q[h + i]);
        Pt to = inter(q[h + i], q[h + (i + 1) % qsz]);
        if (!samePt(from, to)) {
            ++active;
        }
    }
    // 有界 ⟺ 每条真正生效的约束都在多边形上贡献了一条边
    out.bounded = !out.poly.empty() && int(out.poly.size()) == active;
    return out;
}
```
