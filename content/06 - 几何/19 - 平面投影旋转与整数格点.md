# 平面投影旋转与整数格点

实数输入有限；整数接口 |x|,|y|<=1e9；格点多边形简单、无洞、非零面积，按边界给点。 实数 P=`Point<long double>`；整数 IP=`Point<long long>`；整数判定及面积用 __int128。 eps 为绝对阈值；quadratic 的空 vector 为无根，nullopt 表示恒等于零有无穷根。

除格点计数/凸性 O(n) 外均 O(1)，排序六个距离/两个根仍 O(1)；空间 O(1)。

```cpp
#ifndef XCPC_TEMPLATE_POINT
#define XCPC_TEMPLATE_POINT

template <class T> struct Point {
    T x;
    T y;

    // 构造点/向量，省略坐标得到原点。
    Point(const T &x_ = 0, const T &y_ = 0) : x(x_), y(y_) {}

    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator+=(const Point &p) & {
        x += p.x;
        y += p.y;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator-=(const Point &p) & {
        x -= p.x;
        y -= p.y;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator*=(const T &v) & {
        x *= v;
        y *= v;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator/=(const T &v) & {
        x /= v;
        y /= v;
        return *this;
    }
    // 返回反向向量。
    Point operator-() const {
        return Point(-x, -y);
    }
    // 返回两向量之和。
    friend Point operator+(Point a, const Point &b) {
        return a += b;
    }
    // 返回两向量之差。
    friend Point operator-(Point a, const Point &b) {
        return a -= b;
    }
    // 返回标量倍向量。
    friend Point operator*(Point a, const T &b) {
        return a *= b;
    }
    // 返回向量除以非零标量。
    friend Point operator/(Point a, const T &b) {
        return a /= b;
    }
    // 返回标量左乘向量。
    friend Point operator*(const T &a, Point b) {
        return b *= a;
    }
    // 返回坐标严格相等，不含 eps。
    friend bool operator==(const Point &a, const Point &b) {
        return a.x == b.x && a.y == b.y;
    }
    // 返回字典序比较（先 x 后 y），不含 eps。
    friend bool operator<(const Point &a, const Point &b) {
        return a.x < b.x || (a.x == b.x && a.y < b.y);
    }
    // 读入 x、y，返回输入流。
    friend std::istream &operator>>(std::istream &is, Point &p) {
        return is >> p.x >> p.y;
    }
    // 输出 x、y，返回输出流。
    friend std::ostream &operator<<(std::ostream &os, const Point &p) {
        return os << p.x << " " << p.y;
    }
};

template <class T> struct Line {
    Point<T> a;
    Point<T> b;

    // 构造有向线段端点；直线运算须保证两端点不同。
    Line(const Point<T> &a_ = Point<T>(), const Point<T> &b_ = Point<T>()) : a(a_), b(b_) {}
};

template <class T>
// 返回两向量点积。
T dot(const Point<T> &a, const Point<T> &b) {
    return a.x * b.x + a.y * b.y;
}

template <class T>
// 返回两向量二维叉积（a 到 b 逆时针为正）。
T cross(const Point<T> &a, const Point<T> &b) {
    return a.x * b.y - a.y * b.x;
}

template <class T>
// 返回向量长度平方。
T square(const Point<T> &p) {
    return dot(p, p);
}

template <class T>
// 返回向量长度，结果为 double。
double length(const Point<T> &p) {
    return std::sqrt(square(p));
}

template <class T>
// 返回两点距离，结果为 double。
double distance(const Point<T> &a, const Point<T> &b) {
    return length(a - b);
}

template <class T>
// 返回点到直线距离；两端点须不同。
double distancePL(const Point<T> &p, const Line<T> &l) {
    return std::abs(cross(l.a - l.b, l.a - p)) / length(l.b - l.a);
}

template <class T>
// 返回点到闭线段距离；端点须不同，零长段会除零。
double distancePS(const Point<T> &p, const Line<T> &l) {
    if (dot(p - l.a, l.b - l.a) < 0) {
        return distance(p, l.a);
    }
    if (dot(p - l.b, l.a - l.b) < 0) {
        return distance(p, l.b);
    }
    return distancePL(p, l);
}

template <class T>
// 返回两非平行直线交点；整数类型用 long double 计算后截断，精度场景改用浮点坐标。
Point<T> lineIntersection(const Line<T> &l1, const Line<T> &l2) {
    long double c1 = (long double)cross(l2.b - l2.a, l1.a - l2.a);
    long double c2 = (long double)cross(l2.b - l2.a, l1.a - l1.b);
    long double t = c1 / c2;
    return Point<T>(l1.a.x + (long double)(l1.b.x - l1.a.x) * t,
                    l1.a.y + (long double)(l1.b.y - l1.a.y) * t);
}

template <class T>
// 判断点是否在闭线段上（含端点；浮点需自行加 eps）。
bool pointOnSegment(const Point<T> &p, const Line<T> &l) {
    return cross(p - l.a, l.b - l.a) == 0 && std::min(l.a.x, l.b.x) <= p.x &&
           p.x <= std::max(l.a.x, l.b.x) && std::min(l.a.y, l.b.y) <= p.y &&
           p.y <= std::max(l.a.y, l.b.y);
}

template <class T>
// 判断点是否在简单多边形闭区域内（边界算内，与绕向无关，O(n)）。
bool pointInPolygon(const Point<T> &a, const std::vector<Point<T>> &p) {
    int n = p.size();
    for (int i = 0; i < n; i++) {
        if (pointOnSegment(a, Line(p[i], p[(i + 1) % n]))) {
            return true;
        }
    }
    int t = 0;
    for (int i = 0; i < n; i++) {
        auto u = p[i];
        auto v = p[(i + 1) % n];
        if (u.x < a.x && v.x >= a.x && cross(v - u, a - u) > 0) {
            t ^= 1;
        }
        if (u.x >= a.x && v.x < a.x && cross(u - v, a - v) > 0) {
            t ^= 1;
        }
    }
    return t == 1;
}

template <class T>
// 返回两闭线段相交分类：0 不相交、1 严格相交、2 重叠（返回两端点）、3 端点相交。
std::tuple<int, Point<T>, Point<T>> segmentIntersection(const Line<T> &l1, const Line<T> &l2) {
    if (l1.a == l1.b)
        return {pointOnSegment(l1.a, l2) ? 3 : 0, l1.a, l1.a};
    if (l2.a == l2.b)
        return {pointOnSegment(l2.a, l1) ? 3 : 0, l2.a, l2.a};
    if (std::max(l1.a.x, l1.b.x) < std::min(l2.a.x, l2.b.x) ||
        std::min(l1.a.x, l1.b.x) > std::max(l2.a.x, l2.b.x) ||
        std::max(l1.a.y, l1.b.y) < std::min(l2.a.y, l2.b.y) ||
        std::min(l1.a.y, l1.b.y) > std::max(l2.a.y, l2.b.y)) {
        return {0, Point<T>(), Point<T>()};
    }
    if (cross(l1.b - l1.a, l2.b - l2.a) == 0) {
        if (cross(l1.b - l1.a, l2.a - l1.a) != 0) {
            return {0, Point<T>(), Point<T>()};
        }
        Point<T> p1(std::max(std::min(l1.a.x, l1.b.x), std::min(l2.a.x, l2.b.x)),
                    std::max(std::min(l1.a.y, l1.b.y), std::min(l2.a.y, l2.b.y)));
        Point<T> p2(std::min(std::max(l1.a.x, l1.b.x), std::max(l2.a.x, l2.b.x)),
                    std::min(std::max(l1.a.y, l1.b.y), std::max(l2.a.y, l2.b.y)));
        if (!pointOnSegment(p1, l1)) {
            std::swap(p1.y, p2.y);
        }
        if (p1 == p2) {
            return {3, p1, p2};
        }
        return {2, p1, p2};
    }
    auto cp1 = cross(l2.a - l1.a, l2.b - l1.a);
    auto cp2 = cross(l2.a - l1.b, l2.b - l1.b);
    auto cp3 = cross(l1.a - l2.a, l1.b - l2.a);
    auto cp4 = cross(l1.a - l2.b, l1.b - l2.b);
    if ((cp1 > 0 && cp2 > 0) || (cp1 < 0 && cp2 < 0) || (cp3 > 0 && cp4 > 0) ||
        (cp3 < 0 && cp4 < 0)) {
        return {0, Point<T>(), Point<T>()};
    }
    Point p = lineIntersection(l1, l2);
    if (cp1 != 0 && cp2 != 0 && cp3 != 0 && cp4 != 0) {
        return {1, p, p};
    }
    return {3, p, p};
}

template <class T>
// 返回两闭线段最短距离，相交或重叠为 0。
double distanceSS(const Line<T> &l1, const Line<T> &l2) {
    if (std::get<0>(segmentIntersection(l1, l2)) != 0) {
        return 0.0;
    }
    return std::min(
        {distancePS(l1.a, l2), distancePS(l1.b, l2), distancePS(l2.a, l1), distancePS(l2.b, l1)});
}

template <class T>
// 返回简单多边形两倍无向面积，顶点须按边界顺序；空集或单点返回 0。
T polygonArea2(const std::vector<Point<T>> &p) {
    T s = 0;
    int n = p.size();
    for (int i = 0; i < n; i++) {
        s += cross(p[i], p[(i + 1) % n]);
    }
    return std::abs(s);
}

namespace GeoSupplement {
    using Real = long double;
    using P = Point<Real>;
    using IP = Point<long long>;
    using Int = __int128;
    namespace detail {
        void valid(IP p) {
            if (p.x < -1000000000LL || p.x > 1000000000LL || p.y < -1000000000LL ||
                p.y > 1000000000LL)
                throw std::invalid_argument("coordinate exceeds 1e9");
        }
        Int dist2(IP a, IP b) {
            Int x = a.x - b.x, y = a.y - b.y;
            return x * x + y * y;
        }
        std::array<Int, 6> distances(std::array<IP, 4> p) {
            std::array<Int, 6> d;
            int k = 0;
            for (IP a : p)
                valid(a);
            for (int i = 0; i < 4; ++i)
                for (int j = i + 1; j < 4; ++j)
                    d[k++] = dist2(p[i], p[j]);
            std::sort(d.begin(), d.end());
            return d;
        }
    }
    // 向量逆时针旋转 rad 弧度；点绕 o 旋转写 o+rotate(p-o,rad)。
    P rotate(P v, Real rad) {
        Real c = std::cos(rad), s = std::sin(rad);
        return {v.x * c - v.y * s, v.x * s + v.y * c};
    }
    // 返回单位向量；长度<=eps 返回空。
    std::optional<P> unit(P v, Real eps = 1e-12L) {
        Real d = std::sqrt(square(v));
        return d <= eps ? std::nullopt : std::optional<P>(v / d);
    }
    // 返回两向量夹角 [0,pi] 弧度；零向量为空，acos 入参夹到 [-1,1]。
    std::optional<Real> angleBetween(P a, P b, Real eps = 1e-12L) {
        auto x = unit(a, eps), y = unit(b, eps);
        if (!x || !y)
            return std::nullopt;
        return std::acos(std::clamp(dot(*x, *y), Real(-1), Real(1)));
    }
    // 点 p 在无限直线 a-b 上的垂足；a==b 抛异常。
    P project(P p, P a, P b) {
        P d = b - a;
        Real l = square(d);
        if (l == 0)
            throw std::invalid_argument("zero line direction");
        return a + d * (dot(p - a, d) / l);
    }
    // 点 p 到闭线段 a-b 的最近点；允许 a==b。
    P nearestSegment(P p, P a, P b) {
        P d = b - a;
        Real l = square(d);
        return l == 0 ? a : a + d * std::clamp(dot(p - a, d) / l, Real(0), Real(1));
    }
    // 两个整数点建最简 ax+by+c=0；首个非零 a/b 取正，共点为空。
    std::optional<std::array<long long, 3>> equation(IP p, IP q) {
        detail::valid(p);
        detail::valid(q);
        long long a = p.y - q.y, b = q.x - p.x, c = -(a * p.x + b * p.y);
        if (a == 0 && b == 0)
            return std::nullopt;
        long long g = std::gcd(std::gcd(a, b), c);
        a /= g;
        b /= g;
        c /= g;
        if (a < 0 || (a == 0 && b < 0)) {
            a = -a;
            b = -b;
            c = -c;
        }
        return std::array<long long, 3>{a, b, c};
    }
    // 最简斜率 dy/dx，分母非负；竖直为 (1,0)，共点返回空。
    std::optional<std::pair<long long, long long>> slope(IP p, IP q) {
        detail::valid(p);
        detail::valid(q);
        long long x = q.x - p.x, y = q.y - p.y;
        if (x == 0)
            return y == 0 ? std::nullopt : std::optional<std::pair<long long, long long>>({1, 0});
        long long g = std::gcd(x, y);
        x /= g;
        y /= g;
        if (x < 0) {
            x = -x;
            y = -y;
        }
        return std::pair{y, x};
    }
    // 一般式 ax+by+c=0 转直线上两个不同实数点；法向近零返回空。
    std::optional<std::pair<P, P>> equationPoints(Real a, Real b, Real c, Real eps = 1e-12L) {
        if (std::hypot(a, b) <= eps)
            return std::nullopt;
        P p = std::abs(a) >= std::abs(b) ? P{-c / a, 0} : P{0, -c / b};
        Real len = std::hypot(a, b);
        return std::pair{p, p + P{-b / len, a / len}};
    }
    // 解 ax²+bx+c=0 的全部实根；q 公式减少大根/小根相消，退化为一次或常数也可。
    std::optional<std::vector<Real>> quadratic(Real a, Real b, Real c, Real eps = 1e-12L) {
        if (std::abs(a) <= eps) {
            if (std::abs(b) > eps)
                return std::vector<Real>{-c / b};
            if (std::abs(c) <= eps)
                return std::nullopt;
            return std::vector<Real>{};
        }
        Real D = b * b - 4 * a * c;
        if (D < -eps)
            return std::vector<Real>{};
        if (std::abs(D) <= eps)
            return std::vector<Real>{-b / (2 * a)};
        Real q = -(b + std::copysign(std::sqrt(D), b)) / 2;
        std::vector<Real> roots{q / a, c / q};
        std::sort(roots.begin(), roots.end());
        return roots;
    }
    // 四个无序整数点能否组成非退化矩形。
    bool isRectangle(std::array<IP, 4> p) {
        auto d = detail::distances(p);
        return d[0] > 0 && d[0] == d[1] && d[2] == d[3] && d[4] == d[5] && d[4] == d[0] + d[2];
    }
    // 四个无序整数点能否组成正方形，重合点返回 false。
    bool isSquare(std::array<IP, 4> p) {
        auto d = detail::distances(p);
        return d[0] > 0 && d[0] == d[3] && d[4] == d[5] && d[4] == 2 * d[0];
    }
    struct Lattice {
        Int boundary, interior, area2;
    };
    // 简单格点多边形用 gcd 边点数与 Pick 定理计数；不支持自交/洞，零面积抛异常。
    Lattice lattice(const std::vector<IP> &p) {
        Int s = 0, B = 0;
        for (IP a : p)
            detail::valid(a);
        for (int i = 0; i < int(p.size()); ++i) {
            IP a = p[i], b = p[(i + 1) % p.size()];
            s += Int(a.x) * b.y - Int(a.y) * b.x;
            B += std::gcd(b.x - a.x, b.y - a.y);
        }
        if (s < 0)
            s = -s;
        Int v = s - B + 2;
        if (s == 0 || v < 0 || v % 2)
            throw std::invalid_argument("not a nondegenerate simple lattice polygon");
        return {B, v / 2, s};
    }
    // 按边界顺序的简单多边形是否凸，允许共线中间点；全共线/少于三点返回 false，输入须简单。
    bool isConvex(const std::vector<IP> &p) {
        for (IP a : p)
            detail::valid(a);
        if (p.size() < 3)
            return false;
        int sign = 0;
        for (int i = 0; i < int(p.size()); ++i) {
            IP a = p[i], b = p[(i + 1) % p.size()], c = p[(i + 2) % p.size()];
            Int v = Int(b.x - a.x) * (c.y - b.y) - Int(b.y - a.y) * (c.x - b.x);
            if (v) {
                int s = v > 0 ? 1 : -1;
                if (sign && sign != s)
                    return false;
                sign = s;
            }
        }
        return sign != 0;
    }
    // 两点是否在非零方向直线 a-b 的严格同侧；线上的点返回 false。
    bool sameSide(P p, P q, P a, P b, Real eps = 1e-12L) {
        if (square(b - a) == 0)
            throw std::invalid_argument("zero line direction");
        Real x = cross(b - a, p - a), y = cross(b - a, q - a);
        return (x > eps && y > eps) || (x < -eps && y < -eps);
    }
    // 两直线方向是否平行；含重合，零方向抛异常；eps 为相对角度阈值。
    bool parallel(P a, P b, P c, P d, Real eps = 1e-12L) {
        P u = b - a, v = d - c;
        if (square(u) == 0 || square(v) == 0)
            throw std::invalid_argument("zero line direction");
        return std::abs(cross(u, v)) <= eps * std::sqrt(square(u) * square(v));
    }
    // 两直线方向是否垂直，eps 为相对角度阈值。
    bool perpendicular(P a, P b, P c, P d, Real eps = 1e-12L) {
        P u = b - a, v = d - c;
        if (square(u) == 0 || square(v) == 0)
            throw std::invalid_argument("zero line direction");
        return std::abs(dot(u, v)) <= eps * std::sqrt(square(u) * square(v));
    }
    // 非零线段 a-b 的中垂线用 (原点,方向) 表示；调用 Line<P坐标> 时用 origin,origin+direction。
    std::pair<P, P> bisector(P a, P b) {
        if (square(b - a) == 0)
            throw std::invalid_argument("zero segment");
        P d = b - a;
        return {(a + b) / Real(2), P{-d.y, d.x}};
    }

}
```
