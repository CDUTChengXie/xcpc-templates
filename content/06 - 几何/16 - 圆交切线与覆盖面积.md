# 圆交切线与覆盖面积

坐标/半径有限，r>=0；多边形为无洞简单多边形，顶点沿边界，可 CW/CCW。 long double；eps 为长度/参数绝对容差，近相切/近重合/大坐标按尺度调整。

commonTangents 的 nullopt=重合圆有无穷公切线、空 vector=无有限公切线；coverage 返回 ans[0..n]，ans[k] 为至少 k 个圆覆盖面积（非恰 k 次）；反演以正半径圆为基准。

交/切 O(1)；polygonIntersectionArea O(m)；coverage O(n² log n) 时间、O(n) 工作空间。

#ifndef XCPC_TEMPLATE_POINT

template <class T> struct Point {
    T x;
    T y;

    // 构造点/向量，省略坐标得到原点。
    Point(const T &x_ = 0, const T &y_ = 0) : x(x_), y(y_) {}

    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator+=(const Point &p) & {
        x += p.x;
        y += p.y;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator-=(const Point &p) & {
        x -= p.x;
        y -= p.y;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator*=(const T &v) & {
        x *= v;
        y *= v;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator/=(const T &v) & {
        x /= v;
        y /= v;
        return *this;
    }
    // 返回反向向量。
    Point operator-() const {
        return Point(-x, -y);
    }
    // 返回两向量之和。
    friend Point operator+(Point a, const Point &b) {
        return a += b;
    }
    // 返回两向量之差。
    friend Point operator-(Point a, const Point &b) {
        return a -= b;
    }
    // 返回标量倍向量。
    friend Point operator*(Point a, const T &b) {
        return a *= b;
    }
    // 返回向量除以非零标量。
    friend Point operator/(Point a, const T &b) {
        return a /= b;
    }
    // 返回标量左乘向量。
    friend Point operator*(const T &a, Point b) {
        return b *= a;
    }
    // 返回坐标严格相等，不含 eps。
    friend bool operator==(const Point &a, const Point &b) {
        return a.x == b.x && a.y == b.y;
    }
    // 返回字典序比较（先 x 后 y），不含 eps。
    friend bool operator<(const Point &a, const Point &b) {
        return a.x < b.x || (a.x == b.x && a.y < b.y);
    }
    // 读入 x、y，返回输入流。
    friend std::istream &operator>>(std::istream &is, Point &p) {
        return is >> p.x >> p.y;
    }
    // 输出 x、y，返回输出流。
    friend std::ostream &operator<<(std::ostream &os, const Point &p) {
        return os << p.x << " " << p.y;
    }
};

template <class T> struct Line {
    Point<T> a;
    Point<T> b;

    // 构造有向线段端点；直线运算须保证两端点不同。
    Line(const Point<T> &a_ = Point<T>(), const Point<T> &b_ = Point<T>()) : a(a_), b(b_) {}
};

template <class T>
// 返回两向量点积。
T dot(const Point<T> &a, const Point<T> &b) {
    return a.x * b.x + a.y * b.y;
}

template <class T>
// 返回两向量二维叉积（a 到 b 逆时针为正）。
T cross(const Point<T> &a, const Point<T> &b) {
    return a.x * b.y - a.y * b.x;
}

template <class T>
// 返回向量长度平方。
T square(const Point<T> &p) {
    return dot(p, p);
}

template <class T>
// 返回向量长度，结果为 double。
double length(const Point<T> &p) {
    return std::sqrt(square(p));
}

template <class T>
// 返回两点距离，结果为 double。
double distance(const Point<T> &a, const Point<T> &b) {
    return length(a - b);
}

template <class T>
// 返回点到直线距离；两端点须不同。
double distancePL(const Point<T> &p, const Line<T> &l) {
    return std::abs(cross(l.a - l.b, l.a - p)) / length(l.b - l.a);
}

template <class T>
// 返回点到闭线段距离；端点须不同，零长段会除零。
double distancePS(const Point<T> &p, const Line<T> &l) {
    if (dot(p - l.a, l.b - l.a) < 0) {
        return distance(p, l.a);
    }
    if (dot(p - l.b, l.a - l.b) < 0) {
        return distance(p, l.b);
    }
    return distancePL(p, l);
}

template <class T>
// 返回两非平行直线交点；整数类型用 long double 计算后截断，精度场景改用浮点坐标。
Point<T> lineIntersection(const Line<T> &l1, const Line<T> &l2) {
    long double c1 = (long double)cross(l2.b - l2.a, l1.a - l2.a);
    long double c2 = (long double)cross(l2.b - l2.a, l1.a - l1.b);
    long double t = c1 / c2;
    return Point<T>(l1.a.x + (long double)(l1.b.x - l1.a.x) * t,
                    l1.a.y + (long double)(l1.b.y - l1.a.y) * t);
}

template <class T>
// 判断点是否在闭线段上（含端点；浮点需自行加 eps）。
bool pointOnSegment(const Point<T> &p, const Line<T> &l) {
    return cross(p - l.a, l.b - l.a) == 0 && std::min(l.a.x, l.b.x) <= p.x &&
           p.x <= std::max(l.a.x, l.b.x) && std::min(l.a.y, l.b.y) <= p.y &&
           p.y <= std::max(l.a.y, l.b.y);
}

template <class T>
// 判断点是否在简单多边形闭区域内（边界算内，与绕向无关，O(n)）。
bool pointInPolygon(const Point<T> &a, const std::vector<Point<T>> &p) {
    int n = p.size();
    for (int i = 0; i < n; i++) {
        if (pointOnSegment(a, Line(p[i], p[(i + 1) % n]))) {
            return true;
        }
    }
    int t = 0;
    for (int i = 0; i < n; i++) {
        auto u = p[i];
        auto v = p[(i + 1) % n];
        if (u.x < a.x && v.x >= a.x && cross(v - u, a - u) > 0) {
            t ^= 1;
        }
        if (u.x >= a.x && v.x < a.x && cross(u - v, a - v) > 0) {
            t ^= 1;
        }
    }
    return t == 1;
}

template <class T>
// 返回两闭线段相交分类：0 不相交、1 严格相交、2 重叠（返回两端点）、3 端点相交。
std::tuple<int, Point<T>, Point<T>> segmentIntersection(const Line<T> &l1, const Line<T> &l2) {
    if (l1.a == l1.b)
        return {pointOnSegment(l1.a, l2) ? 3 : 0, l1.a, l1.a};
    if (l2.a == l2.b)
        return {pointOnSegment(l2.a, l1) ? 3 : 0, l2.a, l2.a};
    if (std::max(l1.a.x, l1.b.x) < std::min(l2.a.x, l2.b.x) ||
        std::min(l1.a.x, l1.b.x) > std::max(l2.a.x, l2.b.x) ||
        std::max(l1.a.y, l1.b.y) < std::min(l2.a.y, l2.b.y) ||
        std::min(l1.a.y, l1.b.y) > std::max(l2.a.y, l2.b.y)) {
        return {0, Point<T>(), Point<T>()};
    }
    if (cross(l1.b - l1.a, l2.b - l2.a) == 0) {
        if (cross(l1.b - l1.a, l2.a - l1.a) != 0) {
            return {0, Point<T>(), Point<T>()};
        }
        Point<T> p1(std::max(std::min(l1.a.x, l1.b.x), std::min(l2.a.x, l2.b.x)),
                    std::max(std::min(l1.a.y, l1.b.y), std::min(l2.a.y, l2.b.y)));
        Point<T> p2(std::min(std::max(l1.a.x, l1.b.x), std::max(l2.a.x, l2.b.x)),
                    std::min(std::max(l1.a.y, l1.b.y), std::max(l2.a.y, l2.b.y)));
        if (!pointOnSegment(p1, l1)) {
            std::swap(p1.y, p2.y);
        }
        if (p1 == p2) {
            return {3, p1, p2};
        }
        return {2, p1, p2};
    }
    auto cp1 = cross(l2.a - l1.a, l2.b - l1.a);
    auto cp2 = cross(l2.a - l1.b, l2.b - l1.b);
    auto cp3 = cross(l1.a - l2.a, l1.b - l2.a);
    auto cp4 = cross(l1.a - l2.b, l1.b - l2.b);
    if ((cp1 > 0 && cp2 > 0) || (cp1 < 0 && cp2 < 0) || (cp3 > 0 && cp4 > 0) ||
        (cp3 < 0 && cp4 < 0)) {
        return {0, Point<T>(), Point<T>()};
    }
    Point p = lineIntersection(l1, l2);
    if (cp1 != 0 && cp2 != 0 && cp3 != 0 && cp4 != 0) {
        return {1, p, p};
    }
    return {3, p, p};
}

template <class T>
// 返回两闭线段最短距离，相交或重叠为 0。
double distanceSS(const Line<T> &l1, const Line<T> &l2) {
    if (std::get<0>(segmentIntersection(l1, l2)) != 0) {
        return 0.0;
    }
    return std::min(
        {distancePS(l1.a, l2), distancePS(l1.b, l2), distancePS(l2.a, l1), distancePS(l2.b, l1)});
}

template <class T>
// 返回简单多边形两倍无向面积，顶点须按边界顺序；空集或单点返回 0。
T polygonArea2(const std::vector<Point<T>> &p) {
    T s = 0;
    int n = p.size();
    for (int i = 0; i < n; i++) {
        s += cross(p[i], p[(i + 1) % n]);
    }
    return std::abs(s);
}

namespace GeoCircle {
    using Real = long double;
    using P = Point<Real>;
    const Real PI = std::acos(Real(-1));
    struct Circle {
        P c;
        Real r;
    };
    namespace detail {
        void valid(Circle c) {
            if (c.r < 0 || !std::isfinite(c.r) || !std::isfinite(c.c.x) || !std::isfinite(c.c.y))
                throw std::invalid_argument("invalid circle");
        }
        Real norm(P p) {
            return std::sqrt(square(p));
        }
        bool same(Circle a, Circle b, Real eps) {
            return norm(a.c - b.c) <= eps && std::abs(a.r - b.r) <= eps;
        }
    }

    // 返回点到圆周的最近点；点在圆心时任选 +x 方向，零半径返回圆心。
    P nearest(Circle c, P p, Real eps = 1e-12L) {
        detail::valid(c);
        P d = p - c.c;
        Real len = detail::norm(d);
        return len <= eps ? c.c + P{c.r, 0} : c.c + d * (c.r / len);
    }
    // 由圆心角 rad（弧度）返回圆周点。
    P pointAt(Circle c, Real rad) {
        detail::valid(c);
        return c.c + P{std::cos(rad) * c.r, std::sin(rad) * c.r};
    }

    // 求圆与直线交点；segment=true 时限制到 a-b，交点按 a->b 排序；零长按圆上点处理。
    std::vector<P> lineIntersections(Circle c, P a, P b, bool segment = false, Real eps = 1e-12L) {
        detail::valid(c);
        P d = b - a;
        Real len2 = square(d);
        if (len2 <= eps * eps)
            return std::abs(detail::norm(a - c.c) - c.r) <= eps ? std::vector<P>{a}
                                                                : std::vector<P>{};
        Real t = dot(c.c - a, d) / len2, h2 = c.r * c.r - square(a + d * t - c.c);
        Real tol = eps * std::max(Real(1), c.r);
        if (h2 < -tol)
            return {};
        Real dt = std::sqrt(std::max(Real(0), h2) / len2);
        std::vector<P> out;
        for (Real u : std::array<Real, 2>{t - dt, t + dt})
            if (!segment || (u >= -eps && u <= 1 + eps)) {
                P p = a + d * (segment ? std::clamp(u, Real(0), Real(1)) : u);
                if (out.empty() || detail::norm(out.back() - p) > eps)
                    out.push_back(p);
            }
        return out;
    }
    // 返回两圆交面积；包含、重合、外切、分离及零半径均支持。
    Real overlapArea(Circle a, Circle b, Real eps = 1e-12L) {
        detail::valid(a);
        detail::valid(b);
        if (a.r > b.r)
            std::swap(a, b);
        Real d = detail::norm(b.c - a.c);
        if (d <= b.r - a.r + eps)
            return PI * a.r * a.r;
        if (d >= a.r + b.r - eps || a.r == 0)
            return 0;
        Real x = std::acos(
            std::clamp((d * d + a.r * a.r - b.r * b.r) / (2 * d * a.r), Real(-1), Real(1)));
        Real y = std::acos(
            std::clamp((d * d + b.r * b.r - a.r * a.r) / (2 * d * b.r), Real(-1), Real(1)));
        return a.r * a.r * (x - std::sin(2 * x) / 2) + b.r * b.r * (y - std::sin(2 * y) / 2);
    }
    // 返回从点 p 向圆作切线的切点；圆内为空，圆上一个，圆外两个。
    std::vector<P> tangentFrom(Circle c, P p, Real eps = 1e-12L) {
        detail::valid(c);
        P d = p - c.c;
        Real D = detail::norm(d);
        if (D < c.r - eps)
            return {};
        if (c.r <= eps)
            return {c.c};
        if (std::abs(D - c.r) <= eps)
            return {c.c + d * (c.r / D)};
        P u = d * (c.r * c.r / (D * D)), v{-d.y, d.x};
        v = v * (c.r * std::sqrt(std::max(Real(0), D * D - c.r * c.r)) / (D * D));
        return {c.c + u + v, c.c + u - v};
    }
    // 返回所有有限内/外公切线的切点对；近重合圆返回 nullopt，结果去重。
    std::optional<std::vector<std::pair<P, P>>> commonTangents(Circle a, Circle b,
                                                               Real eps = 1e-12L) {
        detail::valid(a);
        detail::valid(b);
        P d = b.c - a.c;
        Real d2 = square(d);
        if (detail::same(a, b, eps))
            return std::nullopt;
        std::vector<std::pair<P, P>> out;
        if (d2 <= eps * eps)
            return out;
        for (int side : std::array<int, 2>{1, -1}) {
            Real dr = a.r - side * b.r, h2 = d2 - dr * dr;
            if (h2 < -eps * std::max(Real(1), std::sqrt(d2)))
                continue;
            for (int s : std::array<int, 2>{1, -1}) {
                Real h = std::sqrt(std::max(Real(0), h2));
                P n{(d.x * dr - d.y * h * s) / d2, (d.y * dr + d.x * h * s) / d2};
                std::pair<P, P> t{a.c + n * a.r, b.c + n * (b.r * side)};
                bool duplicate = false;
                for (auto e : out)
                    if (detail::norm(e.first - t.first) <= eps &&
                        detail::norm(e.second - t.second) <= eps)
                        duplicate = true;
                if (!duplicate)
                    out.push_back(t);
            }
        }
        return out;
    }
    // 返回圆与简单多边形交的无符号面积，允许凹多边形与连续重复点。
    Real polygonIntersectionArea(Circle c, const std::vector<P> &p, Real eps = 1e-12L) {
        detail::valid(c);
        Real sum = 0;
        int n = int(p.size());
        if (c.r == 0)
            return 0;
        for (int i = 0; i < n; ++i) {
            P a = p[i] - c.c, b = p[(i + 1) % n] - c.c, d = b - a;
            Real len2 = square(d);
            if (len2 <= eps * eps)
                continue;
            std::vector<Real> ts{0, 1};
            Real t = -dot(a, d) / len2, h2 = c.r * c.r - square(a + d * t);
            if (h2 > 0) {
                Real dt = std::sqrt(h2 / len2);
                for (Real u : std::array<Real, 2>{t - dt, t + dt})
                    if (u > 0 && u < 1)
                        ts.push_back(u);
            }
            std::sort(ts.begin(), ts.end());
            for (int j = 1; j < int(ts.size()); ++j) {
                P u = a + d * ts[j - 1], v = a + d * ts[j], mid = a + d * ((ts[j - 1] + ts[j]) / 2);
                if (square(mid) <= c.r * c.r)
                    sum += cross(u, v);
                else
                    sum += c.r * c.r * std::atan2(cross(u, v), dot(u, v));
            }
        }
        return std::abs(sum) / 2;
    }
    // 返回至少 k 重圆覆盖面积，ans[0]=0；重复圆保留覆盖次数，零半径圆不贡献面积。
    std::vector<Real> coverage(const std::vector<Circle> &circles, Real eps = 1e-12L) {
        int n = int(circles.size());
        for (auto c : circles)
            detail::valid(c);
        std::vector<Real> ans(n + 1);
        for (int i = 0; i < n; ++i) {
            auto a = circles[i];
            if (a.r == 0)
                continue;
            int mult = 1, base = 0;
            bool skip = false;
            std::vector<std::pair<Real, int>> events{{0, 0}, {2 * PI, 0}};
            auto interval = [&](Real l, Real r) {
                events.push_back({l, 1});
                events.push_back({r, -1});
            };
            for (int j = 0; j < n; ++j)
                if (i != j) {
                    auto b = circles[j];
                    Real d = detail::norm(b.c - a.c);
                    if (detail::same(a, b, eps)) {
                        if (j < i)
                            skip = true;
                        ++mult;
                        continue;
                    }
                    if (d + a.r <= b.r + eps) {
                        ++base;
                        continue;
                    }
                    if (d + b.r <= a.r + eps || d >= a.r + b.r - eps || d <= eps)
                        continue;
                    Real theta = std::atan2(b.c.y - a.c.y, b.c.x - a.c.x);
                    if (theta < 0)
                        theta += 2 * PI;
                    Real alpha = std::acos(std::clamp(
                             (a.r * a.r + d * d - b.r * b.r) / (2 * a.r * d), Real(-1), Real(1))),
                         l = theta - alpha, r = theta + alpha;
                    if (l < 0) {
                        interval(0, r);
                        interval(l + 2 * PI, 2 * PI);
                    } else if (r > 2 * PI) {
                        interval(l, 2 * PI);
                        interval(0, r - 2 * PI);
                    } else
                        interval(l, r);
                }
            if (skip)
                continue;
            std::sort(events.begin(), events.end());
            int count = base;
            for (int j = 0; j + 1 < int(events.size()); ++j) {
                count += events[j].second;
                Real l = events[j].first, r = events[j + 1].first;
                if (r <= l)
                    continue;
                Real integral = (a.r * a.r * (r - l) + a.c.x * a.r * (std::sin(r) - std::sin(l)) -
                                 a.c.y * a.r * (std::cos(r) - std::cos(l))) /
                                2;
                for (int k = count + 1; k <= count + mult; ++k)
                    ans[k] += integral;
            }
        }
        for (int k = 1; k <= n; ++k)
            if (ans[k] < 0 && ans[k] > -eps)
                ans[k] = 0;
        return ans;
    }
    struct InversionLine {
        P p, d;
    }; // 无限直线 p+t*d，d 非零；与二维 Line 的端点表示不同
    using InverseImage = std::variant<Circle, InversionLine>;
    namespace detail {
        void inversion(Circle base) {
            valid(base);
            if (base.r == 0)
                throw std::invalid_argument("inversion radius must be positive");
        }
    }
    // 以 base 圆反演点 p；反演中心（距离<=eps）无有限像，抛 invalid_argument。
    P inversePoint(Circle base, P p, Real eps = 1e-12L) {
        detail::inversion(base);
        if (!std::isfinite(p.x) || !std::isfinite(p.y))
            throw std::invalid_argument("nonfinite inversion point");
        P d = p - base.c;
        Real len2 = square(d);
        if (len2 <= eps * eps)
            throw std::invalid_argument("inversion center has no finite image");
        return base.c + d * (base.r * base.r / len2);
    }
    // 反演无限直线；过中心返回直线，否则返回过中心圆；按几何闭包表示，不含无穷点。
    InverseImage inverse(Circle base, InversionLine line, Real eps = 1e-12L) {
        detail::inversion(base);
        if (!std::isfinite(line.p.x) || !std::isfinite(line.p.y) || !std::isfinite(line.d.x) ||
            !std::isfinite(line.d.y) || square(line.d) == 0)
            throw std::invalid_argument("invalid inversion line");
        P u = line.p - base.c, foot = u - line.d * (dot(u, line.d) / square(line.d));
        Real len = detail::norm(foot);
        if (len <= eps)
            return InversionLine{base.c, line.d};
        Real k = base.r * base.r / (2 * len * len);
        return Circle{base.c + foot * k, base.r * base.r / (2 * len)};
    }
    // 反演圆周（可为点圆）；过中心圆变直线，其余变圆，中心点圆无有限像并抛异常。
    InverseImage inverse(Circle base, Circle circle, Real eps = 1e-12L) {
        detail::inversion(base);
        detail::valid(circle);
        P d = circle.c - base.c;
        Real len2 = square(d), delta = len2 - circle.r * circle.r;
        if (circle.r == 0) {
            return Circle{inversePoint(base, circle.c, eps), 0};
        }
        Real tol = eps * std::max(Real(1), detail::norm(d) + circle.r);
        if (std::abs(delta) <= tol) {
            if (len2 == 0)
                throw std::invalid_argument("unresolved near-center circle");
            return InversionLine{base.c + d * (base.r * base.r / (2 * len2)), P{-d.y, d.x}};
        }
        Real k = base.r * base.r / delta;
        return Circle{base.c + d * k, std::abs(k) * circle.r};
    }

}
