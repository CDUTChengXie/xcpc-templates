# 凸包对数支撑点与双切线（整数）

输入为简单严格凸 CCW 多边形，起点任意、不重复首尾；也支持空/点/段，整数坐标与方向分量绝对值<=1e9。 浮点版限制同上，用 long double 原始符号比较且二分不加 eps，非精确判断函数，须保证舍入后仍严格凸、查询转向分区单调。

下标始终对应构造时顶点 [0,n)，不重新排序；同极值边任选一端；整数用 __int128、浮点用 long double，均无 eps。

外点切线返回 {left,right}（指向 left 时凸包在闭左侧、指向 right 时在闭右侧），内/边界返回空；平行双切线用 {max,min} 两法向极值顶点，零法向 support 任选 0，零直线方向抛异常。

构造验证 O(n)；所有查询 O(log n)，点/段 O(1)，空间 O(n) 保存点；没有查询时全扫描兜底。

```cpp
#ifndef XCPC_TEMPLATE_POINT
#define XCPC_TEMPLATE_POINT

template <class T> struct Point {
    T x;
    T y;

    // 构造点/向量，省略坐标得到原点。
    Point(const T &x_ = 0, const T &y_ = 0) : x(x_), y(y_) {}

    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator+=(const Point &p) & {
        x += p.x;
        y += p.y;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator-=(const Point &p) & {
        x -= p.x;
        y -= p.y;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator*=(const T &v) & {
        x *= v;
        y *= v;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator/=(const T &v) & {
        x /= v;
        y /= v;
        return *this;
    }
    // 返回反向向量。
    Point operator-() const {
        return Point(-x, -y);
    }
    // 返回两向量之和。
    friend Point operator+(Point a, const Point &b) {
        return a += b;
    }
    // 返回两向量之差。
    friend Point operator-(Point a, const Point &b) {
        return a -= b;
    }
    // 返回标量倍向量。
    friend Point operator*(Point a, const T &b) {
        return a *= b;
    }
    // 返回向量除以非零标量。
    friend Point operator/(Point a, const T &b) {
        return a /= b;
    }
    // 返回标量左乘向量。
    friend Point operator*(const T &a, Point b) {
        return b *= a;
    }
    // 返回坐标严格相等，不含 eps。
    friend bool operator==(const Point &a, const Point &b) {
        return a.x == b.x && a.y == b.y;
    }
    // 返回字典序比较（先 x 后 y），不含 eps。
    friend bool operator<(const Point &a, const Point &b) {
        return a.x < b.x || (a.x == b.x && a.y < b.y);
    }
    // 读入 x、y，返回输入流。
    friend std::istream &operator>>(std::istream &is, Point &p) {
        return is >> p.x >> p.y;
    }
    // 输出 x、y，返回输出流。
    friend std::ostream &operator<<(std::ostream &os, const Point &p) {
        return os << p.x << " " << p.y;
    }
};

template <class T> struct Line {
    Point<T> a;
    Point<T> b;

    // 构造有向线段端点；直线运算须保证两端点不同。
    Line(const Point<T> &a_ = Point<T>(), const Point<T> &b_ = Point<T>()) : a(a_), b(b_) {}
};

template <class T>
// 返回两向量点积。
T dot(const Point<T> &a, const Point<T> &b) {
    return a.x * b.x + a.y * b.y;
}

template <class T>
// 返回两向量二维叉积（a 到 b 逆时针为正）。
T cross(const Point<T> &a, const Point<T> &b) {
    return a.x * b.y - a.y * b.x;
}

template <class T>
// 返回向量长度平方。
T square(const Point<T> &p) {
    return dot(p, p);
}

template <class T>
// 返回向量长度，结果为 double。
double length(const Point<T> &p) {
    return std::sqrt(square(p));
}

template <class T>
// 返回两点距离，结果为 double。
double distance(const Point<T> &a, const Point<T> &b) {
    return length(a - b);
}

template <class T>
// 返回点到直线距离；两端点须不同。
double distancePL(const Point<T> &p, const Line<T> &l) {
    return std::abs(cross(l.a - l.b, l.a - p)) / length(l.b - l.a);
}

template <class T>
// 返回点到闭线段距离；端点须不同，零长段会除零。
double distancePS(const Point<T> &p, const Line<T> &l) {
    if (dot(p - l.a, l.b - l.a) < 0) {
        return distance(p, l.a);
    }
    if (dot(p - l.b, l.a - l.b) < 0) {
        return distance(p, l.b);
    }
    return distancePL(p, l);
}

template <class T>
// 返回两非平行直线交点；整数类型用 long double 计算后截断，精度场景改用浮点坐标。
Point<T> lineIntersection(const Line<T> &l1, const Line<T> &l2) {
    long double c1 = (long double)cross(l2.b - l2.a, l1.a - l2.a);
    long double c2 = (long double)cross(l2.b - l2.a, l1.a - l1.b);
    long double t = c1 / c2;
    return Point<T>(l1.a.x + (long double)(l1.b.x - l1.a.x) * t,
                    l1.a.y + (long double)(l1.b.y - l1.a.y) * t);
}

template <class T>
// 判断点是否在闭线段上（含端点；浮点需自行加 eps）。
bool pointOnSegment(const Point<T> &p, const Line<T> &l) {
    return cross(p - l.a, l.b - l.a) == 0 && std::min(l.a.x, l.b.x) <= p.x &&
           p.x <= std::max(l.a.x, l.b.x) && std::min(l.a.y, l.b.y) <= p.y &&
           p.y <= std::max(l.a.y, l.b.y);
}

template <class T>
// 判断点是否在简单多边形闭区域内（边界算内，与绕向无关，O(n)）。
bool pointInPolygon(const Point<T> &a, const std::vector<Point<T>> &p) {
    int n = p.size();
    for (int i = 0; i < n; i++) {
        if (pointOnSegment(a, Line(p[i], p[(i + 1) % n]))) {
            return true;
        }
    }
    int t = 0;
    for (int i = 0; i < n; i++) {
        auto u = p[i];
        auto v = p[(i + 1) % n];
        if (u.x < a.x && v.x >= a.x && cross(v - u, a - u) > 0) {
            t ^= 1;
        }
        if (u.x >= a.x && v.x < a.x && cross(u - v, a - v) > 0) {
            t ^= 1;
        }
    }
    return t == 1;
}

template <class T>
// 返回两闭线段相交分类：0 不相交、1 严格相交、2 重叠（返回两端点）、3 端点相交。
std::tuple<int, Point<T>, Point<T>> segmentIntersection(const Line<T> &l1, const Line<T> &l2) {
    if (l1.a == l1.b)
        return {pointOnSegment(l1.a, l2) ? 3 : 0, l1.a, l1.a};
    if (l2.a == l2.b)
        return {pointOnSegment(l2.a, l1) ? 3 : 0, l2.a, l2.a};
    if (std::max(l1.a.x, l1.b.x) < std::min(l2.a.x, l2.b.x) ||
        std::min(l1.a.x, l1.b.x) > std::max(l2.a.x, l2.b.x) ||
        std::max(l1.a.y, l1.b.y) < std::min(l2.a.y, l2.b.y) ||
        std::min(l1.a.y, l1.b.y) > std::max(l2.a.y, l2.b.y)) {
        return {0, Point<T>(), Point<T>()};
    }
    if (cross(l1.b - l1.a, l2.b - l2.a) == 0) {
        if (cross(l1.b - l1.a, l2.a - l1.a) != 0) {
            return {0, Point<T>(), Point<T>()};
        }
        Point<T> p1(std::max(std::min(l1.a.x, l1.b.x), std::min(l2.a.x, l2.b.x)),
                    std::max(std::min(l1.a.y, l1.b.y), std::min(l2.a.y, l2.b.y)));
        Point<T> p2(std::min(std::max(l1.a.x, l1.b.x), std::max(l2.a.x, l2.b.x)),
                    std::min(std::max(l1.a.y, l1.b.y), std::max(l2.a.y, l2.b.y)));
        if (!pointOnSegment(p1, l1)) {
            std::swap(p1.y, p2.y);
        }
        if (p1 == p2) {
            return {3, p1, p2};
        }
        return {2, p1, p2};
    }
    auto cp1 = cross(l2.a - l1.a, l2.b - l1.a);
    auto cp2 = cross(l2.a - l1.b, l2.b - l1.b);
    auto cp3 = cross(l1.a - l2.a, l1.b - l2.a);
    auto cp4 = cross(l1.a - l2.b, l1.b - l2.b);
    if ((cp1 > 0 && cp2 > 0) || (cp1 < 0 && cp2 < 0) || (cp3 > 0 && cp4 > 0) ||
        (cp3 < 0 && cp4 < 0)) {
        return {0, Point<T>(), Point<T>()};
    }
    Point p = lineIntersection(l1, l2);
    if (cp1 != 0 && cp2 != 0 && cp3 != 0 && cp4 != 0) {
        return {1, p, p};
    }
    return {3, p, p};
}

template <class T>
// 返回两闭线段最短距离，相交或重叠为 0。
double distanceSS(const Line<T> &l1, const Line<T> &l2) {
    if (std::get<0>(segmentIntersection(l1, l2)) != 0) {
        return 0.0;
    }
    return std::min(
        {distancePS(l1.a, l2), distancePS(l1.b, l2), distancePS(l2.a, l1), distancePS(l2.b, l1)});
}

template <class T>
// 返回简单多边形两倍无向面积，顶点须按边界顺序；空集或单点返回 0。
T polygonArea2(const std::vector<Point<T>> &p) {
    T s = 0;
    int n = p.size();
    for (int i = 0; i < n; i++) {
        s += cross(p[i], p[(i + 1) % n]);
    }
    return std::abs(s);
}

namespace ConvexQuery {
    using P = Point<long long>;
    using Int = __int128;
    using FloatingP = Point<long double>;
    template <class T>
    using Wide = std::conditional_t<std::is_same_v<T, long long>, Int, long double>;
    namespace detail {
        template <class T> Wide<T> det(Point<T> a, Point<T> b) {
            return Wide<T>(a.x) * b.y - Wide<T>(a.y) * b.x;
        }
        template <class T> Wide<T> turn(Point<T> a, Point<T> b, Point<T> c) {
            return det(b - a, c - a);
        }
        template <class T> void valid(Point<T> p) {
            if (!std::isfinite(static_cast<long double>(p.x)) ||
                !std::isfinite(static_cast<long double>(p.y)) || p.x < -1000000000LL ||
                p.x > 1000000000LL || p.y < -1000000000LL || p.y > 1000000000LL)
                throw std::invalid_argument("ConvexQuery: coordinate/direction exceeds 1e9");
        }
        template <class T> bool on(Point<T> q, Point<T> a, Point<T> b) {
            return turn(a, b, q) == 0 && std::min(a.x, b.x) <= q.x && q.x <= std::max(a.x, b.x) &&
                   std::min(a.y, b.y) <= q.y && q.y <= std::max(a.y, b.y);
        }
    }
    template <class T> class BasicHull {
        static_assert(std::is_same_v<T, long long> || std::is_same_v<T, long double>);
        using P = Point<T>;
        using Value = Wide<T>;
        std::vector<P> p;
        // ponytail: 浮点原始符号只保证分区仍单调的输入；近退化需升级自适应/精确符号谓词。
        // dir 为常量或 ±(u-q)，按转向分区二分定位极点；不能传任意非单调函数。
        template <class F> int extreme(F dir) const {
            int n = int(p.size());
            auto check = [&](int i) { return detail::det(dir(p[i]), p[(i + 1) % n] - p[i]) >= 0; };
            bool first = check(0);
            P d = dir(p[0]);
            if (!first && check(n - 1))
                return 0;
            auto before = [&](int i) {
                if (i == 0)
                    return true;
                bool current = check(i);
                Value side = detail::det(d, p[i] - p[0]);
                if (i == 1 && current == first && side == 0)
                    return true;
                return current != (current == first && side <= 0);
            };
            int l = 0, r = n;
            while (l < r) {
                int m = (l + r) / 2;
                if (before(m))
                    l = m + 1;
                else
                    r = m;
            }
            return l % n;
        }
        bool contains(P q) const {
            int n = int(p.size());
            if (n == 0)
                return false;
            if (n == 1)
                return q == p[0];
            if (n == 2)
                return detail::on(q, p[0], p[1]);
            Value a = detail::turn(p[0], p[1], q), b = detail::turn(p[0], p.back(), q);
            if (a < 0 || b > 0)
                return false;
            if (a == 0)
                return detail::on(q, p[0], p[1]);
            if (b == 0)
                return detail::on(q, p[0], p.back());
            int l = 1, r = n - 1;
            while (r - l > 1) {
                int m = (l + r) / 2;
                if (detail::turn(p[0], p[m], q) >= 0)
                    l = m;
                else
                    r = m;
            }
            return detail::turn(p[l], p[r], q) >= 0;
        }

      public:
        // 保存顶点原编号，验证严格 CCW 转向；输入须简单，空/点/段可用，重复顶点抛异常。
        explicit BasicHull(std::vector<P> polygon) : p(std::move(polygon)) {
            for (P q : p)
                detail::valid(q);
            int n = int(p.size());
            if (n == 2 && p[0] == p[1])
                throw std::invalid_argument("duplicate segment endpoints");
            if (n >= 3)
                for (int i = 0; i < n; ++i)
                    if (detail::turn(p[i], p[(i + 1) % n], p[(i + 2) % n]) <= 0)
                        throw std::invalid_argument("not strict CCW convex");
        }
        // 最大化 direction·p[i]，O(log n)；方向为零取0，空凸包返回空；极值边任选端点。
        std::optional<int> support(P direction) const {
            detail::valid(direction);
            int n = int(p.size());
            if (!n)
                return std::nullopt;
            if (direction == P{})
                return 0;
            if (n <= 2) {
                if (n == 1)
                    return 0;
                return Value(direction.x) * (p[1].x - p[0].x) +
                                   Value(direction.y) * (p[1].y - p[0].y) >
                               0
                           ? 1
                           : 0;
            }
            P d{direction.y, -direction.x};
            return extreme([&](P) { return d; });
        }
        // 外点双切点，按闭左/闭右顺序返回；内部/边界返回空，支持点和线段退化。
        std::optional<std::array<int, 2>> tangentsFrom(P q) const {
            detail::valid(q);
            int n = int(p.size());
            if (!n || contains(q))
                return std::nullopt;
            if (n == 1)
                return std::array<int, 2>{0, 0};
            if (n == 2) {
                Value side = detail::turn(q, p[0], p[1]);
                return side >= 0 ? std::array<int, 2>{0, 1} : std::array<int, 2>{1, 0};
            }
            return std::array<int, 2>{extreme([&](P u) { return q - u; }),
                                      extreme([&](P u) { return u - q; })};
        }
        // 返回两条平行 direction 的支撑线顶点，顺序为法向 (-dy,dx) 的最大/最小；零方向抛异常。
        std::optional<std::array<int, 2>> parallelTangents(P direction) const {
            detail::valid(direction);
            if (direction == P{})
                throw std::invalid_argument("zero tangent direction");
            P normal{-direction.y, direction.x};
            auto a = support(normal), b = support(-normal);
            if (!a)
                return std::nullopt;
            return std::array<int, 2>{*a, *b};
        }
    };
    using Hull = BasicHull<long long>;
    using FloatingHull = BasicHull<long double>;
}
```
