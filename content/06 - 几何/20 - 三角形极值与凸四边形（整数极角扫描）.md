# 三角形极值与凸四边形（整数极角扫描）

整数坐标绝对值<=1e9；maximumQuadrilateral 的多边形须严格凸 CCW，无重复首尾。 `Point<long long>`；返回 __int128 的两倍面积（非面积），无 eps；minmaxWithIds 另返回原输入三编号；重复点和共线会使最小值为零。

三点不足返回 {0,0}；凸四边形不足四个点返回0；maximumQuadrilateralPoints 另处理任意点集、凹四边形与重复顶点退化。

minmax O(n² log n) 时间/O(n²) 空间；maximumQuadrilateral O(n²) 时间/O(1) 额外空间；任意点集版 O(n log n+h²) 时间/O(n) 空间。

#ifndef XCPC_TEMPLATE_POINT

template <class T> struct Point {
    T x;
    T y;

    // 构造点/向量，省略坐标得到原点。
    Point(const T &x_ = 0, const T &y_ = 0) : x(x_), y(y_) {}

    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator+=(const Point &p) & {
        x += p.x;
        y += p.y;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator-=(const Point &p) & {
        x -= p.x;
        y -= p.y;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator*=(const T &v) & {
        x *= v;
        y *= v;
        return *this;
    }
    // 原地向量运算；标量除数须非零，返回本对象引用。
    Point &operator/=(const T &v) & {
        x /= v;
        y /= v;
        return *this;
    }
    // 返回反向向量。
    Point operator-() const {
        return Point(-x, -y);
    }
    // 返回两向量之和。
    friend Point operator+(Point a, const Point &b) {
        return a += b;
    }
    // 返回两向量之差。
    friend Point operator-(Point a, const Point &b) {
        return a -= b;
    }
    // 返回标量倍向量。
    friend Point operator*(Point a, const T &b) {
        return a *= b;
    }
    // 返回向量除以非零标量。
    friend Point operator/(Point a, const T &b) {
        return a /= b;
    }
    // 返回标量左乘向量。
    friend Point operator*(const T &a, Point b) {
        return b *= a;
    }
    // 返回坐标严格相等，不含 eps。
    friend bool operator==(const Point &a, const Point &b) {
        return a.x == b.x && a.y == b.y;
    }
    // 返回字典序比较（先 x 后 y），不含 eps。
    friend bool operator<(const Point &a, const Point &b) {
        return a.x < b.x || (a.x == b.x && a.y < b.y);
    }
    // 读入 x、y，返回输入流。
    friend std::istream &operator>>(std::istream &is, Point &p) {
        return is >> p.x >> p.y;
    }
    // 输出 x、y，返回输出流。
    friend std::ostream &operator<<(std::ostream &os, const Point &p) {
        return os << p.x << " " << p.y;
    }
};

template <class T> struct Line {
    Point<T> a;
    Point<T> b;

    // 构造有向线段端点；直线运算须保证两端点不同。
    Line(const Point<T> &a_ = Point<T>(), const Point<T> &b_ = Point<T>()) : a(a_), b(b_) {}
};

template <class T>
// 返回两向量点积。
T dot(const Point<T> &a, const Point<T> &b) {
    return a.x * b.x + a.y * b.y;
}

template <class T>
// 返回两向量二维叉积（a 到 b 逆时针为正）。
T cross(const Point<T> &a, const Point<T> &b) {
    return a.x * b.y - a.y * b.x;
}

template <class T>
// 返回向量长度平方。
T square(const Point<T> &p) {
    return dot(p, p);
}

template <class T>
// 返回向量长度，结果为 double。
double length(const Point<T> &p) {
    return std::sqrt(square(p));
}

template <class T>
// 返回两点距离，结果为 double。
double distance(const Point<T> &a, const Point<T> &b) {
    return length(a - b);
}

template <class T>
// 返回点到直线距离；两端点须不同。
double distancePL(const Point<T> &p, const Line<T> &l) {
    return std::abs(cross(l.a - l.b, l.a - p)) / length(l.b - l.a);
}

template <class T>
// 返回点到闭线段距离；端点须不同，零长段会除零。
double distancePS(const Point<T> &p, const Line<T> &l) {
    if (dot(p - l.a, l.b - l.a) < 0) {
        return distance(p, l.a);
    }
    if (dot(p - l.b, l.a - l.b) < 0) {
        return distance(p, l.b);
    }
    return distancePL(p, l);
}

template <class T>
// 返回两非平行直线交点；整数类型用 long double 计算后截断，精度场景改用浮点坐标。
Point<T> lineIntersection(const Line<T> &l1, const Line<T> &l2) {
    long double c1 = (long double)cross(l2.b - l2.a, l1.a - l2.a);
    long double c2 = (long double)cross(l2.b - l2.a, l1.a - l1.b);
    long double t = c1 / c2;
    return Point<T>(l1.a.x + (long double)(l1.b.x - l1.a.x) * t,
                    l1.a.y + (long double)(l1.b.y - l1.a.y) * t);
}

template <class T>
// 判断点是否在闭线段上（含端点；浮点需自行加 eps）。
bool pointOnSegment(const Point<T> &p, const Line<T> &l) {
    return cross(p - l.a, l.b - l.a) == 0 && std::min(l.a.x, l.b.x) <= p.x &&
           p.x <= std::max(l.a.x, l.b.x) && std::min(l.a.y, l.b.y) <= p.y &&
           p.y <= std::max(l.a.y, l.b.y);
}

template <class T>
// 判断点是否在简单多边形闭区域内（边界算内，与绕向无关，O(n)）。
bool pointInPolygon(const Point<T> &a, const std::vector<Point<T>> &p) {
    int n = p.size();
    for (int i = 0; i < n; i++) {
        if (pointOnSegment(a, Line(p[i], p[(i + 1) % n]))) {
            return true;
        }
    }
    int t = 0;
    for (int i = 0; i < n; i++) {
        auto u = p[i];
        auto v = p[(i + 1) % n];
        if (u.x < a.x && v.x >= a.x && cross(v - u, a - u) > 0) {
            t ^= 1;
        }
        if (u.x >= a.x && v.x < a.x && cross(u - v, a - v) > 0) {
            t ^= 1;
        }
    }
    return t == 1;
}

template <class T>
// 返回两闭线段相交分类：0 不相交、1 严格相交、2 重叠（返回两端点）、3 端点相交。
std::tuple<int, Point<T>, Point<T>> segmentIntersection(const Line<T> &l1, const Line<T> &l2) {
    if (l1.a == l1.b)
        return {pointOnSegment(l1.a, l2) ? 3 : 0, l1.a, l1.a};
    if (l2.a == l2.b)
        return {pointOnSegment(l2.a, l1) ? 3 : 0, l2.a, l2.a};
    if (std::max(l1.a.x, l1.b.x) < std::min(l2.a.x, l2.b.x) ||
        std::min(l1.a.x, l1.b.x) > std::max(l2.a.x, l2.b.x) ||
        std::max(l1.a.y, l1.b.y) < std::min(l2.a.y, l2.b.y) ||
        std::min(l1.a.y, l1.b.y) > std::max(l2.a.y, l2.b.y)) {
        return {0, Point<T>(), Point<T>()};
    }
    if (cross(l1.b - l1.a, l2.b - l2.a) == 0) {
        if (cross(l1.b - l1.a, l2.a - l1.a) != 0) {
            return {0, Point<T>(), Point<T>()};
        }
        Point<T> p1(std::max(std::min(l1.a.x, l1.b.x), std::min(l2.a.x, l2.b.x)),
                    std::max(std::min(l1.a.y, l1.b.y), std::min(l2.a.y, l2.b.y)));
        Point<T> p2(std::min(std::max(l1.a.x, l1.b.x), std::max(l2.a.x, l2.b.x)),
                    std::min(std::max(l1.a.y, l1.b.y), std::max(l2.a.y, l2.b.y)));
        if (!pointOnSegment(p1, l1)) {
            std::swap(p1.y, p2.y);
        }
        if (p1 == p2) {
            return {3, p1, p2};
        }
        return {2, p1, p2};
    }
    auto cp1 = cross(l2.a - l1.a, l2.b - l1.a);
    auto cp2 = cross(l2.a - l1.b, l2.b - l1.b);
    auto cp3 = cross(l1.a - l2.a, l1.b - l2.a);
    auto cp4 = cross(l1.a - l2.b, l1.b - l2.b);
    if ((cp1 > 0 && cp2 > 0) || (cp1 < 0 && cp2 < 0) || (cp3 > 0 && cp4 > 0) ||
        (cp3 < 0 && cp4 < 0)) {
        return {0, Point<T>(), Point<T>()};
    }
    Point p = lineIntersection(l1, l2);
    if (cp1 != 0 && cp2 != 0 && cp3 != 0 && cp4 != 0) {
        return {1, p, p};
    }
    return {3, p, p};
}

template <class T>
// 返回两闭线段最短距离，相交或重叠为 0。
double distanceSS(const Line<T> &l1, const Line<T> &l2) {
    if (std::get<0>(segmentIntersection(l1, l2)) != 0) {
        return 0.0;
    }
    return std::min(
        {distancePS(l1.a, l2), distancePS(l1.b, l2), distancePS(l2.a, l1), distancePS(l2.b, l1)});
}

template <class T>
// 返回简单多边形两倍无向面积，顶点须按边界顺序；空集或单点返回 0。
T polygonArea2(const std::vector<Point<T>> &p) {
    T s = 0;
    int n = p.size();
    for (int i = 0; i < n; i++) {
        s += cross(p[i], p[(i + 1) % n]);
    }
    return std::abs(s);
}

namespace TriangleExtrema {
    using P = Point<long long>;
    using Int = __int128;
    namespace detail {
        Int det(P a, P b) {
            return Int(a.x) * b.y - Int(a.y) * b.x;
        }
        Int area(P a, P b, P c) {
            Int v = det(b - a, c - a);
            return v < 0 ? -v : v;
        }
        void valid(const std::vector<P> &p) {
            for (P a : p)
                if (a.x < -1000000000LL || a.x > 1000000000LL || a.y < -1000000000LL ||
                    a.y > 1000000000LL)
                    throw std::invalid_argument("TriangleExtrema: coordinate exceeds 1e9");
        }
    }
    struct TriangleResult {
        Int minimum = 0, maximum = 0;
        std::array<int, 3> maximumIds{-1, -1, -1};
    };
    // 极角扫描求最小/最大两倍面积及最大三角形原输入编号；不足三点编号为-1，共线/重复可返回零面积三点。
    TriangleResult minmaxWithIds(const std::vector<P> &input) {
        detail::valid(input);
        int original = int(input.size());
        if (original < 3)
            return {};
        TriangleResult result;
        result.maximumIds = {0, 1, 2};
        std::vector<int> ids(original);
        std::iota(ids.begin(), ids.end(), 0);
        std::sort(ids.begin(), ids.end(),
                  [&](int a, int b) { return input[a] == input[b] ? a < b : input[a] < input[b]; });
        std::vector<P> p;
        std::vector<int> originalIds;
        for (int id : ids)
            if (p.empty() || !(input[id] == p.back())) {
                p.push_back(input[id]);
                originalIds.push_back(id);
            }
        int n = int(p.size());
        if (n < 3)
            return result;
        struct Event {
            int a, b;
            P d;
        };
        std::vector<Event> events;
        for (int i = 0; i < n; ++i)
            for (int j = i + 1; j < n; ++j) {
                P d = p[j] - p[i];
                if (d.x < 0 || (d.x == 0 && d.y < 0))
                    d = -d;
                events.push_back({i, j, d});
            }
        std::sort(events.begin(), events.end(), [](Event a, Event b) {
            Int v = detail::det(a.d, b.d);
            if (v != 0)
                return v > 0;
            return std::tie(a.a, a.b) < std::tie(b.a, b.b);
        });
        std::vector<int> order(n), pos(n);
        std::iota(order.begin(), order.end(), 0);
        std::iota(pos.begin(), pos.end(), 0);
        Int mn = original > n ? 0 : (Int(1) << 120), mx = 0;
        for (int l = 0; l < int(events.size());) {
            int r = l + 1;
            while (r < int(events.size()) && detail::det(events[l].d, events[r].d) == 0)
                ++r;
            std::vector<int> tied;
            for (int i = l; i < r; ++i) {
                auto e = events[i];
                int a = pos[e.a], b = pos[e.b];
                if (a > b)
                    std::swap(a, b);
                if (b - a > 1)
                    mn = 0;
                if (a > 0)
                    mn = std::min(mn, detail::area(p[e.a], p[e.b], p[order[a - 1]]));
                if (b + 1 < n)
                    mn = std::min(mn, detail::area(p[e.a], p[e.b], p[order[b + 1]]));
                for (int c : std::array<int, 2>{order.front(), order.back()}) {
                    Int value = detail::area(p[e.a], p[e.b], p[c]);
                    if (value > mx) {
                        mx = value;
                        result.maximumIds = {originalIds[e.a], originalIds[e.b], originalIds[c]};
                    }
                }
                tied.push_back(e.a);
                tied.push_back(e.b);
            }
            P d = events[l].d;
            std::sort(tied.begin(), tied.end(), [&](int a, int b) {
                Int A = detail::det(d, p[a]), B = detail::det(d, p[b]);
                return A != B ? A < B : pos[a] < pos[b];
            });
            tied.erase(std::unique(tied.begin(), tied.end()), tied.end());
            for (int a = 0; a < int(tied.size());) {
                int b = a + 1;
                while (b < int(tied.size()) &&
                       detail::det(d, p[tied[a]]) == detail::det(d, p[tied[b]]))
                    ++b;
                int first = pos[tied[a]], last = pos[tied[b - 1]];
                std::reverse(order.begin() + first, order.begin() + last + 1);
                for (int k = first; k <= last; ++k)
                    pos[order[k]] = k;
                a = b;
            }
            l = r;
        }
        result.minimum = mn;
        result.maximum = mx;
        return result;
    }
    // 只需最小/最大两倍面积时调用此包装；同一极角扫描实现，不重复算法。
    std::pair<Int, Int> minmax(const std::vector<P> &p) {
        auto r = minmaxWithIds(p);
        return {r.minimum, r.maximum};
    }

    // 严格凸 CCW 多边形最大四边形两倍面积；每条对角线两侧分别推进三角形极值点。
    Int maximumQuadrilateral(const std::vector<P> &p) {
        detail::valid(p);
        int n = int(p.size());
        if (n < 4)
            return 0;
        for (int i = 0; i < n; ++i)
            if (detail::det(p[(i + 1) % n] - p[i], p[(i + 2) % n] - p[(i + 1) % n]) <= 0)
                throw std::invalid_argument("not strict CCW convex");
        auto area = [&](int a, int b, int c) { return detail::area(p[a % n], p[b % n], p[c % n]); };
        Int ans = 0;
        for (int i = 0; i < n; ++i) {
            int k = i + 1, t = i + 3;
            for (int j = i + 2; j <= i + n - 2; ++j) {
                while (k + 1 < j && area(i, j, k + 1) >= area(i, j, k))
                    ++k;
                t = std::max(t, j + 1);
                while (t + 1 < i + n && area(i, j, t + 1) >= area(i, j, t))
                    ++t;
                ans = std::max(ans, area(i, j, k) + area(i, j, t));
            }
        }
        return ans;
    }
    // 任意整数点集的最大简单四边形两倍面积；允许凹形/重复/共线，少于四个输入点返回零。
    Int maximumQuadrilateralPoints(std::vector<P> p) {
        detail::valid(p);
        if (p.size() < 4)
            return 0;
        int original = int(p.size());
        std::sort(p.begin(), p.end());
        p.erase(std::unique(p.begin(), p.end()), p.end());
        std::vector<P> h;
        for (P a : p) {
            while (h.size() > 1 && detail::det(h.back() - h[h.size() - 2], a - h.back()) <= 0)
                h.pop_back();
            h.push_back(a);
        }
        int lower = int(h.size());
        for (int i = int(p.size()) - 2; i >= 0; --i) {
            while (int(h.size()) > lower &&
                   detail::det(h.back() - h[h.size() - 2], p[i] - h.back()) <= 0)
                h.pop_back();
            h.push_back(p[i]);
        }
        if (h.size() > 1)
            h.pop_back();
        if (h.size() < 3)
            return 0;
        if (h.size() >= 4)
            return maximumQuadrilateral(h);
        Int total = detail::area(h[0], h[1], h[2]);
        if (original > int(p.size()))
            return total;
        Int ans = 0;
        for (P q : p) {
            if (q == h[0] || q == h[1] || q == h[2])
                continue;
            Int cut = std::min({detail::area(q, h[0], h[1]), detail::area(q, h[1], h[2]),
                                detail::area(q, h[2], h[0])});
            ans = std::max(ans, total - cut);
        }
        return ans;
    }

}
