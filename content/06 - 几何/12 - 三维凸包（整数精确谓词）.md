# 三维凸包（整数精确谓词）

整数坐标绝对值<=1e6 用 __int128 精确判定；浮点坐标有限且绝对值<=1e100。 浮点先平移缩放，eps（默认 1e-12，须 0<=eps<1）为归一化叉积/六倍体积阈值而非距离容差，有效阈值 max(eps,64*机器 epsilon)。

独立 Hull3D 点类型；严格去重，points 为排序后原坐标而非输入编号；dimension=-1/0/1/2/3 为 空/点/线段/平面/立体，boundary 仅低维给下标。

3D faces 外向、共面大面可拆多个三角形，2D 单面扇形剖分；浮点结果为近似（禁 fast-math），近退化/不稳定 3D 抛 runtime_error。

排序 O(n log n)；增量面扫描与有序 horizon 最坏 O(n² log n)，浮点末尾支撑/闭合检查同阶；空间 O(n)。

```cpp
namespace Hull3D {
    using Int = __int128;
    template <class T> struct BasicPoint {
        T x, y, z;
    };
    template <class T> struct BasicHull {
        int dimension = -1;
        std::vector<BasicPoint<T>> points; // 排序去重后的点，保留原坐标；下标不是输入点号
        std::vector<std::array<int, 3>> faces; // 三角面的三个 points 下标；立体时按外向绕序
        std::vector<int> boundary; // 低维凸包的边界顶点，存 points 下标；立体时为空
    };
    using Point = BasicPoint<long long>;
    using Hull = BasicHull<long long>;
    using FloatingPoint = BasicPoint<long double>;
    using FloatingHull = BasicHull<long double>;
    namespace detail {
        template <class T>
        using Wide = std::conditional_t<std::is_same_v<T, long long>, Int, long double>;
        template <class T> BasicPoint<T> sub(BasicPoint<T> a, BasicPoint<T> b) {
            return {a.x - b.x, a.y - b.y, a.z - b.z};
        }
        template <class T> std::array<Wide<T>, 3> cross(BasicPoint<T> a, BasicPoint<T> b) {
            return {Wide<T>(a.y) * b.z - Wide<T>(a.z) * b.y,
                    Wide<T>(a.z) * b.x - Wide<T>(a.x) * b.z,
                    Wide<T>(a.x) * b.y - Wide<T>(a.y) * b.x};
        }
        template <class T> Wide<T> dot(std::array<Wide<T>, 3> n, BasicPoint<T> a) {
            return n[0] * a.x + n[1] * a.y + n[2] * a.z;
        }
        template <class T>
        Wide<T> orient(BasicPoint<T> a, BasicPoint<T> b, BasicPoint<T> c, BasicPoint<T> d) {
            return dot(cross(sub(b, a), sub(c, a)), sub(d, a));
        }
        template <class T> bool same(BasicPoint<T> a, BasicPoint<T> b) {
            return a.x == b.x && a.y == b.y && a.z == b.z;
        }
        template <class V> bool nonzero(std::array<V, 3> n, V eps = 0) {
            return n[0] > eps || n[0] < -eps || n[1] > eps || n[1] < -eps || n[2] > eps ||
                   n[2] < -eps;
        }
        // 浮点入口使用归一化工作坐标。
        template <class T> BasicHull<T> build(std::vector<BasicPoint<T>> p, long double eps) {
            using P = BasicPoint<T>;
            using Value = Wide<T>;
            Value tolerance = Value(eps);
            constexpr bool floating = std::is_floating_point_v<T>;
            for (P a : p) {
                if constexpr (floating) {
                    if (!std::isfinite(a.x) || !std::isfinite(a.y) || !std::isfinite(a.z) ||
                        std::abs(a.x) > 1e100L || std::abs(a.y) > 1e100L || std::abs(a.z) > 1e100L)
                        throw std::invalid_argument(
                            "Hull3D: nonfinite/oversized floating coordinate");
                } else if (a.x < -1000000 || a.x > 1000000 || a.y < -1000000 || a.y > 1000000 ||
                           a.z < -1000000 || a.z > 1000000)
                    throw std::invalid_argument("Hull3D: coordinate exceeds 1e6");
            }
            std::sort(p.begin(), p.end(),
                      [](P a, P b) { return std::tie(a.x, a.y, a.z) < std::tie(b.x, b.y, b.z); });
            p.erase(std::unique(p.begin(), p.end(), same<T>), p.end());
            BasicHull<T> h;
            h.points = p;
            int n = int(p.size());
            if (!n)
                return h;
            h.dimension = 0;
            h.boundary = {0};
            if (n == 1)
                return h;
            if constexpr (floating) {
                P low = p[0], high = p[0];
                for (P a : p) {
                    low = {std::min(low.x, a.x), std::min(low.y, a.y), std::min(low.z, a.z)};
                    high = {std::max(high.x, a.x), std::max(high.y, a.y), std::max(high.z, a.z)};
                }
                T scale = std::max({high.x - low.x, high.y - low.y, high.z - low.z});
                for (P &a : p)
                    a = {(a.x - low.x) / scale, (a.y - low.y) / scale, (a.z - low.z) / scale};
            }
            int b = 1, k = -1, l = -1;
            // 浮点选较远基线/最大叉积/最大体积作种子，避免近重复误判维数
            if constexpr (floating) {
                long double best = -1;
                for (int i = 1; i < n; ++i) {
                    P v = sub(p[i], p[0]);
                    auto d = v.x * v.x + v.y * v.y + v.z * v.z;
                    if (d > best) {
                        best = d;
                        b = i;
                    }
                }
            }
            long double bestCross = eps, bestVolume = eps;
            for (int i = 1; i < n; ++i) {
                auto normal = cross(sub(p[b], p[0]), sub(p[i], p[0]));
                if constexpr (floating) {
                    auto score =
                        std::max({std::abs(normal[0]), std::abs(normal[1]), std::abs(normal[2])});
                    if (score > bestCross) {
                        bestCross = score;
                        k = i;
                    }
                } else if (nonzero(normal)) {
                    k = i;
                    break;
                }
            }
            if (k < 0) {
                h.dimension = 1;
                h.boundary = {0, n - 1};
                if constexpr (floating) {
                    auto d = sub(p[b], p[0]);
                    auto projection = [&](int i) {
                        return p[i].x * d.x + p[i].y * d.y + p[i].z * d.z;
                    };
                    int first = 0, last = 0;
                    for (int i = 1; i < n; ++i) {
                        if (projection(i) < projection(first))
                            first = i;
                        if (projection(i) > projection(last))
                            last = i;
                    }
                    h.boundary = {first, last};
                }
                return h;
            }
            for (int i = 1; i < n; ++i) {
                auto v = orient(p[0], p[b], p[k], p[i]);
                if constexpr (floating) {
                    if (std::abs(v) > bestVolume) {
                        bestVolume = std::abs(v);
                        l = i;
                    }
                } else if (v) {
                    l = i;
                    break;
                }
            }
            if (l < 0) {
                h.dimension = 2;
                auto normal = cross(sub(p[b], p[0]), sub(p[k], p[0]));
                int drop;
                if constexpr (floating)
                    drop = int(
                        std::max_element(normal.begin(), normal.end(),
                                         [](auto a, auto b) { return std::abs(a) < std::abs(b); }) -
                        normal.begin());
                else
                    drop = normal[0] ? 0 : normal[1] ? 1 : 2;
                auto xy = [&](int i) {
                    return drop == 0   ? std::pair{p[i].y, p[i].z}
                           : drop == 1 ? std::pair{p[i].x, p[i].z}
                                       : std::pair{p[i].x, p[i].y};
                };
                auto turn = [&](int a, int b, int c) {
                    auto [x, y] = xy(a);
                    auto [u, v] = xy(b);
                    auto [s, t] = xy(c);
                    return Value(u - x) * (t - y) - Value(v - y) * (s - x);
                };
                std::vector<int> ids(n), out;
                std::iota(ids.begin(), ids.end(), 0);
                std::sort(ids.begin(), ids.end(), [&](int a, int b) { return xy(a) < xy(b); });
                for (int i : ids) {
                    while (out.size() > 1 && turn(out[out.size() - 2], out.back(), i) <= tolerance)
                        out.pop_back();
                    out.push_back(i);
                }
                int lower = int(out.size());
                for (int i = n - 2; i >= 0; --i) {
                    while (int(out.size()) > lower &&
                           turn(out[out.size() - 2], out.back(), ids[i]) <= tolerance)
                        out.pop_back();
                    out.push_back(ids[i]);
                }
                out.pop_back();
                h.boundary = out;
                for (int i = 1; i + 1 < int(out.size()); ++i)
                    h.faces.push_back({out[0], out[i], out[i + 1]});
                return h;
            }
            h.dimension = 3;
            h.boundary.clear();
            std::array<int, 4> seed{0, b, k, l};
            P centerSum{p[0].x + p[b].x + p[k].x + p[l].x, p[0].y + p[b].y + p[k].y + p[l].y,
                        p[0].z + p[b].z + p[k].z + p[l].z};
            auto add = [&](int a, int b, int c) {
                auto normal = cross(sub(p[b], p[a]), sub(p[c], p[a]));
                if (!nonzero(normal, tolerance))
                    return;
                P w{centerSum.x - 4 * p[a].x, centerSum.y - 4 * p[a].y, centerSum.z - 4 * p[a].z};
                if (dot(normal, w) > 0)
                    std::swap(b, c);
                h.faces.push_back({a, b, c});
            };
            add(0, b, k);
            add(0, b, l);
            add(0, k, l);
            add(b, k, l);
            for (int i = 0; i < n; ++i) {
                if (std::find(seed.begin(), seed.end(), i) != seed.end())
                    continue;
                std::vector<std::array<int, 3>> keep;
                std::map<std::pair<int, int>, int> horizon;
                for (auto f : h.faces) {
                    if (orient(p[f[0]], p[f[1]], p[f[2]], p[i]) <= tolerance) {
                        keep.push_back(f);
                        continue;
                    }
                    for (int t = 0; t < 3; ++t) {
                        int a = f[t], b = f[(t + 1) % 3];
                        auto rev = horizon.find({b, a});
                        if (rev != horizon.end())
                            horizon.erase(rev);
                        else
                            horizon[{a, b}] = 1;
                    }
                }
                h.faces = std::move(keep);
                for (auto [edge, unused] : horizon)
                    add(edge.first, edge.second, i);
            }
            if constexpr (floating) {
                std::map<std::pair<int, int>, int> edges;
                std::set<int> vertices;
                auto unstable = []() {
                    throw std::runtime_error(
                        "Hull3D: unstable floating topology; adjust scale/eps or use exact predicates");
                };
                for (auto f : h.faces) {
                    for (P a : p)
                        if (orient(p[f[0]], p[f[1]], p[f[2]], a) > eps)
                            unstable();
                    for (int i = 0; i < 3; ++i) {
                        ++edges[{f[i], f[(i + 1) % 3]}];
                        vertices.insert(f[i]);
                    }
                }
                for (auto [e, count] : edges) {
                    auto rev = edges.find({e.second, e.first});
                    if (count != 1 || rev == edges.end() || rev->second != 1)
                        unstable();
                }
                if (int(vertices.size()) - int(edges.size()) / 2 + int(h.faces.size()) != 2)
                    unstable();
            }
            return h;
        }
    }
    // 整数建包；空、重复、共线、共面均支持，所有判定在指定范围内精确。
    Hull build(std::vector<Point> p) {
        return detail::build(std::move(p), 0);
    }
    // 浮点建包；返回原坐标点与近似面/低维边界，近退化按 eps 分类；非法输入/不稳定3D网格抛异常。
    // ponytail: 归一化绝对阈值不认证近退化几何；需更强保证时升级为自适应/精确方向谓词。
    FloatingHull buildFloating(std::vector<FloatingPoint> p, long double eps = 1e-12L) {
        if (!std::isfinite(eps) || eps < 0 || eps >= 1)
            throw std::invalid_argument("Hull3D: eps must be finite in [0,1)");
        return detail::build(std::move(p),
                             std::max(eps, 64 * std::numeric_limits<long double>::epsilon()));
    }
    // 返回三角面面积和；2D 仅计单面，3D 计封闭外表面，两种数值 Hull 均可传。
    template <class T> long double surfaceArea(const BasicHull<T> &h) {
        long double s = 0;
        for (auto f : h.faces) {
            auto n = detail::cross(detail::sub(h.points[f[1]], h.points[f[0]]),
                                   detail::sub(h.points[f[2]], h.points[f[0]]));
            s += std::hypot(static_cast<long double>(n[0]), static_cast<long double>(n[1]),
                            static_cast<long double>(n[2])) /
                 2;
        }
        return s;
    }
    // 返回3D体积，低维为零；整数六倍体积精确累加，浮点先减去同一原点再累加以减小消去误差。
    template <class T> long double volume(const BasicHull<T> &h) {
        if (h.dimension != 3)
            return 0;
        detail::Wide<T> v = 0;
        auto origin = h.points[0];
        for (auto f : h.faces)
            v += detail::orient(origin, h.points[f[0]], h.points[f[1]], h.points[f[2]]);
        return std::abs(static_cast<long double>(v)) / 6;
    }
}
```
