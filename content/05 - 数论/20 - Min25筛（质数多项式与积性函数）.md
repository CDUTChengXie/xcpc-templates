# Min25筛（质数多项式与积性函数）

0<=n<=10^12；模 998244353；f(1)=1，f(p)=c0+c1*p+c2*p*p。 primePower(p,e) 返回 f(p^e) 的模值，e>=1 且 e=1 时必须与三个系数一致；传入的函数须支持 i64 质数。 所有答案取模；对象固定对应构造时的 n，无任意 x 查询接口；n=0 的前缀和为 0。

质数表 O(n^(3/4)/log n)，积性回溯常用估计 O(n^(3/4)/log n)；O(sqrt n) 空间，回调成本另计。

```cpp
namespace min25 {
    constexpr int mod = 998244353;
    // 返回规范到 [0,mod) 的 int 余数。
    inline int norm(long long x) {
        x %= mod;
        if (x < 0)
            x += mod;
        return x;
    }
    // 整除分块点上的 Min25 筛表。
    struct Sieve {
      private:
        long long n;
        int rt;
        std::vector<int> primes, small, large;
        std::vector<long long> w;
        std::vector<std::array<int, 3>> g, prefix;
        int id(long long x) const {
            return x <= rt ? small[x] : large[n / x];
        }

      public:
        // 预处理所有整除分块点上的质数幂和。
        explicit Sieve(long long n) : n(n) {
            assert(n >= 0 && n <= 1000000000000LL);
            rt = std::sqrt((long double)n);
            while (1LL * (rt + 1) * (rt + 1) <= n)
                ++rt;
            while (1LL * rt * rt > n)
                --rt;
            small.assign(rt + 1, 0);
            large.assign(rt + 1, 0);
            std::vector<bool> composite(rt + 1);
            prefix.push_back({0, 0, 0});
            for (int i = 2; i <= rt; ++i) {
                if (!composite[i]) {
                    primes.push_back(i);
                    auto t = prefix.back();
                    t[0] = norm(t[0] + 1);
                    t[1] = norm(t[1] + i);
                    t[2] = norm(t[2] + 1LL * i * i % mod);
                    prefix.push_back(t);
                }
                for (int p : primes) {
                    if (1LL * i * p > rt)
                        break;
                    composite[i * p] = true;
                    if (i % p == 0)
                        break;
                }
            }
            for (long long l = 1; l <= n;) {
                long long x = n / l;
                int k = w.size();
                w.push_back(x);
                if (x <= rt)
                    small[x] = k;
                else
                    large[n / x] = k;
                long long a = x % mod, b = (a + 1) % mod;
                g.push_back({norm(a - 1), norm(a * b % mod * ((mod + 1) / 2) % mod - 1),
                             norm(a * b % mod * ((2 * a + 1) % mod) % mod * 166374059 % mod - 1)});
                l = n / x + 1;
            }
            for (int j = 0; j < (int)primes.size(); ++j) {
                long long p = primes[j];
                std::array<long long, 3> pk{1, p, p * p % mod};
                for (int k = 0; k < (int)w.size() && p * p <= w[k]; ++k) {
                    int t = id(w[k] / p);
                    for (int d = 0; d < 3; ++d)
                        g[k][d] = norm(g[k][d] - pk[d] * norm(g[t][d] - prefix[j][d]) % mod);
                }
            }
        }
        // 返回不超过 n 的质数 0/1/2 次幂和。
        std::array<int, 3> primeSums() const {
            return n < 2 ? std::array<int, 3>{0, 0, 0} : g[id(n)];
        }
        // 返回 f[1..n] 的和模 998244353。
        template <class F> int sum(std::array<int, 3> coeff, F primePower) const {
            if (!n)
                return 0;
            for (auto &v : coeff)
                v = norm(v);
            auto dot = [&](const std::array<int, 3> &a) {
                long long s = 0;
                for (int d = 0; d < 3; ++d)
                    s = (s + 1LL * a[d] * coeff[d]) % mod;
                return (int)s;
            };
            auto dfs = [&](auto &&self, long long x, int j) -> int {
                if (x < 2 || (j < (int)primes.size() && primes[j] > x))
                    return 0;
                long long ans = norm(dot(g[id(x)]) - dot(prefix[j]));
                for (int i = j; i < (int)primes.size() && 1LL * primes[i] * primes[i] <= x; ++i) {
                    long long p = primes[i], pe = p;
                    int e = 1;
                    while (pe <= x / p) {
                        ans = (ans + 1LL * norm(primePower(p, e)) * self(self, x / pe, i + 1) +
                               norm(primePower(p, e + 1))) %
                              mod;
                        pe *= p;
                        ++e;
                    }
                }
                return ans;
            };
            return norm(dfs(dfs, n, 0) + 1);
        }
    };
}
```
