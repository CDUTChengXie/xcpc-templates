# 高斯消元

a 非空且各行同宽，b.size()==a.size()；模域须为素数域；实数 eps>=0，自定义域需能除以非零主元。 a[i][j] 是第 i 条方程中第 j 个未知数的系数，b[i] 是右端常数；下标从 0 开始。

n 行 m 列时返回 0 唯一解、1 多解、-1 无解；a 不变，b 被改写；仅状态 0 时 b[0..m) 对应未知数。 gauss 返回的向量长度为 n，仅前 m 项为答案，末尾多余方程项不用读；多解时不提供一组变量解。

O(n*m*min(n,m)) 时间，O(n*m) 空间。

```cpp
#ifndef XCPC_MODINT

// 返回取模类型的快速幂。
template <class T> constexpr T power(T a, i64 b) {
    T res = 1;
    for (; b; b /= 2, a *= a) {
        if (b % 2) {
            res *= a;
        }
    }
    return res;
}

// 返回 a*b mod p 的原始余数。
constexpr i64 mul(i64 a, i64 b, i64 p) {
    return i64(i128(a) * b % p);
}

// 大模数取模类，模数可编译期固定或运行时设置。
template <i64 P> struct MLong {
    i64 x;

    // 构造值为 0 的取模数。
    constexpr MLong() : x{} {}
    // 用整数构造并规范到 [0,mod)。
    constexpr MLong(i64 x) : x{norm(x % getMod())} {}

    // 动态模数存储。
    static i64 Mod;
    // 返回当前模数。
    constexpr static i64 getMod() {
        return P > 0 ? P : Mod;
    }
    // 设置动态模数，仅 P=0 时生效。
    constexpr static void setMod(i64 Mod_) {
        Mod = Mod_;
    }
    // 只加减一次规范到 [0,mod)；输入须在 [-mod,2*mod) 内。
    constexpr i64 norm(i64 x) const {
        if (x < 0) {
            x += getMod();
        }
        if (x >= getMod()) {
            x -= getMod();
        }
        return x;
    }
    // 返回底层值。
    constexpr i64 val() const {
        return x;
    }
    // 转换为 i64。
    explicit constexpr operator i64() const {
        return x;
    }
    // 返回相反数。
    constexpr MLong operator-() const {
        MLong res;
        res.x = norm(getMod() - x);
        return res;
    }
    // 返回模逆元，要求模数为素数且自身非零。
    constexpr MLong inv() const {
        assert(x != 0);
        return power(*this, getMod() - 2);
    }
    // 乘并赋值。
    constexpr MLong &operator*=(MLong rhs) & {
        x = mul(x, rhs.x, getMod());
        return *this;
    }
    // 加并赋值。
    constexpr MLong &operator+=(MLong rhs) & {
        x = i64((i128(x) + rhs.x) % getMod());
        return *this;
    }
    // 减并赋值。
    constexpr MLong &operator-=(MLong rhs) & {
        x = norm(x - rhs.x);
        return *this;
    }
    // 除并赋值。
    constexpr MLong &operator/=(MLong rhs) & {
        return *this *= rhs.inv();
    }
    // 返回两数乘积。
    friend constexpr MLong operator*(MLong lhs, MLong rhs) {
        MLong res = lhs;
        res *= rhs;
        return res;
    }
    // 返回两数之和。
    friend constexpr MLong operator+(MLong lhs, MLong rhs) {
        MLong res = lhs;
        res += rhs;
        return res;
    }
    // 返回两数之差。
    friend constexpr MLong operator-(MLong lhs, MLong rhs) {
        MLong res = lhs;
        res -= rhs;
        return res;
    }
    // 返回两数之商。
    friend constexpr MLong operator/(MLong lhs, MLong rhs) {
        MLong res = lhs;
        res /= rhs;
        return res;
    }
    // 从流中读入一个取模数。
    friend constexpr std::istream &operator>>(std::istream &is, MLong &a) {
        i64 v;
        if (is >> v)
            a = MLong(v);
        return is;
    }
    // 向流中输出规范值。
    friend constexpr std::ostream &operator<<(std::ostream &os, const MLong &a) {
        return os << a.val();
    }
    // 按规范值判断相等。
    friend constexpr bool operator==(MLong lhs, MLong rhs) {
        return lhs.val() == rhs.val();
    }
    // 按规范值判断不等。
    friend constexpr bool operator!=(MLong lhs, MLong rhs) {
        return lhs.val() != rhs.val();
    }
};

// 自定义消元骨架，返回 0 唯一解 / 1 多解 / -1 无解，并原地写回 b。
template <class T, class NonZero, class Better>
int gaussImpl(std::vector<std::vector<T>> a, std::vector<T> &b, NonZero nz, Better better) {
    int n = int(a.size()), m = int(a[0].size());
    int row = 0;
    for (int col = 0; col < m && row < n; col++) {
        int r = row;
        for (int i = row; i < n; i++) {
            if (better(a[i][col], a[r][col])) {
                r = i;
            }
        }
        if (!nz(a[r][col])) {
            continue;
        }
        std::swap(a[row], a[r]);
        std::swap(b[row], b[r]);
        T f = a[row][col];
        for (int j = col; j < m; j++) {
            a[row][j] = a[row][j] / f;
        }
        b[row] = b[row] / f;
        for (int i = 0; i < n; i++) {
            if (i == row) {
                continue;
            }
            T g = a[i][col];
            if (!nz(g)) {
                continue;
            }
            for (int j = col; j < m; j++) {
                a[i][j] -= g * a[row][j];
            }
            b[i] -= g * b[row];
        }
        row++;
    }
    // 剩余行系数全为 0，常数项非零即矛盾。
    for (int i = row; i < n; i++) {
        if (nz(b[i])) {
            return -1;
        }
    }
    if (row < m) {
        return 1;
    }
    return 0;
}

// 实数/有序类型高斯消元，按绝对值最大选主元。
template <class T> int gaussState(std::vector<std::vector<T>> a, std::vector<T> &b, T eps = T(0)) {
    return gaussImpl(
        std::move(a), b, [&](const T &x) { return abs(x) > eps; },
        [&](const T &x, const T &y) { return abs(x) > abs(y); });
}

// 模域高斯消元，主元只判是否为 0。
template <class T> int gaussMod(std::vector<std::vector<T>> a, std::vector<T> &b) {
    return gaussImpl(
        std::move(a), b, [](const T &x) { return x != T(0); },
        [](const T &x, const T &y) { return x != T(0) && y == T(0); });
}

// 返回模域唯一解；仅前 m 项 solution[j] 为未知数，非唯一时 assert。
std::vector<Z> gauss(std::vector<std::vector<Z>> a, std::vector<Z> b) {
    int st = gaussMod(std::move(a), b);
    assert(st == 0 && "gauss(模版) 要求唯一解；无解或多解请改用 gaussMod");
    return b;
}

// 返回实数唯一解；仅前 m 项 solution[j] 为未知数，非唯一时 assert。
std::vector<double> gauss(std::vector<std::vector<double>> a, std::vector<double> b) {
    int st = gaussState(std::move(a), b, 1e-10);
    assert(st == 0 && "gauss(实数版) 要求唯一解；无解或多解请改用 gaussState");
    return b;
}
```
