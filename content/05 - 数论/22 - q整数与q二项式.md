# q整数与q二项式

prime 为素数且<=INT_MAX，0<=n<=LLONG_MAX；workLimit>=0，查询所需小数字超过上限抛 length_error。 q 自动规范到 [0,prime)；k 越界返回 0；q=0 时合法二项式为 1，q=1 时退化为普通组合数。

liftedInteger 要求 n>=1；返回 [n]_q 去掉 prime 因子后的非零 unit 和因子次数，模 prime 幂精度不足时为空。

求阶 O(sqrt p+log²p)，integer O(log n)，liftedInteger O(log n+log_p LLONG_MAX)，choose O(workLimit·log_p n+log p) 上界。

```cpp
namespace q_combinatorics {
    // 固定 q 与素数模的 q 二项式计算器。
    class Gaussian {
        int q, p, limit;
        long long order;
        long long power(long long a, long long e) const {
            long long r = 1;
            for (; e; e >>= 1, a = a * a % p)
                if (e & 1)
                    r = r * a % p;
            return r;
        }
        int ordinary(long long n, long long k) const {
            long long ans = 1;
            while (n || k) {
                long long a = n % p, b = k % p;
                if (b > a)
                    return 0;
                b = std::min(b, a - b);
                if (b > limit)
                    throw std::length_error("Lucas digit exceeds work bound");
                long long num = 1, den = 1;
                for (long long i = 1; i <= b; ++i)
                    num = num * (a - i + 1) % p, den = den * i % p;
                ans = ans * num % p * power(den, p - 2) % p;
                n /= p;
                k /= p;
            }
            return ans;
        }

      public:
        // 建立固定 q 的计算器。
        Gaussian(int q, int prime = 998244353, int workLimit = 2000000)
            : q(q), p(prime), limit(workLimit), order(1) {
            assert(p >= 2 && limit >= 0);
            this->q %= p;
            if (this->q < 0)
                this->q += p;
            if (this->q <= 1)
                return;
            order = p - 1;
            int x = p - 1;
            for (int f = 2; f <= x / f; ++f)
                if (x % f == 0) {
                    while (x % f == 0)
                        x /= f;
                    while (order % f == 0 && power(this->q, order / f) == 1)
                        order /= f;
                }
            if (x > 1)
                while (order % x == 0 && power(this->q, order / x) == 1)
                    order /= x;
        }
        // 返回 [n]_q。
        int integer(long long n) const {
            assert(n >= 0);
            if (q == 1)
                return n % p;
            return (power(q, n) + p - 1) % p * power((q + p - 1LL) % p, p - 2) % p;
        }
        // 返回 [n]_q 去除 p 因子后的 unit 与指数。
        std::optional<std::pair<int, int>> liftedInteger(long long n) const {
            assert(n >= 1);
            if (q == 0)
                return std::pair{1, 0};
            if (q == 1) {
                int e = 0;
                while (n % p == 0)
                    n /= p, ++e;
                return std::pair{(int)(n % p), e};
            }
            long long M = p;
            while (M <= LLONG_MAX / p)
                M *= p;
            long long rpow = 1, rsum = 0, bpow = q, bsum = 1;
            for (long long e = n; e; e >>= 1) {
                if (e & 1) {
                    rsum = ((__int128)rsum + (__int128)rpow * bsum) % M;
                    rpow = (__int128)rpow * bpow % M;
                }
                bsum = ((__int128)bsum + (__int128)bpow * bsum) % M;
                bpow = (__int128)bpow * bpow % M;
            }
            if (!rsum)
                return {};
            int e = 0;
            while (rsum % p == 0)
                rsum /= p, ++e;
            return std::pair{(int)(rsum % p), e};
        }
        // 返回 q 二项式，支持单位根。
        int choose(long long n, long long k) const {
            assert(n >= 0);
            if (k < 0 || k > n)
                return 0;
            if (q == 0)
                return 1;
            if (q == 1)
                return ordinary(n, k);
            long long a = n % order, b = k % order;
            if (b > a)
                return 0;
            b = std::min(b, a - b);
            if (b > limit)
                throw std::length_error("q digit exceeds work bound");
            long long num = 1, den = 1, pw = power(q, a - b + 1), dw = q;
            for (long long i = 1; i <= b; ++i) {
                num = num * (1 - pw + p) % p;
                den = den * (1 - dw + p) % p;
                pw = pw * q % p;
                dw = dw * q % p;
            }
            return num * power(den, p - 2) % p * ordinary(n / order, k / order) % p;
        }
    };
}
```
