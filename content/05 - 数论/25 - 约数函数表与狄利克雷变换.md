# 约数函数表与狄利克雷变换

tables 使用欧拉筛 minp[0..n]，n>=1 且 minp[i] 是 i>=2 的最小质因子；先 sieve(n) 得到该表。 divisors 的 n>=0；变换数组至少保留下标 0；T 支持 +=、-=，所有中间和须可表示。

下标 1..n 表示整数，下标 0 不参与；不自动取模，取模时可用取模类系数；四种变换原地修改数组。

函数表O(n)，约数集合/四变换O(n log n)，空间O(n)/O(n log n)。

```cpp
namespace divisor_transform {
    // 返回约数个数与约数和表。
    std::pair<std::vector<int>, std::vector<long long>> tables(const std::vector<int> &minp) {
        int n = minp.size();
        assert(n >= 2);
        std::vector<int> tau(n), exponent(n);
        std::vector<long long> sigma(n), power(n), sum(n);
        tau[1] = sigma[1] = power[1] = sum[1] = 1;
        for (int i = 2; i < n; ++i) {
            int p = minp[i];
            assert(p >= 2 && i % p == 0);
            int j = i / p;
            if (j % p == 0) {
                exponent[i] = exponent[j] + 1;
                power[i] = power[j] * p;
                sum[i] = sum[j] + power[i];
                int rest = i / power[i];
                tau[i] = tau[rest] * (exponent[i] + 1);
                sigma[i] = sigma[rest] * sum[i];
            } else {
                exponent[i] = 1;
                power[i] = p;
                sum[i] = p + 1LL;
                tau[i] = 2 * tau[j];
                sigma[i] = (p + 1LL) * sigma[j];
            }
        }
        return {tau, sigma};
    }
    // 返回每个 i∈[1,n] 的递增正约数。
    std::vector<std::vector<int>> divisors(int n) {
        assert(n >= 0);
        std::vector<std::vector<int>> d(n + 1);
        for (int i = 1; i <= n; ++i)
            for (int j = i; j <= n; j += i)
                d[j].push_back(i);
        return d;
    }
    // 原地做除数和变换。
    template <class T> void divisorZeta(std::vector<T> &a) {
        int n = a.size() - 1;
        for (int i = n; i >= 1; --i)
            for (int j = i * 2; j <= n; j += i)
                a[j] += a[i];
    }
    // 原地做除数和逆变换。
    template <class T> void divisorMobius(std::vector<T> &a) {
        int n = a.size() - 1;
        for (int i = 1; i <= n; ++i)
            for (int j = i * 2; j <= n; j += i)
                a[j] -= a[i];
    }
    // 原地做倍数和变换。
    template <class T> void multipleZeta(std::vector<T> &a) {
        int n = a.size() - 1;
        for (int i = 1; i <= n; ++i)
            for (int j = i * 2; j <= n; j += i)
                a[i] += a[j];
    }
    // 原地做倍数和逆变换。
    template <class T> void multipleMobius(std::vector<T> &a) {
        int n = a.size() - 1;
        for (int i = n; i >= 1; --i)
            for (int j = i * 2; j <= n; j += i)
                a[i] -= a[j];
    }
}
```
