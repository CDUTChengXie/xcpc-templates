# 康托展开与排列还原

精确版要求元素互异、perm 是 a 的完整排列、n<=20，排名从 0 起且小于 n!。模排名要求 perm 是 {1..n} 的排列，返回从 1 起的排名模 998244353。

空排列精确排名为 0，模排名为 1；非法模排列抛 invalid_argument。

精确版 O(n²) 时间、O(n) 空间；模排名 O(n log n) 时间、O(n) 空间。

```cpp
// 返回排列在 a 的全部排列中的 0-based 字典序排名。
// ponytail: 精确排名只支持 n<=20，线性扫描 O(n^2)；大规模模排名用 cantorRankMod 的树状数组。
i64 cantorRank(const std::vector<int> &a, const std::vector<int> &perm) {
    int n = (int)a.size();
    std::vector<int> sorted = a;
    std::sort(sorted.begin(), sorted.end());
    std::vector<char> used(n, false);
    i64 rank = 0;
    std::vector<i64> fac(n + 1, 1);
    for (int i = 1; i <= n; i++) {
        fac[i] = fac[i - 1] * i;
    }
    for (int i = 0; i < n; i++) {
        int idx = (int)(std::lower_bound(sorted.begin(), sorted.end(), perm[i]) - sorted.begin());
        i64 smaller = 0;
        for (int j = 0; j < idx; j++) {
            if (!used[j]) {
                smaller++;
            }
        }
        rank += smaller * fac[n - i - 1];
        used[idx] = true;
    }
    return rank;
}

// 由 0-based 字典序排名还原排列。
std::vector<int> unrankPermutation(const std::vector<int> &a, i64 rank) {
    int n = (int)a.size();
    std::vector<int> pool = a;
    std::sort(pool.begin(), pool.end());
    std::vector<i64> fac(n + 1, 1);
    for (int i = 1; i <= n; i++) {
        fac[i] = fac[i - 1] * i;
    }
    std::vector<int> res;
    for (int i = 0; i < n; i++) {
        i64 f = fac[n - i - 1];
        i64 idx = rank / f;
        rank %= f;
        res.push_back(pool[(size_t)idx]);
        pool.erase(pool.begin() + (size_t)idx);
    }
    return res;
}

// 返回 {1..n} 排列的 1-based 字典序排名模 998244353。
int cantorRankMod(const std::vector<int> &perm) {
    if (perm.size() > size_t(INT_MAX / 2))
        throw std::length_error("cantor permutation too large");
    int n = (int)perm.size();
    constexpr int mod = 998244353;
    std::vector<int> bit(n + 1);
    std::vector<char> used(n + 1);
    i64 rank = 0, factorial = 1;
    for (int i = n - 1; i >= 0; --i) {
        int x = perm[i];
        if (x < 1 || x > n || used[x])
            throw std::invalid_argument("cantor requires a permutation of 1..n");
        used[x] = true;
        int smaller = 0;
        for (int j = x - 1; j > 0; j -= j & -j)
            smaller += bit[j];
        rank = (rank + smaller * factorial) % mod;
        factorial = factorial * (n - i) % mod;
        for (int j = x; j <= n; j += j & -j)
            ++bit[j];
    }
    return int((rank + 1) % mod);
}
```

## 容斥：不被指定数整除的个数

忽略非正除数、去掉重复；出现 1 返回 0，limit<=0 返回 0。去重后大于 1 的除数数目 k<31，2^k 必须可承受。

O(2^k*k) 时间，O(k) 空间。

```cpp
// 返回 [1,limit] 中不被任何 divisor 整除的整数个数。
i64 countNotDivisible(i64 limit, const std::vector<i64> &divisors) {
    if (limit <= 0) {
        return 0;
    }
    for (i64 d : divisors) {
        if (d == 1) {
            return 0;
        }
    }
    std::vector<i64> ds;
    for (i64 d : divisors) {
        if (d > 1) {
            ds.push_back(d);
        }
    }
    std::sort(ds.begin(), ds.end());
    ds.erase(std::unique(ds.begin(), ds.end()), ds.end());
    int k = (int)ds.size();
    i128 sub = 0;
    for (int mask = 1; mask < (1 << k); mask++) {
        i64 l = 1;
        int bits = 0;
        for (int i = 0; i < k; i++) {
            if (mask >> i & 1) {
                i128 next = (i128)(l / std::gcd(l, ds[i])) * ds[i];
                if (next > limit) {
                    l = 0;
                    break;
                } // 先判断再收窄，避免溢出
                l = (i64)next;
                bits++;
                if (l == 0 || l > limit) {
                    break;
                }
            }
        }
        if (l == 0 || l > limit) {
            continue;
        }
        sub += (bits & 1) ? (limit / l) : -(limit / l);
    }
    return (i64)((i128)limit - sub);
}
```

## 整数分拆 DP

n、kMax 非负，计数及中间累加须能放进 i64；n=0 有一种空分拆。partitionExact 按份数返回，partitionAtMost 的第 s 项是总和为 s 的方案数。

最坏 O(n*kMax) 时间，O(n+kMax) 空间。

```cpp
// 返回恰好 k 个正整数无序分拆的方案数。
std::vector<i64> partitionExact(int n, int kMax) {
    assert(n >= 0 && kMax >= 0);
    kMax = std::min(kMax, n);
    std::vector<i64> result(kMax + 2, 0), dp(n + 1, 0);
    result[0] = (n == 0);
    dp[0] = 1;
    for (int k = 1; k <= kMax; k++) {
        for (int s = k; s <= n; s++) {
            dp[s] += dp[s - k];
        }
        result[k] = dp[n - k];
    }
    return result;
}

// 返回每份不超过 kMax 的无序分拆方案数。
std::vector<i64> partitionAtMost(int n, int kMax) {
    std::vector<i64> f(n + 1, 0);
    f[0] = 1;
    for (int piece = 1; piece <= kMax; piece++) {
        for (int s = piece; s <= n; s++) {
            f[s] += f[s - piece];
        }
    }
    return f;
}
```
