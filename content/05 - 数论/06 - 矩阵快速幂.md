# 矩阵快速幂

系数为 Z，默认模 P=1000000007；矩阵相加、相乘要求行数、列数相同；向量长度等于阶数；k>=0。 下标 0..n-1；Matrix(n) 全零，Matrix(n,v) 的对角线为 v；幂为 0 返回单位矩阵。 A*vec 的实际计算是行向量 vec 左乘 A：result[j]=sum_i vec[i]*A.a[i][j]。

乘法/加法 O(n^3)/O(n^2)，幂运算 O(n^3 log k)，空间 O(n^2)。

```cpp
#ifndef XCPC_MODINT
#define XCPC_MODINT

// 返回取模类型的快速幂。
template <class T> constexpr T power(T a, i64 b) {
    T res = 1;
    for (; b; b /= 2, a *= a) {
        if (b % 2) {
            res *= a;
        }
    }
    return res;
}

// 返回 a*b mod p 的原始余数。
constexpr i64 mul(i64 a, i64 b, i64 p) {
    return i64(i128(a) * b % p);
}

// 大模数取模类，模数可编译期固定或运行时设置。
template <i64 P> struct MLong {
    i64 x;

    // 构造值为 0 的取模数。
    constexpr MLong() : x{} {}
    // 用整数构造并规范到 [0,mod)。
    constexpr MLong(i64 x) : x{norm(x % getMod())} {}

    // 动态模数存储。
    static i64 Mod;
    // 返回当前模数。
    constexpr static i64 getMod() {
        return P > 0 ? P : Mod;
    }
    // 设置动态模数，仅 P=0 时生效。
    constexpr static void setMod(i64 Mod_) {
        Mod = Mod_;
    }
    // 只加减一次规范到 [0,mod)；输入须在 [-mod,2*mod) 内。
    constexpr i64 norm(i64 x) const {
        if (x < 0) {
            x += getMod();
        }
        if (x >= getMod()) {
            x -= getMod();
        }
        return x;
    }
    // 返回底层值。
    constexpr i64 val() const {
        return x;
    }
    // 转换为 i64。
    explicit constexpr operator i64() const {
        return x;
    }
    // 返回相反数。
    constexpr MLong operator-() const {
        MLong res;
        res.x = norm(getMod() - x);
        return res;
    }
    // 返回模逆元，要求模数为素数且自身非零。
    constexpr MLong inv() const {
        assert(x != 0);
        return power(*this, getMod() - 2);
    }
    // 乘并赋值。
    constexpr MLong &operator*=(MLong rhs) & {
        x = mul(x, rhs.x, getMod());
        return *this;
    }
    // 加并赋值。
    constexpr MLong &operator+=(MLong rhs) & {
        x = i64((i128(x) + rhs.x) % getMod());
        return *this;
    }
    // 减并赋值。
    constexpr MLong &operator-=(MLong rhs) & {
        x = norm(x - rhs.x);
        return *this;
    }
    // 除并赋值。
    constexpr MLong &operator/=(MLong rhs) & {
        return *this *= rhs.inv();
    }
    // 返回两数乘积。
    friend constexpr MLong operator*(MLong lhs, MLong rhs) {
        MLong res = lhs;
        res *= rhs;
        return res;
    }
    // 返回两数之和。
    friend constexpr MLong operator+(MLong lhs, MLong rhs) {
        MLong res = lhs;
        res += rhs;
        return res;
    }
    // 返回两数之差。
    friend constexpr MLong operator-(MLong lhs, MLong rhs) {
        MLong res = lhs;
        res -= rhs;
        return res;
    }
    // 返回两数之商。
    friend constexpr MLong operator/(MLong lhs, MLong rhs) {
        MLong res = lhs;
        res /= rhs;
        return res;
    }
    // 从流中读入一个取模数。
    friend constexpr std::istream &operator>>(std::istream &is, MLong &a) {
        i64 v;
        if (is >> v)
            a = MLong(v);
        return is;
    }
    // 向流中输出规范值。
    friend constexpr std::ostream &operator<<(std::ostream &os, const MLong &a) {
        return os << a.val();
    }
    // 按规范值判断相等。
    friend constexpr bool operator==(MLong lhs, MLong rhs) {
        return lhs.val() == rhs.val();
    }
    // 按规范值判断不等。
    friend constexpr bool operator!=(MLong lhs, MLong rhs) {
        return lhs.val() != rhs.val();
    }
};

template <> i64 MLong<0LL>::Mod = 1000000007LL;

// 小模数取模类，模数可编译期固定或运行时设置。
template <int P> struct MInt {
    int x;

    // 构造值为 0 的取模数。
    constexpr MInt() : x{} {}
    // 用整数构造并规范到 [0,mod)。
    constexpr MInt(i64 x) : x{norm(int(x % getMod()))} {}

    // 动态模数存储。
    static int Mod;
    // 返回当前模数。
    constexpr static int getMod() {
        return P > 0 ? P : Mod;
    }
    // 设置动态模数，仅 P=0 时生效。
    constexpr static void setMod(int Mod_) {
        Mod = Mod_;
    }
    // 只加减一次规范到 [0,mod)；输入须在 [-mod,2*mod) 内。
    constexpr int norm(int x) const {
        if (x < 0) {
            x += getMod();
        }
        if (x >= getMod()) {
            x -= getMod();
        }
        return x;
    }
    // 返回底层值。
    constexpr int val() const {
        return x;
    }
    // 转换为 int。
    explicit constexpr operator int() const {
        return x;
    }
    // 返回相反数。
    constexpr MInt operator-() const {
        MInt res;
        res.x = norm(getMod() - x);
        return res;
    }
    // 返回模逆元，要求模数为素数且自身非零。
    constexpr MInt inv() const {
        assert(x != 0);
        return power(*this, getMod() - 2);
    }
    // 乘并赋值。
    constexpr MInt &operator*=(MInt rhs) & {
        x = int(1LL * x * rhs.x % getMod());
        return *this;
    }
    // 加并赋值。
    constexpr MInt &operator+=(MInt rhs) & {
        x = int((i64(x) + rhs.x) % getMod());

    constexpr MInt &operator-=(MInt rhs) & {

    // 除并赋值。

    }
    // 返回两数乘积。
    friend constexpr MInt operator*(MInt lhs, MInt rhs) {
        MInt res = lhs;
        res *= rhs;
        return res;
    }
    // 返回两数之和。
    friend constexpr MInt operator+(MInt lhs, MInt rhs) {
        MInt res = lhs;
        res += rhs;
        return res;
    }
    // 返回两数之差。
    friend constexpr MInt operator-(MInt lhs, MInt rhs) {
        MInt res = lhs;
        res -= rhs;
        return res;
    }
    // 返回两数之商。
    friend constexpr MInt operator/(MInt lhs, MInt rhs) {
        MInt res = lhs;
        res /= rhs;
        return res;
    }
    // 从流中读入一个取模数。
    friend constexpr std::istream &operator>>(std::istream &is, MInt &a) {
        i64 v;
        if (is >> v)
            a = MInt(v);
        return is;
    }
    // 向流中输出规范值。
    friend constexpr std::ostream &operator<<(std::ostream &os, const MInt &a) {

    // 按规范值判断相等。
    friend constexpr bool operator==(MInt lhs, MInt rhs) {
        return lhs.val() == rhs.val();

template <> int MInt<0>::Mod = 1000000007;

template <int V, int P> constexpr MInt<P> CInv = MInt<P>(V).inv();

// 库级默认模数。
#ifndef XCPC_MODINT_P
#define XCPC_MODINT_P
constexpr int P = 1000000007;
#endif
using Z = MInt<P>;

#endif // XCPC_MODINT

// 模素数下的方阵，支持乘法、加法与快速幂。
struct Matrix {
    int n;
    std::vector<std::vector<Z>> a; // a[row][col] 为系数，行、列下标都从 0 开始

    // 构造 0 阶矩阵。
    Matrix() : n(0) {}
    // 构造 n*n 零矩阵。
    explicit Matrix(int n_) : n(n_), a(n_, std::vector<Z>(n_)) {}
    // 构造对角线上全为 v 的 n*n 矩阵。
    Matrix(int n_, Z v) : Matrix(n_) {
        for (int i = 0; i < n; i++) {
            a[i][i] = v;
        }
    }

    // 原地乘矩阵并返回自身。
    Matrix &operator*=(const Matrix &rhs) {
        Matrix c(n);
        for (int i = 0; i < n; i++) {
            for (int k = 0; k < n; k++) {
                if (a[i][k] == 0) {
                    continue;
                }
                for (int j = 0; j < n; j++) {
                    c.a[i][j] += a[i][k] * rhs.a[k][j];
                }
            }
        }
        return *this = c;
    }

    // 原地加矩阵并返回自身。
    Matrix &operator+=(const Matrix &rhs) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                a[i][j] += rhs.a[i][j];
            }
        }
        return *this;
    }

    // 返回矩阵乘积。
    friend Matrix operator*(Matrix lhs, const Matrix &rhs) {
        return lhs *= rhs;
    }
    // 返回矩阵之和。
    friend Matrix operator+(Matrix lhs, const Matrix &rhs) {
        return lhs += rhs;
    }

    // 返回矩阵的 b 次幂。
    friend Matrix operator^(Matrix a, i64 b) {
        Matrix res(a.n, 1);
        for (; b; b /= 2, a *= a) {
            if (b % 2) {
                res *= a;
            }
        }
        return res;
    }

    // 判断阶数与元素是否都相等。
    friend bool operator==(const Matrix &lhs, const Matrix &rhs) {
        return lhs.n == rhs.n && lhs.a == rhs.a;
    }
    // 判断两个矩阵是否不同。
    friend bool operator!=(const Matrix &lhs, const Matrix &rhs) {
        return !(lhs == rhs);
    }

    // 输出矩阵的每一行。
    friend std::ostream &operator<<(std::ostream &os, const Matrix &mat) {
        for (int i = 0; i < mat.n; i++) {
            os << '[';
            for (int j = 0; j < mat.n; j++) {
                os << mat.a[i][j].val() << (j + 1 == mat.n ? "" : " ");
            }
            os << "]\n";
        }
        return os;
    }

    // 返回行向量 v 左乘本矩阵，result[j]=sum_i v[i]*a[i][j]。
    std::vector<Z> operator*(const std::vector<Z> &v) const {
        std::vector<Z> res(n);
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                res[j] += v[i] * a[i][j];
            }
        }
        return res;
    }
};
```
