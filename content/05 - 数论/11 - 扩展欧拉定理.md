# 扩展欧拉定理

a>=0、m>=1；指数为非负 i64 或非空十进制数字串；不要求 a 与 m 互素。 字符串允许前导 0，调用者保证全为数字；m=1 返回 0，指数为 0 时返回 1%m。 i64 模乘、10*phi(m)、2*phi(m) 和试除中的 p*p 须可放 i64。

O(|b| + sqrt m)。

```cpp
// 返回正整数的欧拉函数值。
i64 phiOf(i64 n) {
    i64 res = n;
    for (i64 p = 2; p * p <= n; p++) {
        if (n % p == 0) {
            res = res / p * (p - 1);
            while (n % p == 0) {
                n /= p;
            }
        }
    }
    if (n > 1) {
        res = res / n * (n - 1);
    }
    return res;
}

// 返回 a^b mod m，b 为十进制字符串。
i64 powerBig(i64 a, const std::string &b, i64 m) {
    if (m == 1) {
        return 0;
    }
    i64 phi = phiOf(m);
    i64 e = 0;
    bool big = false;
    for (char ch : b) {
        e = e * 10 + ch - '0';
        if (e >= phi) {
            big = true;
            e %= phi;
        }
    }
    if (big) {
        e += phi;
    }
    a %= m;
    i64 res = 1;
    while (e) {
        if (e & 1) {
            res = res * a % m;
        }
        a = a * a % m;
        e >>= 1;
    }
    return res;
}

// 返回 a^b mod m，b 为 i64。
i64 powerBig(i64 a, i64 b, i64 m) {
    if (m == 1) {
        return 0;
    }
    i64 phi = phiOf(m);
    i64 e = b >= phi ? b % phi + phi : b;
    a %= m;
    i64 res = 1;
    while (e) {
        if (e & 1) {
            res = res * a % m;
        }
        a = a * a % m;
        e >>= 1;
    }
    return res;
}
```
