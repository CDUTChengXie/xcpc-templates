# 动态树上最大权独立集

树上选一些互不相邻的点，使权值和最大，而且点权会修改。先不考虑修改：若选 `u`，每个孩子都只能不选；若不选 `u`，每个孩子可以独立取两种状态的较大值。两个状态算到根，较大者就是答案，也允许什么都不选。

难在改一个点后，逐个重算祖先在长链上很慢。把每个点的轻儿子贡献先加好，只留下重儿子的两个状态作输入。沿重链，每个点就都是一次「加权再取最大值」的两状态转移，可以用矩阵合并，在线段树中维护整条链。修改时只把链顶答案的差额传给上一条链，不必重扫兄弟子树。

先构造 `DynamicTreeIndependentSet(parent, weight)`；`update(u, value)` 改点权，`answer()` 查询。节点从 0 编号，parent、weight 等长，parent 为非空树且 parent[0]=-1。每次修改前后权值绝对值总和 < 2^59。

初始化 O(n)，单次修改 O(log² n)，查询 O(log n)，空间 O(n)。

```cpp
struct DynamicTreeIndependentSet {
    static constexpr i64 LOW = -(1LL << 60);
    struct Matrix {
        i64 a[2][2] = {{0, LOW}, {LOW, 0}};
    };
    std::vector<int> parent, heavy, head, pos, finish, size;
    std::vector<i64> weight, g0, g1;
    std::vector<Matrix> seg;
    int n = 0, base = 1;

    // parent[0]=-1；其余 parent[i] 指定父节点，构造带权树
    DynamicTreeIndependentSet(const std::vector<int> &par, const std::vector<i64> &w) {
        init(par, w);
    }
    // 按顺序合并两个 max-plus 转移矩阵
    static Matrix multiply(const Matrix &x, const Matrix &y) {
        Matrix z;
        for (int i = 0; i < 2; ++i)
            for (int j = 0; j < 2; ++j) {
                z.a[i][j] = LOW;
                for (int k = 0; k < 2; ++k)
                    if (x.a[i][k] != LOW && y.a[k][j] != LOW)
                        z.a[i][j] = std::max(z.a[i][j], x.a[i][k] + y.a[k][j]);
            }
        return z;
    }
    // 将节点 u 的轻子树贡献编码为转移矩阵
    Matrix node(int u) const {
        Matrix m;
        m.a[0][0] = m.a[0][1] = g0[u];
        m.a[1][0] = g1[u];
        m.a[1][1] = LOW;
        return m;
    }
    // 将 DFS 位置 p 的转移矩阵替换为 m，并更新线段树
    void setMatrix(int p, const Matrix &m) {
        p += base;
        seg[p] = m;
        for (p >>= 1; p; p >>= 1)
            seg[p] = multiply(seg[p * 2], seg[p * 2 + 1]);
    }
    // 返回 DFS 半开区间 [l,r) 内按顺序相乘的转移矩阵
    Matrix product(int l, int r) const {
        Matrix left, right;
        for (l += base, r += base; l < r; l >>= 1, r >>= 1) {
            if (l & 1)
                left = multiply(left, seg[l++]);
            if (r & 1)
                right = multiply(seg[--r], right);
        }
        return multiply(left, right);
    }
    // h 须为重链头；返回该子树根 h 不选/选时的最优权值，分别读取 [0]/[1]
    std::array<i64, 2> chain(int h) const {
        auto m = product(pos[h], finish[h]);
        return {m.a[0][0], m.a[1][0]};
    }
    // 重建整棵树并清除之前的修改，无返回值
    void init(const std::vector<int> &par, const std::vector<i64> &w) {
        n = int(par.size());
        assert(n > 0 && w.size() == par.size() && par[0] == -1);
        parent = par;
        weight = w;
        std::vector<std::vector<int>> child(n);
        for (int u = 1; u < n; ++u) {
            assert(parent[u] >= 0 && parent[u] < n && parent[u] != u);
            child[parent[u]].push_back(u);
        }
        std::vector<int> order{0};
        for (size_t i = 0; i < order.size(); ++i)
            for (int v : child[order[i]])
                order.push_back(v);
        assert(int(order.size()) == n);
        heavy.assign(n, -1);
        size.assign(n, 1);
        g0.assign(n, 0);
        g1 = weight;
        std::vector<std::array<i64, 2>> dp(n);
        for (int i = n - 1; i >= 0; --i) {
            int u = order[i];
            dp[u] = {0, weight[u]};
            for (int v : child[u]) {
                size[u] += size[v];
                if (heavy[u] == -1 || size[v] > size[heavy[u]])
                    heavy[u] = v;
                dp[u][0] += std::max(dp[v][0], dp[v][1]);
                dp[u][1] += dp[v][0];
            }
            for (int v : child[u])
                if (v != heavy[u]) {
                    g0[u] += std::max(dp[v][0], dp[v][1]);
                    g1[u] += dp[v][0];
                }
        }
        head.resize(n);
        pos.resize(n);
        finish.resize(n);
        int clock = 0;
        std::vector<int> starts{0};
        while (!starts.empty()) {
            int h = starts.back();
            starts.pop_back();
            for (int u = h; u != -1; u = heavy[u]) {
                head[u] = h;
                pos[u] = clock++;
                for (int v : child[u])
                    if (v != heavy[u])
                        starts.push_back(v);
            }
            finish[h] = clock;
        }
        base = 1;
        while (base < n)
            base <<= 1;
        seg.assign(base * 2, Matrix{});
        for (int u = 0; u < n; ++u)
            seg[base + pos[u]] = node(u);
        for (int p = base - 1; p > 0; --p)
            seg[p] = multiply(seg[p * 2], seg[p * 2 + 1]);
    }
    // 将节点 u 的权值改成 value；返回修改后的最大独立集权值和
    i64 update(int u, i64 value) {
        assert(u >= 0 && u < n);
        g1[u] += value - weight[u];
        weight[u] = value;
        while (u != -1) {
            int h = head[u];
            auto before = chain(h);
            setMatrix(pos[u], node(u));
            auto after = chain(h);
            int p = parent[h];
            if (p != -1) {
                g0[p] += std::max(after[0], after[1]) - std::max(before[0], before[1]);
                g1[p] += after[0] - before[0];
            }
            u = p;
        }
        return answer();
    }
    // 返回当前最大权值和；相邻节点不能同时选，可以一个也不选
    i64 answer() const {
        auto root = chain(0);
        return std::max(root[0], root[1]);
    }
};
```
