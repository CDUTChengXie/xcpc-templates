# 三分与黄金分割搜索

f(x) 为题目函数；求最小要求先降后升，求最大要求先升后降；浮点求值要稳定。 搜索区间均为闭区间；三分用于严格单谷，平台区间不保证最靠左；浮点 eps>0。 ternaryInt 并列取最终候选范围中最左位置；goldenSearch 返回位置和函数值。

实数接口端点须有限，坐标量级建议不超过 1e18；ternarySearchMin / ternarySearchMax 分别求最小、最大值位置。

浮点 O(log((r - l) / eps))；整数 O(log n) 后扫边界附近，乘以单次 eval 代价。

```cpp
// 实数三分：严格单谷函数的最小值位置
double ternaryFloat(double l, double r, const std::function<double(double)> &f,
                    double eps = 1e-12) {
    if (!std::isfinite(l) || !std::isfinite(r) || l > r || !std::isfinite(eps) || eps <= 0)
        throw std::invalid_argument("invalid range or eps");
    while (r - l > eps) {
        double m1 = std::lerp(l, r, 1.0 / 3);
        double m2 = std::lerp(l, r, 2.0 / 3);
        if (m1 <= l || m2 >= r || m1 >= m2)
            break;
        if (f(m1) > f(m2)) {
            l = m1;
        } else {
            r = m2;
        }
    }
    return std::midpoint(l, r);
}

// 整数三分：离散严格单谷序列的最小值位置
template <class F> int ternaryInt(int l, int r, F &&f) {
    if (l > r)
        throw std::invalid_argument("invalid range");
    while (i64(r) - l > 2) {
        int m1 = int(l + (i64(r) - l) / 3);
        int m2 = int(r - (i64(r) - l) / 3);
        if (f(m1) > f(m2)) {
            l = m1;
        } else {
            r = m2;
        }
    }
    int best = l;
    for (i64 i = i64(l) + 1; i <= r; i++) {
        if (f(i) < f(best)) {
            best = int(i);
        }
    }
    return best;
}

// 黄金分割求单峰/单谷极值；返回 {极值位置,函数值}，每轮仅新增一次f求值
template <class F>
std::pair<double, double> goldenSearch(double l, double r, F f, bool maximum = false,
                                       int iterations = 100) {
    if (!std::isfinite(l) || !std::isfinite(r) || l > r || iterations < 1)
        throw std::invalid_argument("invalid search range");
    constexpr double ratio = 0.6180339887498948482;
    double x = std::lerp(l, r, 1 - ratio), y = std::lerp(l, r, ratio), fx = f(x), fy = f(y);
    for (int i = 0; i < iterations && l < r; ++i) {
        if (x <= l || y >= r || x >= y)
            break;
        if (maximum ? fx < fy : fx > fy) {
            l = x;
            x = y;
            fx = fy;
            y = std::lerp(l, r, ratio);
            fy = f(y);
        } else {
            r = y;
            y = x;
            fy = fx;
            x = std::lerp(l, r, 1 - ratio);
            fx = f(x);
        }
    }
    std::pair<double, double> answer{x, fx};
    auto consider = [&](double p, double value) {
        if (maximum ? value > answer.second : value < answer.second)
            answer = {p, value};
    };
    consider(y, fy);
    consider(l, f(l));
    consider(r, f(r));
    return answer;
}

// 实数三分求严格单谷函数的最小值位置
inline double ternarySearchMin(double l, double r, double eps,
                               const std::function<double(double)> &f) {
    while (r - l > eps) {
        double m1 = l + (r - l) / 3;
        double m2 = r - (r - l) / 3;
        if (m1 <= l || m2 >= r || m1 >= m2)
            break;
        if (f(m1) < f(m2)) {
            r = m2;
        } else {
            l = m1;
        }
    }
    return (l + r) / 2;
}

// 实数三分求严格单峰函数的最大值位置
inline double ternarySearchMax(double l, double r, double eps,
                               const std::function<double(double)> &f) {
    while (r - l > eps) {
        double m1 = l + (r - l) / 3;
        double m2 = r - (r - l) / 3;
        if (m1 <= l || m2 >= r || m1 >= m2)
            break;
        if (f(m1) > f(m2)) {
            r = m2;
        } else {
            l = m1;
        }
    }
    return (l + r) / 2;
}
```
