# 整块整数快读写

只支持按空白分隔的整数（不支持大整数、分数、科学计数法），且不能与 cin/cout 混用。 io 是约 64MB 的全局对象，输入最多 32MiB；不要在栈上或函数内声明 FastIO 局部变量。 纯 I/O，不含算法。readI64/readU64 到输入末尾返回 false 且不修改目标值；readI64 还拒绝孤立负号与越界值。

输出缓冲满时自动 flush，析构时 flush 剩余内容；需要立即输出请显式 flush()。

每次读 O(该数的位数)，总 O(输入长度)；空间 2 * 32MB 常驻。

```cpp
struct FastIO {
    static constexpr int SZ = 1 << 25;

    char in[SZ + 1], out[SZ];
    char *s = in, *ein = in;         // s 当前读指针，ein 指向输入有效末尾
    char *t = out, *eout = out + SZ; // t 当前写位置，eout 指向输出缓冲末尾

    // 懒加载：首次读取才 fread，避免无输入环境阻塞
    bool loaded = false;

    // 首次调用时一次性读入全部输入
    void ensureLoaded() {
        if (loaded) {
            return;
        }
        size_t len = fread(in, 1, SZ, stdin);
        in[len] = 0; // SZ+1 保证读满时这行仍合法
        ein = in + len;
        loaded = true;
    }

    // 构造懒加载
    FastIO() {}
    // 析构时刷新输出缓冲
    ~FastIO() {
        flush();
    }

    // 立即写出输出缓冲
    void flush() {
        if (t != out) {
            fwrite(out, 1, t - out, stdout);
            t = out;
        }
    }
    // 写入一个字符；缓冲满时先 flush
    void put(char c) {
        if (t == eout) {
            flush();
        }
        *t++ = c;
    }

    // 跳过非数字（含 '+'）；到达输入末尾返回 false
    bool readU64(u64 &x) {
        ensureLoaded();
        while (s < ein && (*s < '0' || *s > '9')) {
            ++s;
        }
        if (s == ein) {
            return false;
        }
        x = 0;
        while (s < ein && *s >= '0' && *s <= '9') {
            x = x * 10 + u64(*s++ - '0');
        }
        return true;
    }

    // 额外识别 '-'；到达输入末尾返回 false
    bool readI64(i64 &x) {
        ensureLoaded();
        while (s < ein && *s != '-' && (*s < '0' || *s > '9')) {
            ++s;
        }
        if (s == ein) {
            return false;
        }
        bool neg = (*s == '-');
        if (neg) {
            ++s;
        }
        if (s == ein || *s < '0' || *s > '9') {
            return false;
        }
        u64 v = 0, limit = u64(LLONG_MAX) + neg;
        bool overflow = false;
        while (s < ein && *s >= '0' && *s <= '9') {
            u64 digit = u64(*s++ - '0');
            if (v > (limit - digit) / 10) {
                overflow = true;
            }
            if (!overflow) {
                v = v * 10 + digit;
            }
        }
        if (overflow) {
            return false;
        }
        x = neg && v ? -i64(v - 1) - 1 : i64(v);
        return true;
    }

    // 输出无符号十进制整数
    void writeU64(u64 x) {
        ensureLoaded();
        char c[24], *i = c + 24;
        do {
            *--i = char('0' + x % 10);
            x /= 10;
        } while (x);
        while (i != c + 24) {
            put(*i++);
        }
    }
    // 输出有符号十进制整数，支持最小负数
    void writeI64(i64 x) {
        ensureLoaded();
        if (x < 0) {
            put('-');
            writeU64(u64(-(x + 1)) + 1); // 避开 LLONG_MIN 直接取反的 UB
        } else {
            writeU64(u64(x));
        }
    }

    // 读入 u32；到输入末尾时保持 x 不变
    FastIO &operator>>(u32 &x) {
        u64 v = 0;
        if (readU64(v)) {
            x = u32(v);
        }
        return *this;
    }
    // 读入 u64；到输入末尾时保持 x 不变
    FastIO &operator>>(u64 &x) {
        readU64(x);
        return *this;
    }
    // 读入 int；到输入末尾时保持 x 不变
    FastIO &operator>>(int &x) {
        i64 v = 0;
        if (readI64(v)) {
            x = int(v);
        }
        return *this;
    }
    // 读入 i64；到输入末尾时保持 x 不变
    FastIO &operator>>(i64 &x) {
        readI64(x);
        return *this;
    }

    // 输出一个字符
    FastIO &operator<<(char x) {
        put(x);
        return *this;
    }
    // 输出 C 风格字符串
    FastIO &operator<<(const char *str) {
        while (*str) {
            put(*str++);
        }
        return *this;
    }
    // 输出 u32
    FastIO &operator<<(u32 x) {
        writeU64(u64(x));
        return *this;
    }
    // 输出 u64
    FastIO &operator<<(u64 x) {
        writeU64(x);
        return *this;
    }
    // 输出 int
    FastIO &operator<<(int x) {
        writeI64(i64(x));
        return *this;
    }
    // 输出 i64
    FastIO &operator<<(i64 x) {
        writeI64(x);
        return *this;
    }
} io;
```
