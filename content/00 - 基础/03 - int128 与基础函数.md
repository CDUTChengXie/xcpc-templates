# 128 位整数输入输出

GNU __int128；依赖统一主模板的 i128。允许首个正负号，数值须在有符号 128 位范围内。

toi128 非法或越界时抛异常；WideTools::read 失败时返回 false、设置 failbit 且不改目标值；输出支持最小负数。

读写与转换 O(十进制位数)，额外空间 O(位数)。

```cpp
using u128 = unsigned __int128;

// __int128 输出流（i128 没有标准 IO 重载）
std::ostream &operator<<(std::ostream &os, i128 n) {
    if (n == 0) {
        return os << '0';
    }
    bool neg = n < 0;
    // 在无符号域取绝对值，避免对最小负数取反
    u128 u = neg ? u128(-(n + 1)) + 1 : u128(n);
    char buf[45];
    int len = 0;
    while (u) {
        buf[len++] = char('0' + int(u % 10));
        u /= 10;
    }
    if (neg) {
        os << '-';
    }
    while (len) {
        os << buf[--len];
    }
    return os;
}

// 字符串转 i128，支持可选正负号；超出 i128 范围抛 out_of_range
i128 toi128(const std::string &s) {
    int i = 0;
    bool neg = false;
    if (i < int(s.size()) && (s[i] == '-' || s[i] == '+')) {
        neg = (s[i] == '-');
        ++i;
    }
    u128 n = 0;
    const u128 limit = (u128(1) << 127) - (neg ? 0 : 1);
    for (; i < int(s.size()); ++i) {
        if (s[i] < '0' || s[i] > '9') {
            break;
        }
        unsigned digit = s[i] - '0';
        if (n > (limit - digit) / 10) {
            throw std::out_of_range("toi128");
        }
        n = n * 10 + digit;
    }
    return neg && n ? -i128(n - 1) - 1 : i128(n);
}

namespace WideTools {
    using U = unsigned __int128;
// 将 token 严格读为 i128；失败设置 failbit，保留原 value
    bool read(std::istream &in, i128 &value) {
        std::string s;
        if (!(in >> s))
            return false;
        size_t p = 0;
        bool negative = s[0] == '-';
        if (s[0] == '+' || s[0] == '-')
            ++p;
        U x = 0, bound = (U(1) << 127) - (negative ? 0 : 1);
        bool ok = p < s.size();
        for (; p < s.size() && ok; ++p) {
            if (s[p] < '0' || s[p] > '9') {
                ok = false;
                break;
            }
            unsigned digit = s[p] - '0';
            if (x > (bound - digit) / 10) {
                ok = false;
                break;
            }
            x = x * 10 + digit;
        }
        if (!ok) {
            in.setstate(std::ios::failbit);
            return false;
        }
        value = negative && x ? -i128(x - 1) - 1 : i128(x);
        return true;
    }
}
```

## 整数平方根

n<=0 返回 0；isqrt128 用除法比较，避免中间平方溢出。

i64 版 O(1) 浮点估计后校正；i128 版 O(log n)，空间 O(1)。

```cpp
// 精确整数开方：floor(sqrt(n))
i64 isqrt(i64 n) {
    if (n <= 0) {
        return 0;
    }
    i64 s = (i64)std::sqrt((long double)n);
    if (s < 0) {
        s = 0;
    }
    // 用 i128 比较：n 接近 LLONG_MAX 时 (s+1)*(s+1) 会在 i64 下溢出
    while (i128(s) * s > n) {
        --s;
    }
    while (i128(s + 1) * (s + 1) <= n) {
        ++s;
    }
    return s;
}

// 精确整数开方（i128 版）：floor(sqrt(n))
i128 isqrt128(i128 n) {
    if (n <= 0) {
        return 0;
    }
    // sqrt(2^127) < 2^64，故上界取 2^64
    i128 lo = 0, hi = i128(1) << 64;
    while (lo < hi) {
        i128 mid = lo + (hi - lo + 1) / 2; // hi - lo + 1 不会溢出 i128
        // 用除法代替 mid * mid：mid 接近 2^64 时平方会溢出 i128
        if (mid <= n / mid) {
            lo = mid;
        } else {
            hi = mid - 1;
        }
    }
    return lo;
}
```

## 整数除法向上与向下取整

第二个参数 m 必须大于 0；返回值为 i64。

时间与空间均 O(1)。

```cpp
// 负数也安全的向上取整（m > 0）
i64 ceilDiv(i64 n, i64 m) {
    i64 q = n / m;
    if (n % m != 0 && ((n < 0) == (m < 0))) {
        ++q;
    }
    return q;
}

// 负数也安全的向下取整（m > 0）
i64 floorDiv(i64 n, i64 m) {
    i64 q = n / m;
    if (n % m != 0 && ((n < 0) != (m < 0))) {
        --q;
    }
    return q;
}
```

## 最值更新（chmax / chmin）

按引用修改第一个参数，无返回值；T 须支持比较和赋值。

时间与空间均 O(1)。

```cpp
// 取最大赋值
template <class T> void chmax(T &a, T b) {
    if (a < b) {
        a = b;
    }
}

// 取最小赋值
template <class T> void chmin(T &a, T b) {
    if (a > b) {
        a = b;
    }
}
```

## 整数向上取对数（i64）

b<=1 返回 0；a<=1 且 b>1 返回 -1；不通过溢出的乘法判断边界。

O(log_a b) 时间，O(1) 空间。

```cpp
// 求满足 a^k >= b 的最小 k（a >= 1, b >= 1）
// b <= 1 返回 0；a <= 1 且 b > 1 时返回 -1
int logCeil(i64 a, i64 b) {
    if (b <= 1) {
        return 0;
    }
    if (a <= 1) {
        return -1; // a == 1 时 a^k 恒为 1，到不了 b > 1
    }
    int t = 0;
    i64 v = 1;
    while (v < b) {
        // 下一次乘法就会 >= b，或乘法将溢出：两种情况都直接收尾，避免溢出成负数
        if (v > (b - 1) / a) {
            return t + 1;
        }
        v *= a;
        ++t;
    }
    return t;
}
```

## 128 位最大公约数

返回无符号 128 位；gcd(0,0)=0，magnitude 返回无符号绝对值。

O(log V) 时间，O(1) 空间。

```cpp
namespace WideTools {
    using U = unsigned __int128;
// 返回任意 i128 的无符号绝对值，支持最小负数
    U magnitude(i128 x) {
        return x < 0 ? U(-(x + 1)) + 1 : U(x);
    }
    // 返回非负 gcd，最小负数参与时可能需要无符号范围
    U gcd(i128 a, i128 b) {
        U x = magnitude(a), y = magnitude(b);
        while (y) {
            U r = x % y;
            x = y;
            y = r;
        }
        return x;
    }
}
```
